Each original row contains $6$ squares. After Neil removes two rows, the column height is $9$, so the original bar had $11$ rows. Jack leaves a $9\times5$ rectangle, containing $\boxed{45}$ squares.
\begin{asy}
usepackage("amsmath");
size(7cm);
for(int i=0;i<=6;++i)draw((i,0)--(i,11),gray);for(int j=0;j<=11;++j)draw((0,j)--(6,j),gray);fill((0,9)--(6,9)--(6,11)--(0,11)--cycle,lightgray+opacity(.7));fill((5,0)--(6,0)--(6,9)--(5,9)--cycle,lightblue+opacity(.7));draw((0,0)--(5,0)--(5,9)--(0,9)--cycle,linewidth(1));label("$5$",(2.5,0),S);label("$9$",(0,4.5),W);
\end{asy}
