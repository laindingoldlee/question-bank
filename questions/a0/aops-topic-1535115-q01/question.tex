\begin{enumerate}
\item[(a)] How many ordered pairs $(x,y)$ of positive integers satisfy $\dfrac{1}{x}+\dfrac{1}{y}=\dfrac{1}{10}$?
\item[(b)] How many ordered pairs $(x,y)$ of nonzero integers satisfy the same equation?
\end{enumerate}
