For a positive integer $n$ let
\[
S_n=1!\dbinom{n}{1}+2!\dbinom{n}{2}+\dots+n!\dbinom{n}{n}.
\]
\begin{enumerate}
\item[(a)] Show that $S_n=n(n-1)\cdots(n-k+1)$ summed over $k=1,\dots,n$, and that $S_n=n\,(S_{n-1}+1)$ for $n\ge2$.
\item[(b)] Prove that $S_n=\lfloor e\cdot n!\rfloor-1$, where $e=\sum_{j\ge0}\dfrac{1}{j!}$.
\end{enumerate}
