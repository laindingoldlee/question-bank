Take vectors $A=0,B=b,C=c$. Then $D=\dfrac{b+c}{2}$, $E=\dfrac{b}{2}$, and $F=\dfrac{c}{2}$. The midpoint of $EF$ is $\dfrac{b+c}{4}=\dfrac{D}{2}$, so it lies on $AD$ and is its midpoint. Thus $\boxed{AP:PD=1:1}$.
\begin{asy}
size(7cm);
pair A=(1,3),B=(0,0),C=(5,0),D=(B+C)/2,E=(A+B)/2,F=(A+C)/2,P=(A+D)/2;draw(A--B--C--cycle);draw(A--D);draw(E--F);label("$A$",A,N);label("$B$",B,SW);label("$C$",C,SE);label("$D$",D,S);label("$E$",E,W);label("$F$",F,E);label("$P$",P,NE);
\end{asy}
