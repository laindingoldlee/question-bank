\begin{enumerate}
\item Direct substitution gives $f(2,0,3)=2+\dfrac{1}{2}=\boxed{\dfrac{5}{2}}$ and $f(0,2,3)=-2+\dfrac{9}{2}=\boxed{\dfrac{5}{2}}$.
\item At $t=0$, $f(x,y,0)=-\dfrac{x^2+y^2}{2}$, which is symmetric in $x,y$.
\item Putting the two terms over a common denominator gives
\[
f(x,y,t)=-\dfrac{(1+t^2)(x^2+y^2)-4txy}{2(1-t^2)}.
\]
Both the numerator and denominator are unchanged on interchanging $x,y$.
\item Set $u=\sin s\ne1$. Directly combining the terms, without dividing by $1+u$, gives
\[
g(x,y,s)=-\dfrac{(1+u^2)(x^2+y^2)-4uxy}{2(1-u)}.
\]
Again the expression is symmetric. This proof includes $u=-1$, which is permitted in the domain of $g$ even though $f(x,y,-1)$ is undefined.
\end{enumerate}
