For real $x,y$ and $t\ne\pm1$, define
\[
f(x,y,t)=\dfrac{x^2-y^2}{2}-\dfrac{(x-yt)^2}{1-t^2}.
\]
\begin{enumerate}
\item Evaluate $f(2,0,3)$ and $f(0,2,3)$.
\item Show that $f(x,y,0)=f(y,x,0)$.
\item Prove that $f(x,y,t)=f(y,x,t)$ throughout its domain.
\item Define
\[
g(x,y,s)=\dfrac{(x^2-y^2)(1+\sin s)}{2}
-\dfrac{(x-y\sin s)^2}{1-\sin s}
\]
for $s\ne\dfrac{\pi}{2}+2\pi k$, $k\in\mathbb Z$. Prove that $g(x,y,s)=g(y,x,s)$.
\end{enumerate}
