\begin{enumerate}
\item[(a)] Find the point $P$ of the curve $y=x^2$ that is closest to the point $(3,0)$.
\item[(b)] Let $P=(a,b)$ be this point. Show that the line through $(3,0)$ and $(a,b)$ is perpendicular to the tangent line of the curve at $(a,b)$ (that is, it is the normal line at $(a,b)$).
\end{enumerate}
