Include both cylinder bases: $2\pi r^2+2\pi rh=4\pi r^2$. Cancelling gives $h=r=\boxed6$.
\begin{asy}
import graph;import geometry;usepackage("amsmath");size(7cm);
draw(ellipse((0,0),2,.5));draw(ellipse((0,2),2,.5));draw((-2,0)--(-2,2)^^(2,0)--(2,2));label("$r=6,h=6$",(0,-1));draw(circle((6,1),2));draw(ellipse((6,1),2,.5),dashed);label("$r=6$",(6,-1.5));
\end{asy}
