The angle sum gives $8x+12=180$, so $x=21$. The angles are $59^\circ,64^\circ,57^\circ$, hence the largest is $\boxed{64^\circ}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),B=(4,0);real h=4/(1/tan(59*pi/180)+1/tan(64*pi/180));pair C=(h/tan(59*pi/180),h);draw(A--B--C--cycle);label("$59^\circ$",A+(0.7,.25));label("$64^\circ$",B+(-.7,.25));label("$57^\circ$",C+(0,-.5));
\end{asy}
