Since $PA=PB$, angles $PAB$ and $ABP$ are equal. If $\angle B=2t$, the bisector gives $\angle ABP=t$, and hence $\angle A=t$. The triangle angle sum gives $3t+60=180$, so $t=40$. Angle $APB$ is $180-2t=100$, and $PC$ extends $PA$ in the opposite direction. Therefore $\angle CPB=180-100=\boxed{80^\circ}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),B=(4,0),C=extension(A,A+dir(40),B,B+dir(100)),P=extension(A,C,(2,0),(2,1));draw(A--B--C--cycle);draw(B--P);draw((2,-.3)--(2,3),dashed);label("$A$",A,SW);label("$B$",B,SE);label("$C$",C,N);label("$P$",P,NW);
\end{asy}
