If $n=\prod p^{e_p}$, then $F(n)=\prod(e_p+1)$, which is divisible by $3$ exactly when some exponent satisfies $e_p\equiv2\pmod3$. For $n\le999$ this means $e_p\in\{2,5,8\}$ for some prime $p$ (then $p^2\le999$, so $p\le31$).

Let $A_p$ be the set of $n\le999$ with $e_p\in\{2,5,8\}$. Then
\[
|A_p|=\sum_{e\in\{2,5,8\}}\left(\left\lfloor\dfrac{999}{p^e}\right\rfloor-\left\lfloor\dfrac{999}{p^{e+1}}\right\rfloor\right).
\]
\begin{itemize}
\item $p=2$: $(249-124)+(31-15)+(3-1)=143$; \ $p=3$: $(111-37)+(4-1)=77$; \ $p=5$: $39-7=32$; \ $p=7$: $20-2=18$;
\item $p=11,13,17,19,23,29,31$: $8,5,3,2,1,1,1$.
\end{itemize}
These add up to $291$.

\textbf{Pairs.} $n\in A_p\cap A_q$ needs $p^2q^2\le999$, so $pq\le31$:
\begin{itemize}
\item $A_2\cap A_3$: $36m$ with $\gcd(m,6)=1$, $m\le27$ ($9$ numbers), plus $2^5\cdot3^2=288$ and $2^2\cdot3^5=972$: $11$;
\item $A_2\cap A_5$: $100m$ with $m\in\{1,3,7,9\}$, plus $2^5\cdot5^2=800$: $5$;
\item $A_2\cap A_7$: $196m$, $m\in\{1,3,5\}$: $3$; \ $A_2\cap A_{11}$: $484$; \ $A_2\cap A_{13}$: $676$;
\item $A_3\cap A_5$: $225m$, $m\in\{1,2,4\}$: $3$; \ $A_3\cap A_7$: $441m$, $m\in\{1,2\}$: $2$.
\end{itemize}
These add up to $11+5+3+1+1+3+2=26$.

\textbf{Triples.} Only $A_2\cap A_3\cap A_5=\{900\}$.

By inclusion--exclusion the count is $291-26+1=\boxed{266}$.
