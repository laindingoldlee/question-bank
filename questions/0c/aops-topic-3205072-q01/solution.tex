The center follows three semicircles with radii $98,62,78$. Semicircle length is $\pi r$, so the total is $\pi(98+62+78)=\boxed{238\pi\text{ inches}}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
draw(arc((100,0),100,180,360),black);draw(arc((260,0),60,0,180),black);draw(arc((400,0),80,180,360),black);draw(arc((100,0),98,180,360),blue+dashed);draw(arc((260,0),62,0,180),blue+dashed);draw(arc((400,0),78,180,360),blue+dashed);label("$100$",(100,-110),S);label("$60$",(260,70),N);label("$80$",(400,-90),S);
\end{asy}
