\begin{enumerate}
\item[(a)] Prove the inclusion--exclusion principle for three finite sets:
\[
|A\cup B\cup C|=|A|+|B|+|C|-|A\cap B|-|B\cap C|-|C\cap A|+|A\cap B\cap C| .
\]
\item[(b)] More generally, prove that for finite sets $A_1,\dots,A_n$,
\[
|A_1\cup\cdots\cup A_n|=\sum_{k=1}^{n}(-1)^{k+1}\sum_{1\le i_1<\cdots<i_k\le n}|A_{i_1}\cap\cdots\cap A_{i_k}| .
\]
\end{enumerate}
