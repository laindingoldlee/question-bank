Let $f(x)=(x-1)^2+2$ for real $x$, and $g(x)=\dfrac{1}{(x-1)}$ for $x\ne1$.
\begin{enumerate}
\item If $h=g\circ f$, find $h(2)$.
\item Solve $f(x)=3$.
\item Solve $g(x)=-1.5$.
\item Find $\lim_{x\to1^+}g(x)$.
\item Is $f$ invertible on $\mathbb R$?
\item Justify the preceding answer using two explicit points of its graph.
\end{enumerate}
