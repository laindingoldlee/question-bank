Let their coordinates be $X,Y\in[0,\pi]$. The sample space is a square of area $\pi^2$. The conditions $Y-X\ge\pi-1$ and $X-Y\ge\pi-1$ form two right triangles, each with leg length $1$ and area $\dfrac{1}{2}$. Their interiors are disjoint, so the probability is $\boxed{\dfrac{1}{\pi^2}}$.

texpreamble("\usepackage{amsmath,amssymb}");

real p=pi;draw(box((0,0),(p,p)));filldraw((0,p-1)--(0,p)--(1,p)--cycle,lightblue);filldraw((p-1,0)--(p,0)--(p,1)--cycle,lightblue);label("$0$",(0,0),(-1,-1));label("$\pi$",(p,0),(0,-1));label("$\pi$",(0,p),(-1,0));label("$X$",(p/2,-.4));label("$Y$",(-.4,p/2));
\end{asy}
