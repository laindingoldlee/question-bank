Let $ABCD$ be a square with side length $2$, and let $K,L,M,N$ be the midpoints of $AB,BC,CD,DA$. Segments $KC$ and $LA$ meet at $P$; $LD$ and $MB$ meet at $Q$; $MA$ and $NC$ meet at $R$; $NB$ and $KD$ meet at $T$. Find the area of the (star-shaped) octagon $KPLQMRNT$.

\begin{asy}
usepackage("amsmath");
size(5.5cm);
defaultpen(fontsize(10pt));
pair A=(0,0),B=(2,0),C=(2,2),D=(0,2);
pair K=(1,0),L=(2,1),M=(1,2),N=(0,1);
pair P=(4/3,2/3),Q=(4/3,4/3),R=(2/3,4/3),T=(2/3,2/3);
fill(K--P--L--Q--M--R--N--T--cycle,rgb(0.88,0.92,1));
draw(A--B--C--D--cycle,linewidth(0.9));
draw(K--C^^L--A^^L--D^^M--B^^M--A^^N--C^^N--B^^K--D,gray);
draw(K--P--L--Q--M--R--N--T--cycle,blue+linewidth(0.9));
label("$A$",A,SW);label("$B$",B,SE);label("$C$",C,NE);label("$D$",D,NW);
label("$K$",K,S);label("$L$",L,E);label("$M$",M,N);label("$N$",N,W);
label("$P$",P,SE);label("$Q$",Q,NE);label("$R$",R,NW);label("$T$",T,SW);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
