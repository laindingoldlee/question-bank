\begin{enumerate}
\item[(a)] Let $a$ and $b$ be positive numbers different from $1$, and let $x>0$. Prove the change of base formula
\[
\log_a x=\dfrac{\log_b x}{\log_b a}.
\]
\item[(b)] Use it to write $\log_4 x$ in terms of base-$7$ logarithms, and evaluate $\log_4 7\cdot\log_7 16$.
\end{enumerate}
