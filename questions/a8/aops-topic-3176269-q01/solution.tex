The angle sum is $180^\circ$, so the third is $180^\circ-60^\circ-75^\circ=\boxed{45^\circ}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair[] pts={(0,0),(1,0),((sqrt(3)+1)/4,(3+sqrt(3))/4)};string[] names={"$A$","$B$","$C$"};path boundary=pts[0];for(int j=1;j<pts.length;++j)boundary=boundary--pts[j];draw(boundary--cycle);for(int j=0;j<pts.length;++j)label(names[j],pts[j],dir(90+360*j/pts.length));
\end{asy}
