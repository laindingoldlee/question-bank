In triangle $ABC$, point $D$ lies on side $AC$, and $AB=BD=10$, $AD=12$, $DC=9$. Find $BC$.

\begin{asy}
usepackage("amsmath");
size(5cm);
defaultpen(fontsize(10pt));
pair A=(0,0),D=(12,0),C=(21,0),B=(6,8);
draw(A--B--C--cycle,linewidth(0.8));draw(B--D,linewidth(0.8));
dot(A);dot(B);dot(C);dot(D);
label("$A$",A,SW);label("$B$",B,N);label("$C$",C,SE);label("$D$",D,S);
label("$10$",(A+B)/2,NW);label("$10$",(B+D)/2,E);label("$12$",(A+D)/2,S);label("$9$",(D+C)/2,S);
\end{asy}
