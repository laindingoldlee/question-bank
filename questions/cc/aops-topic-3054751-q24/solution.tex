The altitude is $AM=\dfrac{\sqrt3}{2}$, so $MP=\dfrac{\sqrt3}{8}$. Since $BM=\dfrac{1}{2}$ and $\angle BMP=90^\circ$, $BP^2=\dfrac{1}{4}+\dfrac{3}{64}=\dfrac{19}{64}$. Hence $\boxed{BP=\dfrac{\sqrt{19}}{8}}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair B=(-0.5,0),C=(0.5,0),M=(0,0),A=(0,sqrt(3)/2),P=(0,sqrt(3)/8);draw(A--B--C--cycle);draw(A--M);draw(B--P,blue);label("$A$",A,N);label("$B$",B,SW);label("$C$",C,SE);label("$M$",M,S);label("$P$",P,E);
\end{asy}
