$\dfrac{5-y}{7-3}=8$ gives $5-y=32$, so $y=\boxed{-27}$.

\begin{asy}
usepackage("amsmath");
size(3cm);
defaultpen(fontsize(10pt));
real k=0.25;pair P=(3,-27*k),Q=(7,5*k);
draw((-0.5,0)--(8.5,0),gray,Arrow(4));draw((0,-7.5)--(0,2),gray,Arrow(4));
draw(P--Q,linewidth(0.8));dot(P);dot(Q);
label("$(3,-27)$",P,W,fontsize(7pt));label("$(7,5)$",Q,NW,fontsize(7pt));label("slope $8$",(P+Q)/2,E,fontsize(7pt));
\end{asy}
