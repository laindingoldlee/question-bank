A circle of radius $a$ has the same centre as a square of side length $r$, where $\dfrac{r}{2}<a<\dfrac{r\sqrt2}{2}$. Find the area of the region inside the square but outside the circle.

\begin{asy}
usepackage("amsmath");
size(4.5cm);
defaultpen(fontsize(10pt));
real r=4.000, a=2.450;
pair O=(0,0);
path sq=(-r/2,-r/2)--(r/2,-r/2)--(r/2,r/2)--(-r/2,r/2)--cycle;
fill(sq,lightgray); fill(circle(O,a),white);
draw(sq,linewidth(0.9)); draw(circle(O,a),blue);
dot(O); label("$O$",O,SW);
draw(O--a*dir(130),dashed); label("$a$",0.5*a*dir(130),NE);
draw((-r/2,-r/2-0.75)--(r/2,-r/2-0.75),Arrows(4)); label("$r$",(0,-r/2-0.75),S);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
