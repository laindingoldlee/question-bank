Each semicircle has radius $\dfrac{1}{\pi}$ and arc length $\pi\cdot\dfrac{1}{\pi}=1 $. Hence the four arcs have total length $\boxed4 $.
\begin{asy}
usepackage("amsmath");
size(7cm);
draw((0,0)--(2,0)--(2,2)--(0,2)--cycle,dashed);draw(arc((1,0),1,180,360));draw(arc((2,1),1,-90,90));draw(arc((1,2),1,0,180));draw(arc((0,1),1,90,270));label("$\dfrac{2}{\pi}$",(1,0),N);label("$A$",(0,0),SW);label("$B$",(2,0),SE);label("$C$",(2,2),NE);label("$D$",(0,2),NW);
\end{asy}
