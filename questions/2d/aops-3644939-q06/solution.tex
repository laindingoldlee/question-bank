Put $A=(0,0),C=(2,0),D=(1,0),B=(1,\sqrt3)$. The square diagonal $DF$ has length $\sqrt2$, and $A$ lies on the segment, so $F=(1-\sqrt2,0)$. Thus $BF^2=(\sqrt2)^2+(\sqrt3)^2=\boxed5$.
\begin{asy}
import graph;import geometry;usepackage("amsmath");size(7cm);
pair A=(0,0),B=(1,sqrt(3)),C=(2,0),D=(1,0),F=(1-sqrt(2),0),E=(D+F)/2+(0,1/sqrt(2)),G=(D+F)/2-(0,1/sqrt(2));draw(A--B--C--cycle);draw(D--E--F--G--cycle,blue);draw(B--F,dashed);label("$A$",A,(0,-1));label("$B$",B,(0,1));label("$C$",C,(1,-1));label("$D$",D,(1,-1));label("$E$",E,(0,1));label("$F$",F,(-1,0));label("$G$",G,(0,-1));
\end{asy}
