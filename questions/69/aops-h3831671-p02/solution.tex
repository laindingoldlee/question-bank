The givens say $x^2+y^2=325$ and $xy=84$. Since $AB=x+y$, its square has area $(x+y)^2=x^2+y^2+2xy=325+168=\boxed{493}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
real x=(sqrt(493)+sqrt(157))/2,y=(sqrt(493)-sqrt(157))/2;draw(box((0,0),(x,x)));draw(box((x,0),(x+y,y)));label("$A$",(0,0),(0,-1));label("$C$",(x,0),(0,-1));label("$B$",(x+y,0),(0,-1));label("$x^2$",(x/2,x/2));label("$y^2$",(x+y/2,y/2));
\end{asy}
