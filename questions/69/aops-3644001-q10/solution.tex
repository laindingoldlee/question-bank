Since $DE\parallel BC$, $\angle ABC=180^\circ-125^\circ=55^\circ$. Therefore $\angle BAC=40^\circ$. Equality $AE=BE$ makes $\angle ABE=\angle BAE=40^\circ$, so $\angle AEB=100^\circ$ and $\angle BEC=80^\circ$. The requested sum is $80^\circ+40^\circ=\boxed{120^\circ}$.
\begin{asy}
import graph;import geometry;usepackage("amsmath");size(7cm);
pair B=(0,0),C=(5,0),A=intersectionpoint(B--B+10*dir(55),C--C+10*dir(95));real t=dot(B-A,B-A)/(2*dot(C-A,B-A));pair E=A+t*(C-A),D=A+t*(B-A);draw(A--B--C--cycle);draw(D--E^^B--E,blue);label("$A$",A,(0,1));label("$B$",B,(-1,-1));label("$C$",C,(1,-1));label("$D$",D,(-1,0));label("$E$",E,(1,0));
\end{asy}
