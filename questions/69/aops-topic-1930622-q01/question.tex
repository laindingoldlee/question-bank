\begin{enumerate}
\item[(a)] Show that
\[
\dfrac{1}{998}=\dfrac{1}{1000}+\dfrac{2}{1000^2}+\dfrac{2^2}{1000^3}+\dfrac{2^3}{1000^4}+\cdots .
\]
\item[(b)] Explain why the decimal expansion of $\dfrac{1}{998}$ begins $0.001\,002\,004\,008\,016\,032\,064\,128\,256\,513\ldots$, and why the block after $256$ is $513$ rather than $512$.
\item[(c)] More generally, write $\dfrac{1}{10^m-c}$ as a geometric series, for a positive integer $m$ and an integer $c$ with $0<c<10^m$.
\end{enumerate}
