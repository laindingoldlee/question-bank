Divide the L-shape into a six-by-two horizontal rectangle and a two-by-three rectangle above its left end. Their areas total $6\cdot2+2\cdot3=\boxed{18}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
filldraw((0,0)--(6,0)--(6,2)--(2,2)--(2,5)--(0,5)--cycle,lightblue);draw((0,2)--(2,2),dashed);label("$6$",(3,0),(0,-1));label("$5$",(0,2.5),(-1,0));label("$2$",(6,1),(1,0));label("$2$",(1,5),(0,1));
\end{asy}
