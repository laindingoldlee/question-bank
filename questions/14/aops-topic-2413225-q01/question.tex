\begin{enumerate}
\item[(a)] Prove that the sum of the first $n$ odd positive integers is $n^2$, that is, $1+3+5+\cdots+(2n-1)=n^2$.
\item[(b)] Find and prove a similar formula for the sum of the first $n$ even positive integers, $2+4+6+\cdots+2n$.
\end{enumerate}
