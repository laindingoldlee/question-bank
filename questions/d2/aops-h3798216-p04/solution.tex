From $\pi r^2=12$ and $r>0$, the radius is $\boxed{\sqrt{\dfrac{12}{\pi}}}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair O=(0,0),A=(2,0);draw(circle(O,2));draw(O--A);dot(O);label("$O$",O,(0,-1));label("$r$",(O+A)/2,(0,1));label("area $12$",(0,-.9));
\end{asy}
