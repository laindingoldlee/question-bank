The distance walked is $AB=20\cdot0.75=15$ m. Triangle $ABC$ has a right angle at $B$, so
\[BC=AB\tan57^\circ=15\tan57^\circ\approx15\cdot1.5399\approx23.10\text{ m}.\]

\begin{asy}
usepackage("amsmath");
size(5.5cm);
defaultpen(fontsize(10pt));
real AB=15, BC=15*Tan(57);
fill((-3,0)--(19,0)--(19,BC)--(-3,BC)--cycle,rgb(0.85,0.92,1));
draw((-3,0)--(19,0),linewidth(0.9)); draw((-3,BC)--(19,BC),linewidth(0.9));
pair A=(0,0), B=(AB,0), C=(AB,BC);
draw(A--B,linewidth(0.9)); draw(B--C,red+linewidth(0.9)); draw(A--C,dashed);
real s=0.9; draw((AB-s,0)--(AB-s,s)--(AB,s));
draw(arc(A,3,0,57)); label("$57^\circ$",A+4.2*dir(28),fontsize(8pt));
dot(A);dot(B);dot(C);
label("$A$",A,S);label("$B$",B,S);label("$C$",C,N);
label("$15$",(A+B)/2,S); label("$BC$",(B+C)/2,E,red);
label("river",(5,BC*0.75),fontsize(8pt));
\end{asy}
