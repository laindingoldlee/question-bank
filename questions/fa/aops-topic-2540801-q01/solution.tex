\textbf{Base case.} For $n=3$: $3^3=27<36=(3!)^2$.

\textbf{A lemma.} For every positive integer $n$, $\left(1+\dfrac1n\right)^n<3$. Indeed, by the binomial theorem,
\[\left(1+\dfrac1n\right)^n=\sum_{k=0}^{n}\binom nk\dfrac1{n^k},\qquad\binom nk\dfrac1{n^k}=\dfrac{1}{k!}\cdot\dfrac{n}{n}\cdot\dfrac{n-1}{n}\cdots\dfrac{n-k+1}{n}\le\dfrac1{k!}\le\dfrac1{2^{k-1}}\quad(k\ge1).\]
So the sum is at most $1+\left(1+\dfrac12+\dfrac14+\cdots\right)<1+2=3$.

\textbf{Inductive step.} Suppose $n\ge3$ and $n^n<(n!)^2$. By the lemma, $(n+1)^n=n^n\left(1+\dfrac1n\right)^n<3n^n\le(n+1)n^n$, so
\[(n+1)^{n+1}=(n+1)\cdot(n+1)^n<(n+1)^2\,n^n<(n+1)^2\,(n!)^2=\left((n+1)!\right)^2.\]
This is the statement for $n+1$, which completes the induction. $\blacksquare$

(The inequality fails for $n=1$ and $n=2$, where both sides are equal.)
