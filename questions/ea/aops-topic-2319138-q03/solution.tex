The midpoint has the averages of the coordinates of the endpoints:
\[M=\left(\dfrac{7+(-2)}{2},\ \dfrac{8+3}{2}\right)=\boxed{\left(\dfrac52,\ \dfrac{11}{2}\right)}.\]

\begin{asy}
usepackage("amsmath");
size(6cm);
defaultpen(fontsize(10pt));
import graph;
xaxis("$x$",-3.5,8.5,Arrow(5)); yaxis("$y$",-0.5,9.5,Arrow(5));
pair A=(7,8), B=(-2,3), M=(A+B)/2;
draw(A--B,linewidth(0.9));
dot(A); dot(B); dot(M,red);
label("$A(7,8)$",A,E); label("$B(-2,3)$",B,W); label("$M\left(\dfrac52,\dfrac{11}{2}\right)$",M,SE,red);
\end{asy}
