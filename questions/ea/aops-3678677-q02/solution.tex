Put $EC=3$ and $ED=x$. The perpendicular vectors from $E$ to $A$ and $B$ have dot product $4^2-3x=0$. Thus $x=\dfrac{16}{3}$, rectangle width $x+3=\dfrac{25}{3}$, and area $\boxed{\dfrac{100}{3}}$.
\begin{asy}
import graph;import geometry;usepackage("amsmath");size(7cm);
pair A=(0,4),B=(25/3,4),C=(25/3,0),D=(0,0),E=(16/3,0);draw(A--B--C--D--cycle);draw(A--E--B);label("$A$",A,(-1,1));label("$B$",B,(1,1));label("$C$",C,(1,-1));label("$D$",D,(-1,-1));label("$E$",E,(0,-1));label("$3$",(E+C)/2,(0,-1));label("$4$",(B+C)/2,(1,0));
\end{asy}
