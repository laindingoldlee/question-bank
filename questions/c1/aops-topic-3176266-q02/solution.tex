The three equal sides total $6$, so each has length $6\div3=\boxed2$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair[] pts={(0,0),(2,0),(1,sqrt(3))};string[] names={"$A$","$B$","$C$"};path boundary=pts[0];for(int j=1;j<pts.length;++j)boundary=boundary--pts[j];draw(boundary--cycle);for(int j=0;j<pts.length;++j)label(names[j],pts[j],dir(90+360*j/pts.length));
\end{asy}
