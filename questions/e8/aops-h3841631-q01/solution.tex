\begin{enumerate}
\item Since $0<t<1$ and $N=a^2+(2a+1)t$,
\[
N-x_0^2=t-t^2=t(1-t)>0.
\]
Both $x_0$ and $\sqrt N$ are positive, so the stated inequality follows.
\item Direct subtraction gives
\[
T(x)-\sqrt N=\dfrac{x^2+N-2x\sqrt N}{2x}
=\dfrac{(x-\sqrt N)^2}{2x}\ge0.
\]
The new error is quadratic in the previous error.
\item With $a=6$,
\[
x_0=\dfrac{90}{13},\quad
x_1=\dfrac{1351}{195}\approx6.9282051282,\quad
x_2=\dfrac{3650401}{526890}\approx6.9282032303.
\]
For a rigorous rounding check, direct squaring shows
$(6.928195)^2<48<(6.928205)^2$. Thus $\boxed{\sqrt{48}\approx6.92820}$ to five decimal places.
\end{enumerate}
