Choose $A=(0,0),B=(b,0)$ with $b>0$. Since the median is vertical and $\angle A=135^\circ$, we must have $C=(-b,b)$ and $M=(0,\dfrac{b}{2})$. Then $MB^2=b^2+(\dfrac{b}{2})^2=25$, so $b^2=20$. The area is half the base $b$ times altitude $b$, hence $\boxed{10}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,0),B=(sqrt(20),0),C=(-sqrt(20),sqrt(20)),M=midpoint(B--C);draw(A--B--C--cycle);draw(A--M,blue);label("$A$",A,S);label("$B$",B,SE);label("$C$",C,NW);label("$M$",M,NE);
\end{asy}
