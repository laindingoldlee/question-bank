Put $t=\angle DCA=\angle DBC$. In triangle $ACD$, $\angle CDA=134^\circ-t$. Since diagonal $DB$ lies inside this angle in a convex quadrilateral, $\angle CDB=89^\circ-t$. Triangle $BCD$ then gives $\angle BCD=180^\circ-t-(89^\circ-t)=\boxed{91^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair D=(0,0),A=(5,0),C=5sin(46*pi/180)/sin(30*pi/180)*dir(104);pair B=extension(D,D+dir(45),C,C+dir(15));draw(A--B--C--D--cycle);draw(A--C,dashed);draw(B--D,dashed);label("$A$",A,(1,0));label("$B$",B,(1,1));label("$C$",C,(-1,1));label("$D$",D,(-1,-1));
\end{asy}
