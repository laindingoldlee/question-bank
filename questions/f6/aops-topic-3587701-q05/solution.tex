Put $A=(0,0)$, $B=(10,0)$, $D=(0,10)$, so $X=(5,5)$ and $C=(10,10)$. Write $E=(t,0)$, $F=(0,t)$. Triangles $AEX$ and $AXF$ each have area $\dfrac{5t}{2}$, so $5t=5$ and $t=1$. Triangles $AEC$ and $ACF$ each have area $5t$, so $[AECF]=10t=\boxed{10}$.
\begin{asy}
size(7cm);
pair A=(0,0),B=(10,0),C=(10,10),D=(0,10),X=(5,5),E=(1,0),F=(0,1);draw(A--B--C--D--cycle);draw(E--X--F);draw(E--C--F,dashed);label("$A$",A,SW);label("$B$",B,SE);label("$C$",C,NE);label("$D$",D,NW);label("$X$",X,E);label("$E$",E,S);label("$F$",F,W);
\end{asy}
