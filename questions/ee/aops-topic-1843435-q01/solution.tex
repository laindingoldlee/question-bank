The roots are $-4$ and $\dfrac{3}{2}$, so $y=a(x+4)\left(x-\dfrac{3}{2}\right)$. At $x=0$: $a\cdot4\cdot\left(-\dfrac{3}{2}\right)=-6a=-15$, so $a=\dfrac{5}{2}$. Hence
\[
y=\dfrac{5}{2}(x+4)\left(x-\dfrac{3}{2}\right)=\boxed{\dfrac{5}{2}x^2+\dfrac{25}{4}x-15}
\]
(equivalently $y=\dfrac{5}{4}(x+4)(2x-3)$).

usepackage("amsmath");

draw((-5.6,0)--(3.2,0),gray(0.6),Arrow(5));
draw((0,-22)--(0,20),gray(0.6),Arrow(5));
draw(graph(f,-5.1,2.6),blue+linewidth(0.9));
dot((-4,0));dot((1.5,0));dot((0,-15));
label("$(-4,0)$",(-4,0),SW);label("$(1.5,0)$",(1.5,0),SE);label("$(0,-15)$",(0,-15),SE);
label("$x$",(3.2,0),E);label("$y$",(0,20),N);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
