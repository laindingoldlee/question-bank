Let $m_1,m_2,\dots,m_n$ be pairwise relatively prime positive integers, let $M=m_1m_2\cdots m_n$, and let $a_1,\dots,a_n$ be integers. For each $i$ let $b_i=\dfrac{M}{m_i}$, and let $y_i$ be an integer with $b_iy_i\equiv1\pmod{m_i}$.
\begin{enumerate}
\item[(a)] Explain why such $y_i$ exist, and prove that $x=\sum_{i=1}^{n}a_ib_iy_i$ satisfies $x\equiv a_i\pmod{m_i}$ for every $i$.
\item[(b)] Prove that the solutions of the system $x\equiv a_i\pmod{m_i}$ ($i=1,\dots,n$) are exactly the integers congruent to this $x$ modulo $M$.
\item[(c)] Use the formula to solve $x\equiv2\pmod3$, $x\equiv3\pmod5$, $x\equiv2\pmod7$.
\end{enumerate}
