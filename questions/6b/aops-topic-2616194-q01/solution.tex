Let the triangle have side $s$. A rectangle of height $h$ inscribed this way leaves a right triangle with a $60^\circ$ angle at each end of its base, with legs $h$ and $\dfrac{h}{\sqrt3}$. So its base is
\[s-\dfrac{2h}{\sqrt3}.\]
From the first rectangle, $1=s-\dfrac4{\sqrt3}$, so $s=1+\dfrac4{\sqrt3}$. The second rectangle has base
\[w=s-\dfrac2{\sqrt3}=1+\dfrac{2}{\sqrt3}=\dfrac{3+2\sqrt3}{3},\]
so $a+b+c=3+2+3=\boxed{8}$.

\begin{asy}
usepackage("amsmath");
size(7cm);
defaultpen(fontsize(10pt));
real s=1+4/sqrt(3),w=1+2/sqrt(3),dx=s+0.7;
pair A=(0,0),B=(s,0),C=(s/2,s*sqrt(3)/2);
draw(A--B--C--cycle,linewidth(0.8));draw(((s-1)/2,0)--((s-1)/2,2)--((s+1)/2,2)--((s+1)/2,0),linewidth(0.8));
label("$1$",(s/2,2),N,fontsize(8pt));label("$2$",((s+1)/2,1),E,fontsize(8pt));label("$\dfrac{2}{\sqrt3}$",((s-1)/4,0),S,fontsize(8pt));
draw(shift(dx,0)*(A--B--C--cycle),linewidth(0.8));draw(shift(dx,0)*(((s-w)/2,0)--((s-w)/2,1)--((s+w)/2,1)--((s+w)/2,0)),linewidth(0.8));
label("$1$",(dx+(s+w)/2,0.5),E,fontsize(8pt));label("$w$",(dx+s/2,1),N,fontsize(8pt));label("$\dfrac{1}{\sqrt3}$",(dx+(s-w)/4,0),S,fontsize(8pt));
draw(((s-1)/2+0.18,0)--((s-1)/2+0.18,0.18)--((s-1)/2,0.18));draw((dx+(s-w)/2+0.18,0)--(dx+(s-w)/2+0.18,0.18)--(dx+(s-w)/2,0.18));
\end{asy}
