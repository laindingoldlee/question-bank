$AM=\dfrac{5}{2}$ and $AD=5$, so triangle $AMD$ is right at $A$, with area $\dfrac{1}{2}\cdot\dfrac{5}{2}\cdot5=\dfrac{25}{4}$. Its hypotenuse $DM=\sqrt{25+\dfrac{25}{4}}=\dfrac{5\sqrt5}{2}$. Therefore the requested altitude is $\dfrac{2\cdot\dfrac{25}{4}}{\dfrac{5\sqrt5}{2}}=\boxed{\sqrt5}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),B=(5,0),C=(5,5),D=(0,5),M=(2.5,0),H=D+dot(A-D,M-D)/dot(M-D,M-D)*(M-D);draw(A--B--C--D--cycle);draw(D--M);draw(A--H,blue);label("$A$",A,SW);label("$B$",B,SE);label("$C$",C,NE);label("$D$",D,NW);label("$M$",M,S);label("$H$",H,NE);label("$5$",(0,2.5),W);
\end{asy}
