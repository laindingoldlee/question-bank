Use $A=(0,0)$, $B=(3,0)$, $D=(0,1)$ and $O=(\dfrac{3}{2},\dfrac{1}{2})$. The circle through $A$ has equation $x^2+y^2+ux+vy=0$. Substituting $D,O$ gives $v=-1,u=-\dfrac{4}{3}$. Its intersections with $AB$ satisfy $x(x-\dfrac{4}{3})=0$, so $E=(\dfrac{4}{3},0)$. Hence $BE=3-\dfrac{4}{3}=\dfrac{5}{3}$ and $\boxed{m+n=8}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");

defaultpen(fontsize(9pt));
pair A=(0,0),B=(3,0),C=(3,1),D=(0,1),O=(1.5,.5),E=(4/3.0,0);draw(A--B--C--D--cycle);draw(A--C,dashed);draw(B--D,dashed);draw(circle((2/3.0,.5),5/6.0),blue);label("A",A,SW);label("B",B,SE);label("C",C,NE);label("D",D,NW);label("O",O,(1,0));label("E",E,S);
\end{asy}
