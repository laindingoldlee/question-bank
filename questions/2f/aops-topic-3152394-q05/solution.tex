Symmetry makes the perpendicular feet the side midpoints. Thus $ DEF $ is the medial triangle, with side length $\dfrac{s}{2}$. Its area is $\dfrac{\sqrt3}{4}(\dfrac{s}{2})^2=\boxed{\dfrac{\sqrt3s^2}{16}}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),B=(4,0),C=(2,2sqrt(3)),H=(2,2sqrt(3)/3),D=(A+B)/2,Ep=(B+C)/2,F=(C+A)/2;draw(A--B--C--cycle);draw(D--Ep--F--cycle,blue);draw(H--D^^H--Ep^^H--F,dashed);label("$A$",A,SW);label("$B$",B,SE);label("$C$",C,N);label("$D$",D,S);label("$E$",Ep,NE);label("$F$",F,NW);dot("$H$",H,E);
\end{asy}
