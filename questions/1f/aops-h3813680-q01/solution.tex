Use coordinates $A=(0,0)$, $B=(w,0)$, $C=(w,h)$, $D=(0,h)$, $P=(x,y)$. Then both $PA^2+PC^2$ and $PB^2+PD^2$ equal $x^2+(x-w)^2+y^2+(y-h)^2$. The equality follows without any restriction on $x,y$, so it covers interior, exterior and boundary points.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,0),B=(4,0),C=(4,3),D=(0,3),P=(2,1.2);draw(A--B--C--D--cycle);draw(A--P--C,dashed);draw(B--P--D,dashed);label("$A$",A,(-1,-1));label("$B$",B,(1,-1));label("$C$",C,(1,1));label("$D$",D,(-1,1));dot(P);label("$P$",P,(0,1));
\end{asy}
