Put $A=(0,0)$ and $M=(8,0)$, with $C$ to the right. Since $MB=5$ and $\angle BMC=60^\circ$, the horizontal coordinate of $B$ is $8+5\cos60^\circ=\dfrac{21}{2}$, regardless of which side of the base it lies on. In an isosceles triangle with $AB=BC$, $B$ lies on the perpendicular bisector of $AC$, so $AC=21$. Therefore $\boxed{AC^2=441}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,0),M=(8,0),C=(21,0),B=(10.5,5sqrt(3)/2);draw(A--B--C--cycle);draw(B--M,blue);draw(B--(10.5,0),dashed);label("$A$",A,(-1,-1));label("$M$",M,(0,-1));label("$C$",C,(1,-1));label("$B$",B,(0,1));label("$8$",(4,0),(0,-1));label("$5$",(B+M)/2,(-1,0));label("$60^\circ$",M+(1,.3));
\end{asy}
