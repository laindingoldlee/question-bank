Let $f(x)=\dfrac{2x+1}{5}$ and $g(x)=5x^2$.
\begin{enumerate}
\item[(a)] For which $x$ is $f(g(x))=f(2x+1)$?
\item[(b)] For which $x$ is $f(g(x+1)+1)=f(g(x)+1)+1$?
\end{enumerate}
