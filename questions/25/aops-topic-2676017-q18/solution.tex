Draw through $P$ lines parallel to the sides. Let $P$ be at distances $x$ and $y$ from the sides $AD$ and $BC$, and at distances $u$ and $v$ from the sides $AB$ and $CD$. By Pythagoras,
\[PA^2=x^2+u^2,\quad PB^2=y^2+u^2,\quad PC^2=y^2+v^2,\quad PD^2=x^2+v^2.\]
Hence $PA^2+PC^2=PB^2+PD^2$, so $PD^2=289+36-225=100$ and $PD=\boxed{10}$.

\begin{asy}
usepackage("amsmath");
size(3cm);
defaultpen(fontsize(10pt));
real c=4.2,bb=sqrt(36-c^2),a=sqrt(100-c^2),d=sqrt(225-bb^2);pair P=(0,0),A=(-a,d),B=(bb,d),C=(bb,-c),D=(-a,-c);
draw(A--B--C--D--cycle,linewidth(0.8));draw(P--A);draw(P--B);draw(P--C);draw(P--D);
draw((-a,0)--(bb,0),dashed);draw((0,-c)--(0,d),dashed);dot(P);
label("$A$",A,NW);label("$B$",B,NE);label("$C$",C,SE);label("$D$",D,SW);label("$P$",P,dir(-125),fontsize(8pt));
label("$17$",A/2,NE,fontsize(8pt));label("$15$",B*0.42+(1.1,0),fontsize(8pt));label("$6$",C/2,NE,fontsize(7pt));
label("$x$",(-a/2,0),N,fontsize(8pt));label("$y$",(bb*0.6,0),N,fontsize(8pt));label("$u$",(0,d*0.6),E,fontsize(8pt));label("$v$",(0,-0.75*c),E,fontsize(7pt));
\end{asy}
