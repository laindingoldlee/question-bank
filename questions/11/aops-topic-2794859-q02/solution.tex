Use $A=(0,0),B=(6,0),D=(0,6),P=(3,-3\sqrt3)$. The angle between the downward ray $DA$ and $DP$ has tangent $\dfrac{3}{6+3\sqrt3}=2-\sqrt3=\tan15^\circ$. Thus $\boxed{\angle ADP=15^\circ}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),B=(6,0),C=(6,6),D=(0,6),P=(3,-3*sqrt(3));draw(A--B--C--D--cycle);draw(A--P--B);draw(D--P,blue);label("A",A,W);label("B",B,E);label("C",C,NE);label("D",D,NW);label("P",P,S);
\end{asy}
