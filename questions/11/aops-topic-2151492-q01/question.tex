\begin{enumerate}
\item[(a)] Show that there is no integer $x$ for which $\dfrac{2^x}{10}$ is an integer.
\item[(b)] If $x$ may be any real number, what is the smallest $x$ for which $\dfrac{2^x}{10}$ is an integer?
\end{enumerate}
