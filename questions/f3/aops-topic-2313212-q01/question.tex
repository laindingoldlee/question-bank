Here $[x]$ denotes the greatest integer not exceeding $x$, so that $[x]\le x<[x]+1$.
\begin{enumerate}
\item[(a)] Show that for every integer $n\ge2$,
\[\sum_{k=2}^{n}\left[\sqrt{k^2-1}\right]=\dfrac{n(n-1)}{2},\]
so that this sum is not equal to $\dfrac{(n-1)n(4n+1)}{6}$.
\item[(b)] Prove that for every integer $n\ge2$,
\[\sum_{k=1}^{n^2-1}\left[\sqrt{k}\right]=\dfrac{(n-1)n(4n+1)}{6}.\]
\end{enumerate}
