\textbf{Domain.} $x^2-5x+6=(x-2)(x-3)$, so the domain is all real $x$ except $2$ and $3$: $\boxed{(-\infty,2)\cup(2,3)\cup(3,\infty)}$.

\textbf{Range.} For $x\ne2,3$ we may cancel one factor $x-3$:
\[f(x)=\dfrac{(x-3)^2}{x-2}.\]
Put $t=x-2$, so $t\ne0$ and $t\ne1$ (the latter because $x\ne3$). Then
\[f(x)=\dfrac{(t-1)^2}{t}=t-2+\dfrac1t.\]
\begin{itemize}
\item For $t>0$: $t+\dfrac1t\ge2$, so $f\ge0$, with $f=0$ only at $t=1$, which is excluded. So $f>0$. Every $y>0$ is attained: $t+\dfrac1t=y+2$ has two positive roots with product $1$, and they are not equal to $1$ since $y\ne0$.
\item For $t<0$: $t+\dfrac1t\le-2$, so $f\le-4$, and every $y\le-4$ is attained (for example $f=-4$ at $t=-1$, i.e. $x=1$).
\end{itemize}
So the range is
\[\boxed{(-\infty,-4]\cup(0,\infty)}.\]
Note that $0$ is missing because of the hole at $x=3$.
