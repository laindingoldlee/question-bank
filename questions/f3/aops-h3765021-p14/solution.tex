In triangle $APB$, the base angles total $\dfrac{A+B}{2}=\dfrac{180^\circ-36^\circ}{2}=72^\circ$. Therefore $\angle APB=180^\circ-72^\circ=\boxed{108^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair C=(0,3/tan(18*pi/180)),A=(-3,0),B=(3,0);real s=abs(C-A);pair P=(s*A+s*B+6*C)/(2*s+6);draw(A--B--C--cycle);draw(A--P--B,blue);label("A",A,SW);label("B",B,SE);label("C",C,N);label("P",P,N);label("$36^\circ$",C+(0,-1.4));
\end{asy}
