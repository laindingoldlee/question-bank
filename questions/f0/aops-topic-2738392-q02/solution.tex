\begin{itemize}
\item (a) Just below $\dfrac\pi2$, $\sin x\to1$ and $\cos x\to0^+$, so $\tan x\to\boxed{+\infty}$.
\item (b) Just above $0$, $\cos x\to1$ and $\sin x\to0^+$, so $\cot x\to\boxed{+\infty}$.
\item (c) Just below $\dfrac{3\pi}{2}$ (in the third quadrant), $\cos x\to0^-$, so $\sec x\to\boxed{-\infty}$.
\item (d) Just below $\pi$ (in the second quadrant), $\sin x\to0^+$, so $\csc x\to\boxed{+\infty}$.
\end{itemize}
