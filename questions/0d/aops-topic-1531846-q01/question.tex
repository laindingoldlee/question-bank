\begin{enumerate}
\item[(a)] Equilateral triangle $ABC$ has side length $1$, and squares $ABDE$, $BCHI$ and $CAFG$ lie outside the triangle. What is the area of hexagon $DEFGHI$?

$\textbf{(A)}\ \dfrac{12+3\sqrt3}{4}\qquad\textbf{(B)}\ \dfrac{9}{2}\qquad\textbf{(C)}\ 3+\sqrt3\qquad\textbf{(D)}\ \dfrac{6+3\sqrt3}{2}\qquad\textbf{(E)}\ 6$
\item[(b)] Now let $ABC$ be any triangle, with squares $ABDE$, $BCHI$ and $CAFG$ drawn outside it. Prove that each of the three triangles $AEF$, $BID$ and $CGH$ between the squares has the same area as triangle $ABC$.
\end{enumerate}
