Set $ A=(0,0),B=(20,0),C=(0,17)$. The angle bisector is $ y=x $, while $ BC $ has equation $\dfrac{x}{20}+\dfrac{y}{17}=1 $. Thus $ D=(\dfrac{340}{37},\dfrac{340}{37})$, giving $\boxed{AD=\dfrac{340\sqrt2}{37}}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),B=(20,0),C=(0,17),D=(340/37,340/37);draw(A--B--C--cycle);draw(A--D,blue);label("$A$",A,SW);label("$B$",B,SE);label("$C$",C,N);label("$D$",D,NE);label("$20$",(10,0),S);label("$17$",(0,8.5),W);
\end{asy}
