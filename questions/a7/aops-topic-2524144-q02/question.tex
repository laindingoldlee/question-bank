\begin{enumerate}
\item[(a)] Show that the sum of $\dfrac1n$ over all positive integers $n$ that are multiples of $2$ or of $3$ is infinite.
\item[(b)] Find the sum of $\dfrac1n$ over all integers $n>1$ whose only prime factors are $2$ and $3$, that is, over $n=2^a3^b$ with $a,b\ge0$ and $a+b\ge1$.
\end{enumerate}
