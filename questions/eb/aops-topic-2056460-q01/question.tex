\begin{enumerate}
\item[(a)] Let \(n\) and \(k\) be positive integers with \(k\le 2n\), so that \(n^2<n^2+k<(n+1)^2\). Prove that
\[
n+\dfrac{k}{2n+1}<\sqrt{n^2+k}<n+\dfrac{k}{2n}.
\]
\item[(b)] Use (a) to find \(\sqrt{2020}\) correct to one decimal place (that is, find the integer part and the first decimal digit).
\end{enumerate}
