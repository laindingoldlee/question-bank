Scale to side $1$, with $A=(0,0),B=(1,0),C=(1,1),D=(0,1)$. Put $C'=(1-d,0)$, $0<d\le1$. The reflection axis is the perpendicular bisector of $CC'$, with normal $(d,1)$. Reflecting the difference $Z-C=(-\dfrac{1}{3},0)$ yields
\[
Z'=C'+\left(-\dfrac{1}{3},0\right)+\dfrac{2d}{3(d^2+1)}(d,1).
\]
Since $PQ$ has equation $x=\dfrac{1}{3}$, we obtain $1-d-\dfrac{1}{3}+\dfrac{2d^2}{3(d^2+1)}=\dfrac{1}{3}$, or $3d^3-3d^2+3d-1=0$.
Let $r=\dfrac{AC'}{BC'}=\dfrac{1-d}{d}$. Substituting $d=\dfrac{1}{r+1}$ reduces the equation to $r^3=2$. This root produces a valid image on the segment: its height is $\dfrac{2d}{3(d^2+1)}\in(0,1)$. Restoring $BC'=1$ gives $\boxed{AC'=\sqrt[3]2}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
real r=2^(1./3),l=1+r,d=1/l,h=(2*l*l-r*r)/2;pair A=(0,0),B=(l,0),C=(l,l),D=(0,l),P=(l/3,0),R=(2*l/3,0),Q=(l/3,l),Z=(2*l/3,l),Cp=(r,0),Zp=(l/3,l*2*d/(3*(1+d*d)));draw(A--B--C--D--cycle);draw(P--Q,gray);draw(R--Z,gray);draw((0,h/l)--(l,(h-l)/l),blue+dashed);draw(Cp--Zp,red);draw(C--Cp,dotted);draw(Z--Zp,dotted);label("$A$",A,(-1,-1));label("$B$",B,(1,-1));label("$C$",C,(1,1));label("$D$",D,(-1,1));label("$P$",P,(0,-1));label("$R$",R,(0,-1));label("$Q$",Q,(0,1));label("$Z$",Z,(0,1));label("$C'$",Cp,(0,-1));label("$Z'$",Zp,(-1,0));
\end{asy}
