Place the right vertex at the origin, with legs on the axes. Then the interior point has coordinates $(4,3)$ and the hypotenuse is $x+y=s$, where $s=AB$. Its distance to that line is $\dfrac{s-7}{\sqrt2}=5$, giving $\boxed{s=7+5\sqrt2}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(220);
defaultpen(fontsize(9pt));
pair A=(0,7+5*sqrt(2)),B=(0,0),C=(7+5*sqrt(2),0); draw(A--B--C--cycle); label("$A$",A,(-1,-1)); label("$B$",B,(1,-1)); label("$C$",C,(0,1)); pair P=(4,3);dot(P);draw(P--(0,3),dashed);draw(P--(4,0),dashed);label("$P$",P,(1,1));
\end{asy}
