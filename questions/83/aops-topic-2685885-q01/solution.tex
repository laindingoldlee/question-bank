Write $\tan x=\dfrac{\sin x}{\cos x}$ (in the unit circle below, triangles $OQP$ and $OUT$ are similar, so $\dfrac{\tan x}{1}=\dfrac{\sin x}{\cos x}$). Then, using $1-\cos^2x=\sin^2x$,
\[\tan^2x-\sin^2x=\dfrac{\sin^2x}{\cos^2x}-\sin^2x=\dfrac{\sin^2x\left(1-\cos^2x\right)}{\cos^2x}=\dfrac{\sin^2x}{\cos^2x}\cdot\sin^2x=\tan^2x\sin^2x.\ \blacksquare\]

\begin{asy}
usepackage("amsmath");
size(4cm);
defaultpen(fontsize(10pt));
real t=35;pair O=(0,0),P=dir(t),Q=(Cos(t),0),T=(1,Tan(t));
draw(arc(O,1,-8,100),gray);draw((-0.15,0)--(1.3,0),gray);draw((0,-0.15)--(0,1.15),gray);
draw(O--T,linewidth(0.8));draw(P--Q,linewidth(0.8));draw((1,0)--T,linewidth(0.8));
draw(Q+(-0.06,0)--Q+(-0.06,0.06)--Q+(0,0.06));draw((1,0)+(-0.06,0)--(1,0)+(-0.06,0.06)--(1,0.06));
draw(arc(O,0.22,0,t));label("$x$",0.3*dir(t/2),fontsize(8pt));
dot(P);label("$P$",P,NW);label("$O$",O,SW);label("$T$",T,NE);label("$U$",(1,0),SE);label("$Q$",Q,S+0.6W,fontsize(8pt));
label("$1$",P/2,NW,fontsize(8pt));label("$\cos x$",(Q.x/2,0),S,fontsize(8pt));label("$\sin x$",(Q+P)/2,W,fontsize(8pt));label("$\tan x$",((1,0)+T)/2,E,fontsize(8pt));
\end{asy}
