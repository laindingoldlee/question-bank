Let $R_k=\underbrace{11\ldots1}_{k}=\dfrac{10^k-1}{9}$.

\textbf{Pigeonhole.} Consider $R_1,R_2,\dots,R_{n+1}$. Two of them, say $R_i$ and $R_j$ with $i<j$, have the same remainder modulo $n$. Then $n\mid R_j-R_i=R_{j-i}\cdot10^i$. Since $n$ is odd and not divisible by $5$, $\gcd(n,10^i)=1$, so $n\mid R_{j-i}$.

\textbf{Euler's theorem.} $\gcd(9n,10)=1$, so $10^{\varphi(9n)}\equiv1\pmod{9n}$. Then $9n\mid10^{\varphi(9n)}-1=9R_{\varphi(9n)}$, so $n\mid R_{\varphi(9n)}$. $\blacksquare$
