\begin{enumerate}
\item[(a)] Prove that if $k\equiv3\pmod4$, then the equation $x^2+y^2=k$ has no integer solutions.
\item[(b)] Find all integer solutions $(x,y)$ of $x^2+y^2=65$. How many ordered pairs are there?
\end{enumerate}
