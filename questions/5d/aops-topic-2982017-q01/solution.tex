\begin{enumerate}
\item The square's four vertices lie on the circle and its sides are chords, whose interior points lie inside the circle: $\boxed{4}$ points.
\item $f^{-1}(4)=\boxed{3}$ by definition of the inverse.
\item Let $O$ be the centre and $\ell$ the line through $T$ perpendicular to $OT$.
\begin{asy}

\end{asy}
For any other point $Z$ of $\ell$, triangle $OTZ$ has a right angle at $T$, so $OZ>OT$ and $Z$ lies outside the circle. Thus $\ell$ meets the circle only at $T$ and is tangent there. $\blacksquare$
\item $4=\boxed{2^2}$.
\end{enumerate}
