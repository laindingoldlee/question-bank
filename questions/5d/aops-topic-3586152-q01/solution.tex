Set $D=(0,0)$, $C=(6,0)$, $A=(0,8)$, $B=(6,8)$, $E=(0,6)$. The circle is $x^2+y^2-6x-8y=0$. On ray $CE$, write $(x,y)=(6-6t,6t)$, $t\ge0$. Substitution gives $72t^2-84t=0$. Apart from $C$ at $t=0$, we have $t=\dfrac{7}{6}$, giving $F=(-1,7)$. Therefore $DF=\sqrt{1+49}=\boxed{5\sqrt2}$ and $FB=\sqrt{49+1}=\boxed{5\sqrt2}$.
\begin{asy}
size(7cm);
pair D=(0,0),C=(6,0),A=(0,8),B=(6,8),E=(0,6),F=(-1,7);draw(circle((3,4),5));draw(A--B--C--D--cycle);draw(C--F);draw(D--F--B,dashed);label("$A$",A,NW);label("$B$",B,NE);label("$C$",C,SE);label("$D$",D,SW);label("$E$",E,E);label("$F$",F,W);
\end{asy}
