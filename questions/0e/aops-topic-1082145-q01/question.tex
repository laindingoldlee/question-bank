\begin{enumerate}
\item[(a)] Is it true that $PA+PB<CA+CB$ for every triangle $ABC$ and every point $P$ inside it?
\item[(b)] Is it true that $PA+PB+PC<DA+DB+DC$ for every tetrahedron $ABCD$ and every point $P$ inside it?
\end{enumerate}
