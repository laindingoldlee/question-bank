\textbf{Yes.} Make two perpendicular vertical cuts through the axis, giving $4$ equal quarter-cylinders, and one horizontal cut at half the height, which halves each of them. The result is $8$ congruent pieces.

draw(circle((0,0),1));draw((-1,0)--(1,0));draw((0,-1)--(0,1));label("top view; third cut horizontal at half height",(0,-1.2),S);
\end{asy}
