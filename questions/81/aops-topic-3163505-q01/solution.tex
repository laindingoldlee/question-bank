The perpendicular legs are a base and its altitude, so the area is $\dfrac{1}{2}\cdot20\cdot18=\boxed{180}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair[] pts={(0,0),(20,0),(0,18)};string[] names={"$A$","$B$","$C$"};path boundary=pts[0];for(int j=1;j<pts.length;++j)boundary=boundary--pts[j];draw(boundary--cycle);for(int j=0;j<pts.length;++j)label(names[j],pts[j],dir(90+360*j/pts.length));
\end{asy}
