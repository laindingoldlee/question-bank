The altitude from $B$ bisects $AC$, giving half-base 21 and height $\sqrt{29^2-21^2}=20$. Thus the area is $\dfrac{1}{2}\cdot42\cdot20=\boxed{420}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(-21,0),C=(21,0),B=(0,20),D=(0,0);draw(A--B--C--cycle);draw(B--D,dashed);label("A",A,SW);label("B",B,N);label("C",C,SE);label("D",D,S);label("29",(A+B)/2,NW);label("29",(B+C)/2,NE);label("42",(0,-2),S);
\end{asy}
