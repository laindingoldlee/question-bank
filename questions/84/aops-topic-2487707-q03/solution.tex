At the start $y=\sin0=0$, and at the end $y=\sin\dfrac{\pi}{2}=1$. So the $y$-coordinate has increased by $\boxed{1}$. (On $\left[0,\dfrac\pi2\right]$ the function $\sin x$ increases steadily from $0$ to its maximum value $1$.)

\begin{asy}
usepackage("amsmath");
size(5cm);
defaultpen(fontsize(10pt));
import graph;
real k=1.2;
draw((-0.5,0)--(3.7,0),Arrow(4)); draw((0,-1.4)--(0,1.6),Arrow(4));
label("$x$",(3.7,0),E); label("$y$",(0,1.6),N);
draw(graph(new pair(real x){return (x,k*sin(x));},-0.3,3.5),blue+linewidth(0.9));
pair Pt=(pi/2,k);
dot((0,0)); dot(Pt);
draw((pi/2,0)--Pt,dashed); draw((0,k)--Pt,dashed);
label("$\dfrac{\pi}{2}$",(pi/2,0),S,fontsize(8pt)); label("$1$",(0,k),W,fontsize(8pt));
label("$\left(\dfrac{\pi}{2},1\right)$",Pt,NE,fontsize(8pt));
label("$\pi$",(pi,0),SW,fontsize(8pt)); label("$y=\sin x$",(3.4,k*sin(3.4)),SE,blue+fontsize(8pt));
\end{asy}
