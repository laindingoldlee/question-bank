\begin{enumerate}
\item[(a)] Let \(a\ne b\) be constants. Show that for every \(k\) with \(k\ne -a,-b\),
\[
\dfrac{1}{(k+a)(k+b)}=\dfrac{1}{b-a}\left(\dfrac{1}{k+a}-\dfrac{1}{k+b}\right).
\]
\item[(b)] Use (a) to evaluate \(\displaystyle\sum_{k=1}^{\infty}\dfrac{1}{k(k+2)}\).
\end{enumerate}
