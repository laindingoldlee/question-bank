\begin{enumerate}
\item The first three terms sum to $\dfrac{5}{2}$, while $4-\varphi<\dfrac{5}{2}$ because $\varphi>\dfrac{3}{2}$. All remaining terms are positive, so the claimed equality is impossible.
\item Since $F_{n+2}\ge2F_n$, both $F_{2k+1}$ and $F_{2k+2}$ are at least $2^k$. Thus the reciprocal series is bounded by $\sum_{k=0}^{\infty}2\cdot2^{-k}=4$, proving convergence. The bound is strict since $F_4=3>2$. Its first five terms sum to $\dfrac{91}{30}>3$, giving the strict lower bound.
\item Binet's formula is $F_m=\dfrac{\varphi^m-(-\varphi^{-1})^m}{\sqrt5}$; it follows by checking the initial values and the recurrence. For even $m$,
\[
\dfrac{1}{F_m}=\dfrac{\sqrt5\,\varphi^m}{\varphi^{2m}-1}
=\sqrt5\left(\dfrac{1}{\varphi^m-1}-\dfrac{1}{\varphi^{2m}-1}\right).
\]
Hence for $K\ge1$,
\[
\sum_{k=0}^K\dfrac{1}{F_{2^k}}
=1+\sqrt5\left(\dfrac{1}{\varphi^2-1}-\dfrac{1}{\varphi^{2^{K+1}}-1}\right).
\]
Letting $K$ tend to infinity and using $\varphi^2-1=\varphi$ gives
\[
1+\dfrac{\sqrt5}{\varphi}=\dfrac{7-\sqrt5}{2}=\boxed{4-\varphi}.
\]
\end{enumerate}
