Since $8^2+15^2=64+225=289=17^2$, the converse of the Pythagorean theorem shows the triangle is right-angled opposite its longest side $BC$. Thus $\boxed{\angle A=90^\circ}$.

texpreamble("\usepackage{amsmath,amssymb}");

defaultpen(fontsize(9pt));
pair A=(0,0),B=(8,0),C=(0,15);draw(A--B--C--cycle);draw((1,0)--(1,1)--(0,1));label("$A$",A,(-1,-1));label("$B$",B,(1,-1));label("$C$",C,(0,1));label("$8$",(A+B)/2,(0,-1));label("$15$",(A+C)/2,(-1,0));label("$17$",(B+C)/2,(1,0));
\end{asy}
