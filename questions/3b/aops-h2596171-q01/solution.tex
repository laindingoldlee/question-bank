Take $B=(0,0)$, $C=(0,24)$, $A=(-12,0)$, $D=(-20,24)$, and $E=(0,14)$. The lines $AC$ and $DE$ are
\[
y=2x+24,\qquad y=14-\dfrac{x}{2},
\]
so $F=(-4,16)$. The vectors from $A$ to $D,F$ are $(-8,24)$ and $(8,16)$, respectively. Hence
\[
[ADF]=\dfrac{1}{2}|(-8)(16)-24(8)|=\boxed{160}.
\]
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(-12,0),B=(0,0),C=(0,24),D=(-20,24),E=(0,14),F=(-4,16);fill(A--D--F--cycle,lightblue);draw(A--B--C--D--cycle);draw(A--C);draw(D--E);label("$A$",A,SW);label("$B$",B,SE);label("$C$",C,NE);label("$D$",D,NW);label("$E$",E,(1,0));label("$F$",F,NE);
\end{asy}
