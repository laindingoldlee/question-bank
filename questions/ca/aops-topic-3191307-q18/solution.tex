$AC^2=4^2+11^2=137$, while $BD^2=2^2+(-8)^2=68$, so $BD$ is shorter. Its slope is $-4$, so its perpendicular bisector has slope $\boxed{\dfrac{1}{4}}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair[] pts={(2,-1),(3,6),(6,10),(5,-2)}; string[] names={"$A$","$B$","$C$","$D$"}; path boundary=pts[0]; for(int j=1;j<pts.length;++j)boundary=boundary--pts[j];draw(boundary--cycle); for(int i=0;i<pts.length;++i) label(names[i],pts[i],dir(90+360*i/pts.length));
\end{asy}
