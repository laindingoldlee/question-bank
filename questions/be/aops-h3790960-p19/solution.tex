Place $D=(0,0)$ and $A=(30,0)$, and take the quadrilateral above $DA$. Since angle $A$ is right, write $B=(30,h)$. The $45^\circ$ angle at $D$ puts $C=(t,t)$. Since $BC\perp CD$ and $BC=16$, the vector $C-B$ is $(-8\sqrt2,8\sqrt2)$. Hence $t=30-8\sqrt2$ and $h=30-16\sqrt2$. Split along $BD$ into the two right triangles $BAD$ and $BCD$. Their total area is $\dfrac{1}{2}(30h+16\cdot\sqrt2t)=\dfrac{1}{2}(900-256)=\boxed{322}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair D=(0,0),A=(30,0),B=(30,30-16*sqrt(2)),C=(30-8*sqrt(2),30-8*sqrt(2));draw(A--B--C--D--cycle);draw(B--D,dashed);label("A",A,SE);label("B",B,E);label("C",C,N);label("D",D,SW);label("30",(A+D)/2,S);label("16",(B+C)/2,NE);
\end{asy}
