Use $A=(0,0)$, $B=(s,0)$, $C=(s,s)$, $D=(0,s)$, $P=(s,t)$ with $0\le t\le s$ and $AP=4$. Distances from $B,C,D$ to line $AP$ are $\dfrac{st}{4}$, $\dfrac{s(s-t)}{4}$, $\dfrac{s^2}{4}$, respectively. Their sum is $\dfrac{s^2}{2}=6$. Triangle $APD$ has base $AD=s$ and horizontal height $s$, so its area is also $\dfrac{s^2}{2}=\boxed{6}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
real s=sqrt(12);pair A=(0,0),B=(s,0),C=(s,s),D=(0,s),P=(s,2);pair foot(pair X){return (X.x*P.x+X.y*P.y)/16*P;}draw(A--B--C--D--cycle);filldraw(A--P--D--cycle,lightblue);draw(B--foot(B),dashed);draw(C--foot(C),dashed);draw(D--foot(D),dashed);draw(A--(1.3*P),blue);label("A",A,SW);label("B",B,SE);label("C",C,NE);label("D",D,NW);label("P",P,E);
\end{asy}
