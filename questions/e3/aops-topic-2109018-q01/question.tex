\begin{enumerate}
\item[(a)] Show that there is a function $f:\mathbb R\to\mathbb R$ such that
\[\left(f(x)-1\right)\left(f(x)-2\right)f(x)=x\qquad\text{for all real }x,\]
and give an example of such a function.
\item[(b)] Prove that no such function is continuous.
\end{enumerate}
