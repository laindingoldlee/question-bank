Since $OA=OB$, triangle $AOB$ is isosceles, giving $\angle AOB=180^\circ-2\cdot24^\circ=132^\circ$. The rays $OA,OC$ are opposite, so $\angle BOC=180^\circ-132^\circ=\boxed{48^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair O=(0,0),A=(-1,0),C=(1,0),B=dir(48);draw(unitcircle);draw(A--C);draw(A--B--O,dashed);label("O",O,S);label("A",A,W);label("B",B,NE);label("C",C,E);label("$24^\circ$",(-.68,.09));label("$48^\circ$",(.28,.12));
\end{asy}
