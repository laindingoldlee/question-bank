A student claims that for every prime $p>2$ the number
\[
f(p)=2p+1+i^{\,p+1}(p-5),
\]
where $i^2=-1$, is prime.
\begin{enumerate}
\item[(a)] Show that $f(p)=p+6$ if $p\equiv1\pmod4$ and $f(p)=3p-4$ if $p\equiv3\pmod4$.
\item[(b)] Find the smallest prime $p>2$ for which $f(p)$ is not prime.
\end{enumerate}
