\begin{enumerate}
\item[(a)] Prove that for all real $\alpha$ and $\beta$,
\[
\cos\alpha+\cos\beta=2\cos\dfrac{\alpha+\beta}{2}\cos\dfrac{\alpha-\beta}{2}.
\]
\item[(b)] Is $\cos\alpha+\cos\beta=\cos(\alpha+\beta)$ true for all $\alpha,\beta$?
\end{enumerate}
