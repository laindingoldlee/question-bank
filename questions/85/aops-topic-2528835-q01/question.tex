Let $x$ and $y$ be real numbers with $y\ne0$ and $x\ne\pm y$.
\begin{enumerate}
\item[(a)] Simplify $\dfrac{x^2}{x^{-1}}$ (for $x\ne0$).
\item[(b)] Simplify $\dfrac{(x+y)y^{-1}-2x(x+y)^{-1}}{(x-y)y^{-1}+2x(x-y)^{-1}}$.
\end{enumerate}
