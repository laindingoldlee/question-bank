The function is defined exactly when $\cos x\ne0$, i.e. $x\ne\dfrac{\pi}{2}+k\pi$ for every integer $k$. On its domain, $f(x)=1+\tan x$, and $\tan x$ takes every real value on each interval $\left(-\dfrac\pi2+k\pi,\dfrac\pi2+k\pi\right)$. So
\[\text{domain}=\boxed{\bigcup_{k\in\mathbb Z}\left(-\dfrac{\pi}{2}+k\pi,\ \dfrac{\pi}{2}+k\pi\right)},\qquad\text{range}=\boxed{(-\infty,\infty)}.\]

\begin{asy}
usepackage("amsmath");
size(7cm);
defaultpen(fontsize(10pt));
import graph;
real ys=0.5;
real f(real x){return ys*(1+tan(x));}
draw((-5,0)--(5.1,0),Arrow(5)); draw((0,-2.3)--(0,3),Arrow(5));
label("$x$",(5.1,0),E); label("$y$",(0,3),N);
for(int k=-1;k<=1;++k){
  real lo=-pi/2+k*pi+atan(1/5), hi=pi/2+k*pi-atan(1/4);
  draw(graph(f,lo,hi),blue+linewidth(0.9));
}
for(int k=-1;k<=2;++k){ real xa=-pi/2+k*pi; draw((xa,-2.3)--(xa,2.8),dashed+gray(0.5)); }
label("$-\dfrac{\pi}{2}$",(-pi/2,-2.3),S,fontsize(7pt)); label("$\dfrac{\pi}{2}$",(pi/2,-2.3),S,fontsize(7pt));
label("$-\dfrac{3\pi}{2}$",(-3pi/2,-2.3),S,fontsize(7pt)); label("$\dfrac{3\pi}{2}$",(3pi/2,-2.3),S,fontsize(7pt));
draw((-0.08,ys)--(0.08,ys)); label("$1$",(-0.08,ys),NW,fontsize(7pt));
\end{asy}
