The vertical parallel sides have lengths $7$ and $9$ and are separated by $4$. The trapezoid area is $\dfrac{7+9}{2}\cdot4=\boxed{32}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair[] pts={(-1,-2),(3,-1),(3,8),(-1,5)}; string[] names={"$A$","$B$","$C$","$D$"}; path boundary=pts[0]; for(int j=1;j<pts.length;++j)boundary=boundary--pts[j];draw(boundary--cycle); for(int i=0;i<pts.length;++i) label(names[i],pts[i],dir(90+360*i/pts.length));
\end{asy}
