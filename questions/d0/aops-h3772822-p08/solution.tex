Let the rhombus side be $1$ and take $A=(0,0)$, $B=(1,0)$, $D=(\cos20^\circ,\sin20^\circ)$, $C=B+D$. Let $M$ and $N$ be the feet on $AD$ and $CD$. Both altitudes have length $\sin20^\circ$, so $BMN$ is isosceles. The directions of $BM$ and $BN$ are $110^\circ$ and $90^\circ$, respectively, giving $\angle MBN=20^\circ$. The other two angles are $80^\circ$, so the largest is $\boxed{80^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,0),B=(1,0),D=dir(20),C=B+D,M=cos(20*pi/180)*D,N=(1,D.y);draw(A--B--C--D--cycle);draw(B--M--N--B,blue);label("A",A,SW);label("B",B,S);label("C",C,NE);label("D",D,NW);label("M",M,W);label("N",N,(0,1));
\end{asy}
