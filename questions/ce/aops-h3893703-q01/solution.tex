Parallelism gives $b=5$. Also $BC^2=2^2+5^2=29$, so $AD^2=a^2+25=29$ and $a^2=4$. Both $D=(2,5)$ and $D=(-2,5)$ satisfy the conditions, the latter giving a parallelogram. In either case $a^2b=4\cdot5=\boxed{20}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,0),B=(10,0),C=(8,5),D=(2,5);draw(A--B--C--D--cycle);label("$A$",A,(-1,-1));label("$B$",B,(1,-1));label("$C$",C,(1,1));label("$D=(2,5)$",D,(-1,1));
\end{asy}
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));

\end{asy}
