The asymptotes are $y=\pm\dfrac bax$. Each makes the angle $\alpha$ with the $x$-axis, where $\tan\alpha=\dfrac ba$, and the angle between them containing the hyperbola's branches is $2\alpha$. Since $a>b$, $\tan\alpha<1$, so $\alpha<45^\circ$ and $2\alpha<90^\circ$. The (acute) angle between the asymptotes is therefore $2\alpha=60^\circ$, so $\alpha=30^\circ$ and
\[\dfrac ba=\tan30^\circ=\dfrac{1}{\sqrt3},\qquad\dfrac ab=\boxed{\sqrt3}.\]

\begin{asy}
usepackage("amsmath");
size(6cm);
defaultpen(fontsize(10pt));
import graph;
real a=sqrt(3),bq=1;
path br(real s){guide g;for(real t=-1.6;t<=1.6;t+=0.05)g=g..(s*a*cosh(t),bq*sinh(t));return g;}
draw(br(1),linewidth(0.8));draw(br(-1),linewidth(0.8));
draw((-4.5,-4.5/a)--(4.5,4.5/a),dashed);draw((-4.5,4.5/a)--(4.5,-4.5/a),dashed);
draw((-4.8,0)--(4.8,0),gray,Arrow);draw((0,-2.8)--(0,2.8),gray,Arrow);
draw(arc((0,0),0.7,-30,30),Arrows);
label("$60^\circ$",(0.72,0),E,fontsize(8pt));
label("$y=\dfrac{b}{a}x$",(4.5,4.5/a),E,fontsize(9pt));label("$y=-\dfrac{b}{a}x$",(4.5,-4.5/a),E,fontsize(9pt));
label("$x$",(4.8,0),E);label("$y$",(0,2.8),N);
\end{asy}
