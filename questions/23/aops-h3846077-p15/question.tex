\begin{enumerate}
\item A function $f:\mathbb N\to\mathbb N$, with $\mathbb N=\{1,2,\ldots\}$, satisfies
\[
\sum_{j=1}^n f(j)^3=2\left(\sum_{j=1}^n f(j)\right)^2
\]
for every $n$. Find $f(1),f(2),f(3)$. Show that if $f(j)=2j$ for $1\le j\le k$, then $f(k+1)=2(k+1)$. Determine $f(2016)$ and prove uniqueness.
\item Determine all functions $g:\mathbb N\to\mathbb N$ with $g(2)=4$ and
\[
\sum_{j=1}^n g(j)^3=\dfrac{1}{2}\bigl(g(n)g(n+1)\bigr)^2
\]
for every $n$.
\end{enumerate}
