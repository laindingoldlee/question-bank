\begin{enumerate}
\item At $n=1$, positivity gives $f(1)^3=2f(1)^2$, so $f(1)=2$. Suppose the first $k$ values are $2j$, with total $S=k(k+1)$. Subtract the identity at $k$ from the identity at $k+1$, writing $t=f(k+1)>0$:
\[
t^3=2((S+t)^2-S^2)=4St+2t^2.
\]
Divide by $t$ and factor:
\[
(t-2k-2)(t+2k)=0.
\]
The positive root is $t=2k+2$. Induction gives the unique possible function $f(n)=2n$. It satisfies the identity because $\sum_{j=1}^n j^3=(\dfrac{n(n+1)}{2})^2$. Thus $\boxed{f(1)=2,f(2)=4,f(3)=6,f(2016)=4032}$.
\item The identity at $n=1$ gives $g(1)^3=8g(1)^2$, hence $g(1)=8$. At $n=2$, it gives $8^3+4^3=8g(3)^2$, so $g(3)^2=72$. This is not an integer square, contradicting the codomain. Therefore $\boxed{\text{no such function }g\text{ exists}}$.
\end{enumerate}
