The area is $\dfrac{1}{2}\cdot1984\cdot2016\sin\theta$, where $\theta$ is the angle between them; it is largest when $\theta=90^\circ$: $\dfrac{1984\cdot2016}{2}=\boxed{1999872}$.
\begin{asy}
usepackage("amsmath");
size(3cm);
defaultpen(fontsize(10pt));
pair A=(0,0),B=(4,0); draw(A--B,linewidth(1)); draw(A--(0,3.9),blue+linewidth(1)); draw(B--(0,3.9),blue); draw(A--3.9*dir(60),gray+dashed); draw(3.9*dir(60)--B,gray+dashed);
draw((0.25,0)--(0.25,0.25)--(0,0.25)); label("$1984$",(2,0),S); label("$2016$",(0,1.95),W); label("$\theta$",A,6*dir(40));
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
