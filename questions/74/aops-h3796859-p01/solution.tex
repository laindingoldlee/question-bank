Each exterior angle supplementary to an interior angle is $180^\circ$ minus that angle. Thus the sum is $5\cdot180^\circ-(72+103+102+104+159)^\circ=\boxed{360^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair[] P;for(int i=0;i<5;++i)P.push(dir(90+72*i));for(int i=0;i<5;++i){draw(P[i]--P[(i+1)%5]);draw(P[(i+1)%5]--(P[(i+1)%5]+.4*(P[(i+1)%5]-P[i])),dashed);}label("exterior turns total 360 degrees",(0,-1.5));
\end{asy}
