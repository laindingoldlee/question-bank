If its sides are $x,y>0$, then $x+y=12$. Since $(x-y)^2\ge0$, $xy\le\dfrac{(x+y)^2}{4}=36$, with equality at $x=y=6$. Thus the largest area is $\boxed{36}$.

usepackage("amsmath");

pair[] pts={(0,0),(6,0),(6,6),(0,6)};string[] names={"$A$","$B$","$C$","$D$"};path boundary=pts[0];for(int j=1;j<pts.length;++j)boundary=boundary--pts[j];draw(boundary--cycle);for(int j=0;j<pts.length;++j)label(names[j],pts[j],dir(90+360*j/pts.length));
\end{asy}
