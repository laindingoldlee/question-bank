Put $B=(0,0)$, $C=(L,0)$, $A=(u,h)$. Then $D=(\dfrac{L}{2},0)$, line $EF$ has height $\dfrac{h}{2}$, and $X=(\dfrac{L}{2},\dfrac{h}{2})$. Extending $AX$ to height zero gives $Y=(L-u,0)$, so $|L-2u|=2$. The distance equations give $AC^2-AB^2=L^2-2Lu=11$, hence $L|L-2u|=11$. Thus $L=\dfrac{11}{2}$ and $p+q=\boxed{13}$.

usepackage("amsmath");

real h=3*sqrt(39)/4;pair A=(7/4,h),Bp=(0,0),C=(11/2,0),D=(11/4,0),Ep=(A+C)/2,F=A/2,X=(11/4,h/2),Y=(15/4,0);draw(A--Bp--C--cycle);draw(Ep--F);draw(D--X^^A--Y,blue);label("$A$",A,N);label("$B$",Bp,SW);label("$C$",C,SE);label("$D$",D,S);label("$E$",Ep,NE);label("$F$",F,NW);label("$X$",X,N);label("$Y$",Y,S);
\end{asy}
