Sine is defined for every real input, so the domain is $\boxed{\mathbb R}$. Since $\sin(-x)=-\sin x$, $f(x)=-\sin^2x$. The square of sine attains every value in $[0,1]$, so the range is $\boxed{[-1,0]}$.
\begin{asy}
usepackage("amsmath");
import graph;
size(7cm);
draw((-pi-0.3,0)--(pi+0.3,0),Arrow);
draw((0,-1.3)--(0,0.4),Arrow);
draw((-pi,-1)--(pi,-1),gray+dashed);
draw(graph(new real(real x){return -sin(x)^2;},-pi,pi),blue);
label("$x$",(pi+0.3,0),S);
label("$y$",(0,0.4),W);
label("$-\pi$",(-pi,0),N);
label("$\pi$",(pi,0),N);
label("$0$",(0,0),NE);
label("$-1$",(0,-1),SW);
label("$y=-\sin^2x$",(pi/2,-1.2),S);
\end{asy}
