The $y$-coordinate of the midpoint is the average of the endpoints' $y$-coordinates:
\[
\dfrac{5+y}{2}=2\qquad\Longrightarrow\qquad y=\boxed{-1}.
\]
(The $x$-coordinates agree: $\dfrac{3+(-4)}{2}=-0.5$.)

\begin{asy}
usepackage("amsmath");
size(5cm);
defaultpen(fontsize(10pt));
pair P=(3,5),Q=(-4,-1),M=(-0.5,2);
draw((-5.5,0)--(4.5,0),gray,Arrow(4)); draw((0,-2)--(0,6.3),gray,Arrow(4));
label("$x$",(4.5,0),S,gray); label("$y$",(0,6.3),W,gray);
draw(P--Q,linewidth(0.9));
dot(P);dot(Q);dot(M,red);
label("$(3,5)$",P,E); label("$(-4,y)=(-4,-1)$",Q,S); label("$(-0.5,2)$",M,NW,red);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
