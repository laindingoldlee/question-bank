Taking base-ten logarithms gives $Y=x\log_{10}2$. This is a straight line through the origin with positive slope $\boxed{\log_{10}2}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
draw((-1,0)--(4,0),Arrow);draw((0,-.5)--(0,1.6),Arrow);draw((-1,-log(2)/log(10))--(4,4*log(2)/log(10)),blue);label("$x$",(4,0),S);label("$Y=\log_{10}y$",(0,1.6),N);label("$Y=x\log_{10}2$",(3,1.3));
\end{asy}
