There are $\binom{10}{5}=252$ unrestricted paths. A path through $(2,7)$ consists of one of $\binom51=5$ first segments and one of $\binom51=5$ final segments, giving 25. Subtract to obtain $\boxed{227}$.
\begin{asy}
import geometry; usepackage("amsmath"); size(240);
for(int x=1;x<=6;++x)draw((x,3)--(x,8),gray);for(int y=3;y<=8;++y)draw((1,y)--(6,y),gray);dot((1,3));dot((6,8));dot((2,7),red);label("$(1,3)$",(1,3),SW);label("$(6,8)$",(6,8),NE);label("$(2,7)$",(2,7),NW);
\end{asy}
