Let the hexagon side be $s$. Using its standard horizontal orientation, the left and right vertices are separated by $2s$, so $s=2$. The half-height is $\dfrac{\sqrt3}{2}s=\sqrt3$. The horizontal segments $AQ,DS$ each have length $\dfrac{3}{2}s=3$, the vertical segments $QC,SF$ each have length $\sqrt3$, and $CD,FA$ each have length $s=2$. Thus the perimeter is $\boxed{10+2\sqrt3}$.

real h=sqrt(3);pair A=(-1,h),B=(1,h),C=(2,0),D=(1,-h),ptE=(-1,-h),F=(-2,0),P=(-2,h),Q=(2,h),RR=(2,-h),ptS=(-2,-h);draw(P--Q--RR--ptS--cycle,gray);draw(A--B--C--D--ptE--F--cycle);draw(A--Q--C--D--ptS--F--cycle,blue);label("$A$",A,N);label("$B$",B,N);label("$C$",C,E);label("$D$",D,S);label("$E$",ptE,S);label("$F$",F,W);label("$P$",P,NW);label("$Q$",Q,NE);label("$R$",RR,SE);label("$S$",ptS,SW);
\end{asy}
