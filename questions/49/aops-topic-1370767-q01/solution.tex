Side $AB$ is horizontal (both points have $y=1$), so take it as the base: $AB=6-1=5$. The height is the vertical distance from $C$ to the line $y=1$, which is $7-1=6$.

\begin{asy}
usepackage("amsmath");
size(5cm);
defaultpen(fontsize(10pt));
path ra(pair P,pair u,pair v,real s=0.2){return P+s*unit(u)--P+s*unit(u)+s*unit(v)--P+s*unit(v);}
import graph;
pair A=(1,1),B=(6,1),C=(3,7),H=(3,1);
for(int i=0;i<=7;++i) draw((i,0)--(i,8),gray(0.85));
for(int j=0;j<=8;++j) draw((0,j)--(7,j),gray(0.85));
draw((0,0)--(7.4,0),Arrow(5)); draw((0,0)--(0,8.4),Arrow(5));
draw(A--B--C--cycle,linewidth(0.9));
draw(C--H,red+dashed); draw(ra(H,B-H,C-H,0.35),red);
dot(A);dot(B);dot(C);
label("$A(1,1)$",A,S); label("$B(6,1)$",B,S); label("$C(3,7)$",C,N);
label("$5$",(3.5,1),S); label("$6$",(3,4),W,red);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}

[ABC]=\dfrac{1}{2}\cdot5\cdot6=\boxed{15}.
\]
