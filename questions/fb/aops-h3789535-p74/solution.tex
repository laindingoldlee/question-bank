The distance from $(x,y)$ to the origin is $\sqrt{x^2+y^2}$. Thus the equation describes a $\boxed{\text{circle centred at }(0,0)\text{ with radius }2\sqrt3}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
draw((-4,0)--(4,0),Arrow);draw((0,-4)--(0,4),Arrow);draw(circle((0,0),sqrt(12)),blue);draw((0,0)--sqrt(12)*dir(45),dashed);label("$2\sqrt3$",sqrt(3)*dir(45),NW);label("O",(0,0),SW);label("$x$",(4,0),E);label("$y$",(0,4),N);
\end{asy}
