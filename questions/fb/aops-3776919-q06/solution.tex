Order the integer sides $a\le b\le c$. Triangle inequality gives $c<\dfrac{7}{2}$, so $c\le3$, while the perimeter forces $c\ge3$. Thus $c=3$ and $a+b=4$, yielding $(a,b)=(1,3)$ or $(2,2)$. There are $\boxed2$ triangles.
\begin{asy}
import graph;
usepackage("amsmath");
import geometry;
size(7cm);
pair[] p={(0,0),(3,0),(1.5,sqrt(7)/2)}; pair labelCentre=(0,0);for(pair labelVertex:p)labelCentre+=labelVertex;labelCentre/=p.length; draw(p[0]--p[1]--p[2]--cycle); dot(p[0]);label("$A$",p[0],unit(p[0]-labelCentre));dot(p[1]);label("$B$",p[1],unit(p[1]-labelCentre));dot(p[2]);label("$C$",p[2],unit(p[2]-labelCentre));
\end{asy}
