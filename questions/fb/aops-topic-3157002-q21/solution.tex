Take $ B=(0,0),C=(17,0),A=(x,y)$. Then $ x^2+y^2=400 $, while $ MC^2=(\dfrac{x}{2}-17)^2+\dfrac{y^2}{4}=100-17x+289=289 $. Consequently $ x=\dfrac{100}{17}$, which lies between 0 and 17, and $\boxed{BP=\dfrac{100}{17}}$.
\begin{asy}
usepackage("amsmath");

pair B=(0,0),C=(17,0),A=(100/17,sqrt(400-(100/17)^2)),M=A/2,P=(A.x,0);draw(A--B--C--cycle);draw(M--C^^A--P,dashed);label("$A$",A,N);label("$B$",B,SW);label("$C$",C,SE);label("$M$",M,W);label("$P$",P,S);
\end{asy}
