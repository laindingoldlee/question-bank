\begin{enumerate}
\item[(a)] Let $c$ and $g$ be positive integers with $d=\gcd(c,g)$, and let $b$, $f$ be integers. Prove that the system
\[
x\equiv b\pmod c,\qquad x\equiv f\pmod g
\]
has a solution if and only if $d\mid b-f$, and that in that case its solutions form exactly one residue class modulo $\operatorname{lcm}(c,g)$.
\item[(b)] Solve $x\equiv3\pmod8$, $x\equiv5\pmod9$.
\item[(c)] Does the system $x\equiv1\pmod6$, $x\equiv4\pmod{10}$ have a solution?
\end{enumerate}
