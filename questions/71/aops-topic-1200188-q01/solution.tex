The exterior angles of any convex polygon add up to $360^\circ$, and in a regular pentagon they are equal, so each is $\dfrac{360^\circ}{5}=\boxed{72^\circ}$.

\begin{asy}
usepackage("amsmath");
size(3cm);
defaultpen(fontsize(10pt));
pair[] P; for(int k=0;k<5;++k) P.push(dir(90+72*k)); draw(P[0]--P[1]--P[2]--P[3]--P[4]--cycle,linewidth(0.9));
draw(P[0]--(P[0]+0.8*(P[0]-P[4])),dashed); draw(arc(P[0],0.3,degrees(P[0]-P[4]),degrees(P[1]-P[0])),red); label("$72^\circ$",P[0]+(-0.45,0.05),W);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
