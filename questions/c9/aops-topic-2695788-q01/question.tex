\begin{enumerate}
\item[(a)] For $x\ge1$, rewrite $\sqrt x-\sqrt{x-1}$ without a difference of square roots.
\item[(b)] Deduce that for every $x>1$,
\[\dfrac{1}{2\sqrt x}<\sqrt x-\sqrt{x-1}<\dfrac{1}{2\sqrt{x-1}}.\]
\item[(c)] Hence find the integer part of $S=\dfrac{1}{\sqrt1}+\dfrac{1}{\sqrt2}+\cdots+\dfrac{1}{\sqrt{10000}}$.
\end{enumerate}
