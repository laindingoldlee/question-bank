Expanding cancels every cross term and leaves $a^2+b^2+c^2$. By the three-dimensional Pythagorean theorem, this is the squared space diagonal, $10^2=\boxed{100}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),Bp=(3,0),C=(3,2),D=(0,2),v=(1,1);draw(A--Bp--C--D--cycle);draw((A+v)--(Bp+v)--(C+v)--(D+v)--cycle);draw(A--(A+v));draw(Bp--(Bp+v));draw(C--(C+v));draw(D--(D+v));draw(A--(C+v),blue);label("$10$",(A+C+v)/2,N);label("$a$",(1.5,0),S);label("$b$",(3,1),E);label("$c$",(.5,.5),W);
\end{asy}
