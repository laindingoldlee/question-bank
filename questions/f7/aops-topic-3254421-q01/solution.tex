Extend $AM$ through $M$ to $N$ with $AN=AC$. Since $AM=\dfrac{1}{2}AC$, $M$ is the midpoint of $AN$ as well as $BC$. Thus quadrilateral $ABNC$ is a parallelogram, so $AB\parallel CN$. Triangle $ACN$ is isosceles with vertex angle $\angle CAN=32^\circ$, giving base angle $\angle ACN=74^\circ$. The angle between $AC$ and its parallel-to-$CN$ side $AB$ is supplementary to that base angle, hence $\boxed{\angle BAC=106^\circ}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),C=(2,0),Np=2*dir(32),M=Np/2,B=Np-C;draw(A--B--C--cycle);draw(A--Np,dashed);draw(C--Np--B,dashed);dot(M);label("$A$",A,SW);label("$B$",B,NW);label("$C$",C,SE);label("$N$",Np,NE);label("$M$",M,NW);label("$32^\circ$",(0.6,0.13));
\end{asy}
