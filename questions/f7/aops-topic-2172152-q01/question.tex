\begin{enumerate}
\item[(a)] How many subsets of $\{1,2,3,4\}$ contain the element $3$?
\item[(b)] Let $n$ be a positive integer and $k\in\{1,2,\ldots,n\}$. Prove that exactly $2^{n-1}$ subsets of $\{1,2,\ldots,n\}$ contain $k$.
\end{enumerate}
