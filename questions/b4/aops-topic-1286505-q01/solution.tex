The scale factor is $3$, and areas scale by its square, $9$. The area is $5\cdot9=\boxed{45\text{ m}^2}$.

\begin{asy}
usepackage("amsmath");
size(6cm);
defaultpen(fontsize(10pt));
pair A=(0,0),B=(3,0),C=(3,0)+3*dir(35),D=3*dir(35); draw(A--B--C--D--cycle,linewidth(0.9)); label("$3$",(1.5,0),S);
pair A2=(4.5,0); draw(shift(A2)*scale(3)*(A--B--C--D--cycle),linewidth(0.9)); label("$9$",A2+(4.5,0),S);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
