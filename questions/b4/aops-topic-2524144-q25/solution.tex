The graph is the circle with centre $(2,-3)$ and radius $\sqrt8=2\sqrt2$. It encloses a disc of area $\pi\left(2\sqrt2\right)^2=\boxed{8\pi}$.

\begin{asy}
usepackage("amsmath");
size(4cm);
defaultpen(fontsize(10pt));
pair O=(2,-3);real r=2*sqrt(2);
fill(circle(O,r),lightgray);
draw((-1.5,0)--(5.5,0),Arrow(4));draw((0,-6.5)--(0,1.5),Arrow(4));
draw(circle(O,r),linewidth(0.8));
dot(O);label("$(2,-3)$",O,S);draw(O--O+r*dir(45));label("$2\sqrt2$",O+0.5*r*dir(45),NW);
label("$x$",(5.5,0),S);label("$y$",(0,1.5),W);
\end{asy}
