In triangle $ABC$, $AB=2$ and $AC=3$. Squares I, II and III are drawn outward on the sides $AB$, $BC$ and $CA$. At each vertex of the triangle, the two squares that meet there, together with the vertex, determine a ``gap'' triangle: at $A$ it is the triangle formed by $A$ and the vertex of square I next to $A$ and the vertex of square III next to $A$, and similarly at $B$ and at $C$. The three gap triangles are shaded, as in the figure (the shape of the triangle is not fixed).

\begin{asy}
usepackage("amsmath");
size(5cm);
defaultpen(fontsize(10pt));
pair A=(0,0), B=(2,0), C=3*dir(68);
transform t=rotate(-90);
pair E=A+t*(B-A), B1=B+t*(B-A);
pair B2=B+t*(C-B), C2=C+t*(C-B);
pair C3=C+t*(A-C), D=A+t*(A-C);
filldraw(A--B--B1--E--cycle,white,gray);
filldraw(B--C--C2--B2--cycle,white,gray);
filldraw(C--A--D--C3--cycle,white,gray);
filldraw(A--E--D--cycle,paleblue,blue);
filldraw(B--B1--B2--cycle,paleblue,blue);
filldraw(C--C2--C3--cycle,paleblue,blue);
draw(A--B--C--cycle,linewidth(1));
label("II",(B+C2)/2); label("III",(C+D)/2);
dot(A); dot(B); dot(C);
label("$B$",B,dir(-20)); label("$C$",C,N);
label("$2$",(A+B)/2,N); label("$3$",(A+C)/2,E);
label("I",(A+B1)/2);
label("$A$",A,dir(205));
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}

\begin{enumerate}
\item[(a)] Show that each shaded triangle has the same area as triangle $ABC$.
\item[(b)] Find the greatest possible value of the total shaded area.
\end{enumerate}
