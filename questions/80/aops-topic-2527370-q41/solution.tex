A unit cube has exactly two painted faces when it lies on an edge of the big cube but not at a corner. Each of the $12$ edges has $7-2=5$ such cubes, so there are $12\cdot5=\boxed{60}$.

In the sketch, the cubes with two painted faces along the three visible edges through the front top right corner are shaded.

\begin{asy}
usepackage("amsmath");
size(5cm);
defaultpen(fontsize(10pt));
real n=7;
pair P(real x,real y,real z){return (x+0.5*y,z+0.4*y);}
for(int i=1;i<=5;++i){
 fill(P(i,0,6)--P(i+1,0,6)--P(i+1,0,7)--P(i,0,7)--cycle,lightred);
 fill(P(6,0,i)--P(7,0,i)--P(7,0,i+1)--P(6,0,i+1)--cycle,lightred);
 fill(P(i,0,7)--P(i+1,0,7)--P(i+1,1,7)--P(i,1,7)--cycle,lightred);
 fill(P(7,0,i)--P(7,1,i)--P(7,1,i+1)--P(7,0,i+1)--cycle,lightred);
 fill(P(7,i,7)--P(7,i+1,7)--P(6,i+1,7)--P(6,i,7)--cycle,lightred);
 fill(P(7,i,7)--P(7,i+1,7)--P(7,i+1,6)--P(7,i,6)--cycle,lightred);
}
for(int i=0;i<=7;++i){
 draw(P(i,0,0)--P(i,0,7),gray);draw(P(0,0,i)--P(7,0,i),gray);
 draw(P(7,i,0)--P(7,i,7),gray);draw(P(7,0,i)--P(7,7,i),gray);
 draw(P(0,i,7)--P(7,i,7),gray);draw(P(i,0,7)--P(i,7,7),gray);
}
draw(P(0,0,0)--P(7,0,0)--P(7,0,7)--P(0,0,7)--cycle,linewidth(0.8));
draw(P(7,0,0)--P(7,7,0)--P(7,7,7)--P(7,0,7),linewidth(0.8));
draw(P(0,0,7)--P(0,7,7)--P(7,7,7),linewidth(0.8));
label("$7$",P(3.5,0,0),S);
\end{asy}
