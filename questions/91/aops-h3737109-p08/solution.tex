Take $A=(0,0),B=(8,0),C=(8,4),D=(0,4)$. Then $E=(4,4)$ and $F=(6,4)$. Subtract triangle $BCF$, of area $\dfrac{1}{2}\cdot4\cdot2=4$, from the rectangle of area $32$. Thus $[ABFD]=\boxed{28}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");

defaultpen(fontsize(9pt));
pair A=(0,0),B=(8,0),C=(8,4),D=(0,4),E=(4,4),F=(6,4);filldraw(A--B--F--D--cycle,lightblue);draw(B--C--D);label("A",A,SW);label("B",B,SE);label("C",C,NE);label("D",D,NW);label("E",E,N);label("F",F,N);label("8",(4,0),S);label("4",(8,2),(1,0));
\end{asy}
