\begin{enumerate}
\item[(a)] Prove that for every odd positive integer $n$,
\[
x^n+1=(x+1)\left(x^{n-1}-x^{n-2}+x^{n-3}-\cdots-x+1\right).
\]
\item[(b)] Deduce that if $n$ has an odd divisor $m>1$, then $x^n+1$ has the factor $x^{\dfrac {n}{m}}+1$. Factor $x^6+1$ in this way.
\item[(c)] Show that $x^2+1$ (more generally $x^{n}+1$ for even $n$) has no factor $x-c$ with $c$ real.
\end{enumerate}
