The unbent lower piece has length $50-x$. The bent piece of length $x$ is the hypotenuse of a right triangle with other legs $50-x$ and $10$. Thus $x^2=(50-x)^2+100$, giving $100x=2600$ and $\boxed{x=26\text{ m}}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,0),B=(0,24),C=(10,0);draw(A--B--C--A);draw(B--(0,50),dashed);label("base",A,SW);label("bend",B,W);label("top",C,SE);label("$50-x$",(A+B)/2,W);label("$x$",(B+C)/2,E);label("10",(A+C)/2,S);
\end{asy}
