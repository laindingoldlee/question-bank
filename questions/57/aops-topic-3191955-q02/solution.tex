The acute angles sum to $90^\circ$, so the other is $90^\circ-35^\circ=\boxed{55^\circ}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair[] pts={(0,0),(4,0),(0,4*tan(35*pi/180))}; string[] names={"$A$","$B$","$C$"}; path boundary=pts[0]; for(int j=1;j<pts.length;++j)boundary=boundary--pts[j];draw(boundary--cycle); for(int i=0;i<pts.length;++i) label(names[i],pts[i],dir(90+360*i/pts.length));
\end{asy}
