If $f(x)=x+1$ and $g(x)=x^2$, find
\begin{enumerate}
\item[(a)] $(f\circ g)(x)$;
\item[(b)] $(f\circ g)(-2)$;
\item[(c)] $(f\circ g)(0)$;
\item[(d)] $(f\circ g)(5)$;
\item[(e)] $(f\circ g)(10)$.
\end{enumerate}
