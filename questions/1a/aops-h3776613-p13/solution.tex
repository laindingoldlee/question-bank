Let $AB=c,BC=a$. The angle-bisector length formula yields $BD=\dfrac{2ac\cos60^\circ}{a+c}=\dfrac{ac}{100}$. Therefore $\boxed{ac=2100}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair B=(0,0),A=30*dir(120),C=(70,0),D=(70*A+30*C)/100;draw(A--B--C--cycle);draw(B--D,blue);label("$A$",A,N);label("$B$",B,SW);label("$C$",C,SE);label("$D$",D,NE);
\end{asy}
