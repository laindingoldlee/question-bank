Each of $b,c,d,e$ divides something, so none of them is $0$; and $a\ne0$ as the leading digit. So all five digits are nonzero and different.

\textbf{Step 1: $b\mid a$, $b\ne a$.} The pairs are
\[
(a,b)=(2,1),(3,1),\dots,(9,1),\ (4,2),(6,2),(8,2),\ (6,3),(9,3),\ (8,4).
\]

\textbf{Step 2: $c\mid\overline{ab}$, then $d\mid\overline{abc}$, then $e\mid\overline{abcd}$}, always with new digits. Work through each pair (a dash means the branch dies):
\begin{itemize}
\item $\overline{ab}=21$: $c\in\{3,7\}$; $213=3\cdot71$ and $217=7\cdot31$ have no new digit divisor $d$ (we need $d\notin\{2,1,c\}$). Dead.
\item $31,41,61,71$: no new digit divides them (they are prime). Dead. $81$: $c\in\{3,9\}$; $813$: no new $d$; $819=9\cdot91$: $d\in\{3,7\}$, then $8193$ and $8197$ have no new digit divisor $e$. Dead. $91=7\cdot13$: $c=7$; $917=7\cdot131$: no new $d$. Dead.
\item $51$: $c=3$; $513=27\cdot19$: $d=9$; $5139$: its digit divisors are $1,3,9$, none new. Dead.
\item $42$: $c\in\{1,3,6,7\}$.
  \begin{itemize}
  \item $421$ is prime: dead. $427=7\cdot61$: $d=1$ only, $4271$ has no new divisor $e$: dead.
  \item $423=9\cdot47$: $d\in\{1,9\}$. $4231$: no new $e$. $4239=9\cdot471$: $e=1$, giving $\mathbf{42391}$.
  \item $426=6\cdot71$: $d\in\{1,3\}$. $4261$: no new $e$. $4263=3\cdot7^2\cdot29$: $e\in\{1,7\}$, giving $\mathbf{42631}$, $\mathbf{42637}$.
  \end{itemize}
\item $62$: $c=1$; $621=27\cdot23$: $d\in\{3,9\}$. $6213$: no new $e$. $6219=9\cdot691$: $e=3$, giving $\mathbf{62193}$.
\item $63$: $c\in\{1,7,9\}$. $631$ is prime: dead. $637=7^2\cdot13$: $d=1$, $6371$: no new $e$: dead. $639=9\cdot71$: $d=1$; $6391=7\cdot913$: $e=7$, giving $\mathbf{63917}$.
\item $82$: $c=1$; $821$ is prime: dead.
\item $84$: $c\in\{1,2,3,6,7\}$.
  \begin{itemize}
  \item $841=29^2$: dead. $843=3\cdot281$: $d=1$, $8431$: no new $e$: dead. $847=7\cdot11^2$: $d=1$, $8471$: no new $e$: dead.
  \item $842=2\cdot421$: $d=1$; $8421=3\cdot7\cdot401$: $e\in\{3,7\}$, giving $\mathbf{84213}$, $\mathbf{84217}$.
  \item $846=2\cdot3^2\cdot47$: $d\in\{1,2,3,9\}$.
    $8461$: no new $e$.
    $8462=2\cdot4231$: $e=1$, giving $\mathbf{84621}$.
    $8463=3\cdot7\cdot13\cdot31$: $e\in\{1,7\}$, giving $\mathbf{84631}$, $\mathbf{84637}$.
    $8469=3^2\cdot941$: $e\in\{1,3\}$, giving $\mathbf{84691}$, $\mathbf{84693}$.
  \end{itemize}
\item $93$: $c=1$; $931=7^2\cdot19$: $d=7$; $9317$: digit divisors $1$ and $7$ only, none new. Dead.
\end{itemize}
(Here "no new $d$" or "no new $e$" means that every digit dividing the number is already used.)

So there are exactly twelve such numbers:
\[
\boxed{42391,\ 42631,\ 42637,\ 62193,\ 63917,\ 84213,\ 84217,\ 84621,\ 84631,\ 84637,\ 84691,\ 84693}.
\]
