draw((-7,-0.12)--(-7,0.12));label("$-7$",(-7,0),S);
label("$A(-7,a)$",A,NW);label("$B(c,d)$",B,E);
label("$c+7$",((A.x+B.x)/2,A.y),S);label("$d-a$",(B.x,(A.y+B.y)/2),E);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}

\[
\text{slope}=\dfrac{d-a}{c-(-7)}=\boxed{\dfrac{d-a}{c+7}},
\]
which is defined because $c\ne-7$.
