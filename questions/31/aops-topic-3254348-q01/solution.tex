The centres are at heights $4$ and $9$ above the line and are $4+9=13$ apart. Their vertical separation is $5$ and horizontal separation is $AB$. By the Pythagorean theorem, $AB^2=13^2-5^2=144$, so $\boxed{AB=12}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),B=(12,0),O1=(0,4),O2=(12,9);draw((-5,0)--(22,0));draw(circle(O1,4));draw(circle(O2,9));draw(O1--O2,dashed);draw(A--O1,dashed);draw(B--O2,dashed);draw(O1--(12,4),dashed);label("$A$",A,S);label("$B$",B,S);label("$O_1$",O1,W);label("$O_2$",O2,E);label("$4$",(0,2),W);label("$9$",(12,4.5),E);label("$13$",(O1+O2)/2,NW);
\end{asy}
