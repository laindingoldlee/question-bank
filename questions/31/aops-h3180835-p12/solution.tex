Take $A=(0,0)$, $B=(w,0)$, $C=(w,h)$, $D=(0,h)$, $P=(x,y)$. The first condition gives $y=x\tan40^\circ$. The second gives $w-x=y\tan50^\circ=x$, so $w=2x$. The third gives $h-y=(w-x)\tan60^\circ=\sqrt3x$. Thus vector $DP=(x,-\sqrt3x)$ makes angle $30^\circ$ with the downward vertical ray $DA$. Hence $\boxed{\angle PDA=30^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
real y=tan(40*pi/180),h=y+sqrt(3);pair A=(0,0),B=(2,0),C=(2,h),D=(0,h),P=(1,y);draw(A--B--C--D--cycle);draw(A--P--B);draw(C--P--D);label("A",A,SW);label("B",B,SE);label("C",C,NE);label("D",D,NW);label("P",P,N);
\end{asy}
