The triangle angle sum gives $\angle K=180^\circ-90^\circ-35^\circ=\boxed{55^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair N=(0,0),M=(4,0),K=(0,4*tan(35*pi/180));draw(N--M--K--cycle);label("N",N,SW);label("M",M,SE);label("K",K,(0,1));label("$35^\circ$",M+(-.8,.25));
\end{asy}
