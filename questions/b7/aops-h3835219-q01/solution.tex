\begin{enumerate}
\item Expanding the three squares cancels $-2a^2b^2$ and $-2c^2d^2$ against the corresponding positive terms from $2(ab-cd)^2$, leaving the stated expression.
\item Set $T_k=2^k\sum_{j=1}^{2^{n-k}}P_{k,j}^{2^{n-k}}$. Since $P_{k,j}=P_{k-1,2j-1}P_{k-1,2j}$, the identity $u^2+v^2-2uv=(u-v)^2$ gives
\[
T_{k-1}-T_k=
2^{k-1}\sum_{j=1}^{2^{n-k}}
\left(P_{k-1,2j-1}^{2^{n-k}}-P_{k-1,2j}^{2^{n-k}}\right)^2.
\]
Summing over $k$ telescopes to $T_0-T_n$, exactly the claimed left side.
\item For nonnegative $x_i$, choose $a_i=x_i^{\dfrac{1}{N}}$, meaning the nonnegative $N$th root. Every square is nonnegative, hence
\[
\boxed{\dfrac{x_1+\cdots+x_N}{N}\ge(x_1\cdots x_N)^{\dfrac{1}{N}}}.
\]
Equality holds precisely when all $x_i$ are equal: vanishing squares successively force equality of adjacent nonnegative entries and then of neighboring blocks.
\end{enumerate}
