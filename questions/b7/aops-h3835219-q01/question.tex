Let $N=2^n$ with integer $n\ge1$, and let $a_1,\ldots,a_N$ be real. Define block products
\[
P_{k,j}=\prod_{i=(j-1)2^k+1}^{j2^k}a_i
\quad(0\le k\le n,\ 1\le j\le2^{n-k}).
\]
\begin{enumerate}
\item Verify $a^4+b^4+c^4+d^4-4abcd=(a^2-b^2)^2+(c^2-d^2)^2+2(ab-cd)^2$.
\item Prove
\[
\sum_{i=1}^{2^n}a_i^{2^n}-2^n\prod_{i=1}^{2^n}a_i
=\sum_{k=1}^n2^{k-1}\sum_{j=1}^{2^{n-k}}
\left(P_{k-1,2j-1}^{2^{n-k}}-P_{k-1,2j}^{2^{n-k}}\right)^2.
\]
\item Deduce AM-GM for $N=2^n$ nonnegative numbers.
\end{enumerate}
