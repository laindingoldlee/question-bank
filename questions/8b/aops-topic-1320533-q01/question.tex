\begin{enumerate}
\item[(a)] Let $n$ be a positive integer with an odd divisor $m>1$, say $n=km$. Show that $x^{2n}+y^{2n}$ is divisible by $x^{2k}+y^{2k}$, and write down the factorisation.
\item[(b)] Factor $x^6+y^6$ and $x^{12}+y^{12}$ as products of two non-constant polynomials with integer coefficients.
\item[(c)] Factor $x^4+y^4$ as a product of two quadratic polynomials with real coefficients.
\end{enumerate}
