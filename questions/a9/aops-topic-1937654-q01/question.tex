For integers $j\ge0$ and $k\ge1$ let $S_j(k)=1^j+2^j+\cdots+k^j$ (so $S_0(k)=k$).
\begin{enumerate}
\item[(a)] Prove that for every integer $p\ge1$,
\[
p\,S_{p-1}(k)=(k+1)^p-1-\sum_{i=2}^{p}\dbinom pi S_{p-i}(k).
\]
\item[(b)] A student wrote this identity without the factor $p$ on the left. Show by an example that the student's version is false.
\item[(c)] Use the identity with $p=4$ to find a closed form for $S_3(k)$.
\end{enumerate}
