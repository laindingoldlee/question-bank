$64.8$ lb $=1036.8$ oz, which is $\dfrac{1036.8}{0.6}=1728$ times the small cube. Volume scales as the cube of the edge, and $1728=12^3$, so the edge is $12$ inches $=\boxed{1\text{ foot}}$.

\begin{asy}
usepackage("amsmath");
size(5cm);
defaultpen(fontsize(10pt));
draw(unitsquare); draw((1,0)--(1.4,0.3)--(1.4,1.3)--(1,1)); draw((0,1)--(0.4,1.3)--(1.4,1.3)); label("$1$ in",(0.5,0),S);
draw(shift(2.5,0)*scale(3)*unitsquare); draw((5.5,0)--(6.7,0.9)--(6.7,3.9)--(5.5,3)); draw((2.5,3)--(3.7,3.9)--(6.7,3.9)); label("$12$ in",(4,0),S);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
