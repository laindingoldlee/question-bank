The area of a parallelogram is base times height: $7\cdot14=\boxed{98}$.

\begin{asy}
usepackage("amsmath");
size(2.8cm);
defaultpen(fontsize(10pt));
pair A=(0,0), B=(7,0), C=(11,14), D=(4,14);
draw(A--B--C--D--cycle,linewidth(0.9));
draw(D--(4,0),dashed); draw((4,0)--(4.6,0)--(4.6,0.6)--(4,0.6));
label("$7$",(A+B)/2,S); label("$14$",(4,7),E);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
