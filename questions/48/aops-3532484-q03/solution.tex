Set $A=(0,0)$, $B=(1,0)$, $D=(0,1)$, $E=(\dfrac{1}{2},0)$, $F=(0,\dfrac{1}{2})$. Lines $DE$ and $BF$ have equations $y=1-2x$ and $y=\dfrac{1}{2}-\dfrac{x}{2}$, so $G=(\dfrac{1}{3},\dfrac{1}{3})$. The determinant area is $\dfrac{1}{2}|\det(F-E,G-E)|=\boxed{\dfrac{1}{24}}$.

import graph;import geometry;usepackage("amsmath");size(7cm);
pair A=(0,0),B=(1,0),C=(1,1),D=(0,1),E=(.5,0),F=(0,.5),G=(1/3,1/3);draw(A--B--C--D--cycle);draw(D--E^^B--F);draw(E--F,dashed);label("$A$",A,(-1,-1));label("$B$",B,(1,-1));label("$C$",C,(1,1));label("$D$",D,(-1,1));label("$E$",E,(0,-1));label("$F$",F,(-1,0));label("$G$",G,(0,1));
\end{asy}
