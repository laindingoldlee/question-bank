Let $\mathbf u,\mathbf v$ be unit vectors from $A$ toward $B,C$, respectively. The section formula gives $E=\dfrac{B+2C}{3}$ and $G=\dfrac{D+2F}{3}$. Hence $E-G=\dfrac{(B-D)+2(C-F)}{3}=\dfrac{100\sqrt3}{3}(\mathbf u+\mathbf v)$. Since the angle between the unit vectors is $60^\circ$, $|\mathbf u+\mathbf v|^2=1+1+2\cos60^\circ=3$. Thus $\boxed{EG=100}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,0),B=(240,0),C=160*dir(60),D=B-100*sqrt(3)*(1,0),F=C-50*sqrt(3)*dir(60),E=(B+2*C)/3,G=(D+2*F)/3;draw(A--B--C--cycle);draw(D--F);draw(E--G,blue);label("A",A,SW);label("B",B,SE);label("C",C,N);label("D",D,S);label("F",F,NW);label("E",E,NE);label("G",G,SW);
\end{asy}
