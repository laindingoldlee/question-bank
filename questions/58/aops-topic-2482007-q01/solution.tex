\textbf{(a)} The formula $2x^2-8x+3$ can be evaluated for every real $x$, so the domain is all real numbers: a parabola $y=ax^2+bx+c$ never ``ends'' horizontally. Completing the square,
\[f(x)=2\left(x^2-4x+4\right)-8+3=2(x-2)^2-5.\]
Since $(x-2)^2$ takes every value $\ge0$, $f(x)$ takes every value $\ge-5$, and no smaller value. So
\[\text{domain}=\boxed{\text{all real }x},\qquad\text{range}=\boxed{y\ge-5}.\]

\begin{asy}
usepackage("amsmath");
size(5.5cm);
defaultpen(fontsize(10pt));
import graph;
real k=0.4;
real f(real x){return k*(2x^2-8x+3);}
draw((-1.2,0)--(5.4,0),Arrow(6)); draw((0,-2.6)--(0,3.6),Arrow(6));
label("$x$",(5.4,0),E); label("$y$",(0,3.6),N);
draw(graph(f,-0.45,4.45),blue+linewidth(0.9));
draw((0,-5k)--(5.4,-5k),dashed+gray);
dot((2,-5k)); label("$(2,-5)$",(2,-5k),S);
label("$y=2x^2-8x+3$",(4.45,f(4.45)),E,blue);
label("range: $y\ge-5$",(5.0,-5k),N,fontsize(8pt));
for(int i=1;i<=4;++i){draw((i,-0.08)--(i,0.08)); label("$"+string(i)+"$",(i,0),SE,fontsize(7pt));}
for(int j=-5;j<=5;j+=5){ if(j!=0){draw((-0.08,k*j)--(0.08,k*j)); label("$"+string(j)+"$",(0,k*j),W,fontsize(7pt));}}
\end{asy}

\textbf{(b)} Here the roles of $x$ and $y$ are swapped: $x=(y-2)^2-3$. Every real $y$ gives a point, so the $y$-coordinates are \textbf{all real numbers}, while $x=(y-2)^2-3$ takes exactly the values $x\ge-3$. So the $x$-coordinates form the half-line $\boxed{x\ge-3}$ (this curve is not the graph of a function of $x$).

\begin{asy}
usepackage("amsmath");
size(5.5cm);
defaultpen(fontsize(10pt));
import graph;
real g(real y){return y^2-4y+1;}
draw((-4.5,0)--(8,0),Arrow(6)); draw((0,-1.8)--(0,6),Arrow(6));
label("$x$",(8,0),E); label("$y$",(0,6),N);
path p=graph(g,-1.2,5.2);
draw(reflect((0,0),(1,1))*p,red+linewidth(0.9));
draw((-3,-1.8)--(-3,6),dashed+gray);
dot((-3,2)); label("$(-3,2)$",(-3,2),W);
label("$x=y^2-4y+1$",(4.2,2.6),E,red);
label("$x\ge-3$",(-3,-1.4),W,fontsize(8pt));
\end{asy}
