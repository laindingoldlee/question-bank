The given angles are 70 and 75 degrees. The remaining angle is $180-70-75=\boxed{35^\circ}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),B=(4,0);real x=4*tan(75*pi/180)/(tan(70*pi/180)+tan(75*pi/180));pair C=(x,x*tan(70*pi/180));draw(A--B--C--cycle);label("$70^\circ$",A,S);label("$75^\circ$",B,S);label("$35^\circ$",C,N);
\end{asy}
