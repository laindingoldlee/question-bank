Triangle $BCD$ is right at $C$, with $BC=1,CD=5$, so $BD^2=26$. Since lines $BA,BD$ coincide, the absolute cosine of $\angle ABC$ is $\dfrac{1}{\sqrt{26}}$. Therefore $\boxed{\sec^2\angle ABC=26}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair B=(0,0),C=(1,0),D=(1,5),A=D/sqrt(26);draw(B--C--D--B);draw(A--C);label("$B$",B,SW);label("$C$",C,SE);label("$D$",D,N);label("$A$",A,W);
\end{asy}
