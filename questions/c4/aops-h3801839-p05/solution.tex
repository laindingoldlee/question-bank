Since $AB=BC$, the base angles are equal, each $\dfrac{180^\circ-40^\circ}{2}=70^\circ$. The bisector $BD$ makes $\angle ABD=20^\circ$. In triangle $ABD$, the remaining angle is $180^\circ-70^\circ-20^\circ=\boxed{90^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(-1,0),C=(1,0),B=(0,1/tan(20*pi/180)),D=(0,0);draw(A--B--C--cycle);draw(B--D,blue);label("$A$",A,(-1,-1));label("$B:40^\circ$",B,(0,1));label("$C$",C,(1,-1));label("$D$",D,(0,-1));
\end{asy}
