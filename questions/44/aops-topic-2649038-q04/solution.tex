The room has side $\sqrt[3]{512}=8$ yards $=24$ feet. Since $24=6\cdot4$, the boxes fit exactly in a $6\times6\times6$ array, and no more can fit because the volume is $\dfrac{24^3}{4^3}=216$ boxes. So $\boxed{216}$ boxes fit.

\begin{asy}
usepackage("amsmath");
size(4cm);
defaultpen(fontsize(10pt));
real n=6;pair o=(0.45,0.3);
for(int i=0;i<=6;++i){draw((i,0)--(i,6),gray);draw((0,i)--(6,i),gray);draw((i,6)--(i,6)+6*o,gray);draw((0,6)+i*o--(6,6)+i*o,gray);draw((6,i)--(6,i)+6*o,gray);draw((6,0)+i*o--(6,6)+i*o,gray);}
draw((0,0)--(6,0)--(6,6)--(0,6)--cycle,linewidth(0.8));draw((0,6)--(0,6)+6*o--(6,6)+6*o--(6,0)+6*o--(6,0),linewidth(0.8));draw((6,6)--(6,6)+6*o,linewidth(0.8));
fill((0,0)--(1,0)--(1,1)--(0,1)--cycle,lightgray);
label("$24$ ft",(3,0),S);label("$24$ ft",(6,3)+6*o,E);label("$4$ ft",(0.5,0),S,fontsize(7pt));
\end{asy}
