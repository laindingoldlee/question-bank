\begin{enumerate}
\item[(a)] Show that $x^6-16x^4+16x^2-1=(x+1)(x-1)\left(x^4-15x^2+1\right)$.
\item[(b)] Show that every root $z$ of $x^4-15x^2+1$ satisfies $z^6=224z^2-15$.
\item[(c)] Find the sum of the sixth powers of the six roots of $x^6-16x^4+16x^2-1=0$.
\end{enumerate}
