Use $A=(0,0)$, $B=(6,0)$, $C=(6,6)$, $D=(0,6)$, $P=(2,0)$, $Q=(0,q)$, $R=(6,r)$. Determinant areas give $[DPR]=12+r=15$, so $r=3$, and $[CPQ]=6+2q=15$, so $q=\dfrac{9}{2}$. Lines $DR$ and $CQ$ are $y=6-\dfrac{x}{2}$ and $y=\dfrac{9}{2}+\dfrac{x}{4}$. They meet at $X=(2,5)$, directly above $P$, so $\boxed{PX=5}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");

defaultpen(fontsize(9pt));
pair A=(0,0),B=(6,0),C=(6,6),D=(0,6),P=(2,0),Q=(0,4.5),R=(6,3),X=(2,5);draw(A--B--C--D--cycle);draw(D--P--R);draw(C--P--Q);draw(D--R);draw(C--Q);draw(P--X,blue);label("A",A,SW);label("B",B,SE);label("C",C,NE);label("D",D,NW);label("P",P,S);label("Q",Q,W);label("R",R,E);label("X",X,N);
\end{asy}
