Write $DE=2t$, $AD=h$. Since $BE=CE=t$, triangle $BEC$ is right isosceles, giving $t\sqrt2=9$ and $t^2=\dfrac{81}{2}$. The right triangle $ADB$ gives $h^2+t^2=144$, so $h^2=\dfrac{207}{2}$. The rectangle area is $2th=2\sqrt{\dfrac{81\cdot207}{4}}=\boxed{27\sqrt{23}}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
real t=9/sqrt(2),h=sqrt(207/2);pair A=(0,h),D=(0,0),E0=(2*t,0),F=(2*t,h),B=(t,0),C=(2*t,t);draw(A--D--E0--F--cycle);draw(A--B--C--cycle);label("A",A,NW);label("D",D,SW);label("E",E0,SE);label("F",F,NE);label("B",B,S);label("C",C,E);
\end{asy}
