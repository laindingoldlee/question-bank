Scale and place $E=(0,0)$, $F=(2,0)$, and $N=(1,0)$. The angle at $E$ is $105^\circ$. Intersecting the rays from $E$ at $105^\circ$ and from $F$ at $150^\circ$ gives $D=(\dfrac{1-\sqrt3}{2},\dfrac{1+\sqrt3}{2})$. Thus $D-N=(-\dfrac{1+\sqrt3}{2},\dfrac{1+\sqrt3}{2})$, which makes $135^\circ$ with the positive horizontal ray $NF$. Therefore $\boxed{\angle DNF=135^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair E=(0,0),F=(2,0),N=(1,0),D=((1-sqrt(3))/2,(1+sqrt(3))/2);draw(D--E--F--cycle);draw(D--N,blue);label("D",D,(0,1));label("E",E,SW);label("N",N,S);label("F",F,SE);
\end{asy}
