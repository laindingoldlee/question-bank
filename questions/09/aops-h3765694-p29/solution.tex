Assume the quadrilateral is convex and write $B=(0,0),C=(t,0)$, with $AB=CD=s$. The angle conditions give $A=(-\dfrac{\sqrt3s}{2},\dfrac{s}{2})$ and $D=(t,s)$, after reflecting if necessary. Thus $M=(\dfrac{t}{2},0)$ and $N=(\dfrac{t}{2}-\dfrac{\sqrt3s}{4},\dfrac{3s}{4})$. The absolute slope of $MN$ is $\sqrt3$, so its acute angle with $BC$ is $\boxed{60^\circ}$.

texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair B=(0,0),C=(5,0),A=(-sqrt(3),1),D=(5,2),M=(B+C)/2,N=(A+D)/2;draw(A--B--C--D--cycle);draw(M--N,blue);label("A",A,NW);label("B",B,SW);label("C",C,SE);label("D",D,NE);label("M",M,S);label("N",N,(0,1));
\end{asy}
