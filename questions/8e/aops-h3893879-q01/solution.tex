Scale to $A=(0,0)$, $M=(1,0)$, $B=(2,0)$ and reflect if needed so $C$ is above the axis. The $45^\circ$ condition gives $C=(1+u,u)$, $u>0$. The $30^\circ$ ray from $A$ gives $u=\dfrac{1+u}{\sqrt3}$, so $u=\dfrac{\sqrt3+1}{2}$. Thus the slope of $BC$ is $\dfrac{u}{u-1}=2+\sqrt3=\tan75^\circ$. The interior angle at $B$ is $105^\circ$, so $\angle C=180^\circ-30^\circ-105^\circ=\boxed{45^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,0),M=(1,0),B=(2,0),C=((3+sqrt(3))/2,(1+sqrt(3))/2);draw(A--B--C--cycle);draw(M--C,blue);label("A",A,SW);label("M",M,S);label("B",B,S);label("C",C,N);label("$30^\circ$",(.6,.13));label("$45^\circ$",(1.4,.18));
\end{asy}
