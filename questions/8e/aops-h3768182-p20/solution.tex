The three layers contain $3^2,2^2,1^2$ cubes. Their total is $9+4+1=\boxed{14}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
for(int n=3;n>=1;--n){real ox=4*(3-n);for(int x=0;x<n;++x)for(int y=0;y<n;++y)draw(box((ox+x,y),(ox+x+1,y+1)));label(string(n)+" by "+string(n),(ox+n/2.0,-.5));}
\end{asy}
