The triangle is $50^\circ$--$50^\circ$--$80^\circ$ or $50^\circ$--$65^\circ$--$65^\circ$. The largest difference $x-y$ is $80-50=\boxed{30}$.
\begin{asy}
usepackage("amsmath");
size(5cm);
defaultpen(fontsize(10pt));
pair A=(0,0),B=(4,0),C=(2,2*Tan(50)); pair D=(6,0),E=(10,0),F=(8,2*Tan(65));
draw(A--B--C--cycle,linewidth(1)); draw(D--E--F--cycle,linewidth(1));
label("\small$50^\circ$",A+1.25*dir(18)); label("\small$50^\circ$",B+1.25*dir(162)); label("\small$80^\circ$",C+1.1*S);
label("\small$50^\circ$",F+1.3*S); label("\small$65^\circ$",D+1.25*dir(22)); label("\small$65^\circ$",E+1.25*dir(158));
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
