A book defines the Fibonacci numbers and asks for a proof that
\[
F_0+F_2+F_4+\cdots+F_{2n}=F_{2n+1}\qquad(n\ge0),
\]
with the solution
\[
F_{2n+1}=F_{2n}+F_{2n-1}=F_{2n}+F_{2n-2}+F_{2n-3}=\cdots=F_{2n}+F_{2n-2}+\cdots+F_2+F_0 .
\]
\begin{enumerate}
\item Show that with $F_0=0$, $F_1=1$ and $F_{k+1}=F_k+F_{k-1}$ the claimed identity is false, find the step where the solution goes wrong, and prove the correct identity.
\item Show that the identity is true if the sequence starts $F_0=F_1=1$ instead.
\end{enumerate}
