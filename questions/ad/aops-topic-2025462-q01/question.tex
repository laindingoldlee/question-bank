\begin{enumerate}
\item[(a)] Find the coefficient of $x^6$ in $(1+x+x^2+x^3+\cdots)(1+x+x^2+x^3+\cdots)$.
\item[(b)] Prove that for positive integers $k$ and nonnegative integers $n$, the coefficient of $x^n$ in $(1+x+x^2+\cdots)^k$ is $\dbinom{n+k-1}{k-1}$, and use this to find the coefficient of $x^6$ in $(1+x+x^2+\cdots)^4$.
\end{enumerate}
