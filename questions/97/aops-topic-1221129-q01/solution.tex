The unpainted cubes form the inner box with each dimension reduced by $2$: $16\cdot12\cdot2=384$ and $30\cdot5\cdot3=450$. The difference is $\boxed{66}$.

\begin{asy}
usepackage("amsmath");
size(3cm);
defaultpen(fontsize(10pt));
pair o=(0.45,0.3); draw((0,0)--(4,0)--(4,1.2)--(0,1.2)--cycle); draw((0,1.2)--(0,1.2)+3*o--(4,1.2)+3*o--(4,0)+3*o--(4,0)); draw((4,1.2)--(4,1.2)+3*o);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
