The space diagonal is $\sqrt{1^2+1^2+1^2}=\boxed{\sqrt3}$; all face diagonals and edges are shorter.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair t=(.4,.4);draw((0,0)--(1,0)--(1,0)+t--t--cycle);draw((0,1)--(1,1)--(1,1)+t--(0,1)+t--cycle);for(pair p:new pair[]{(0,0),(1,0),(1.4,.4),(.4,.4)})draw(p--p+(0,1));draw((0,0)--(1.4,1.4),blue);label("side 1",(.5,-.2));
\end{asy}
