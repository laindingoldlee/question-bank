The horizontal and vertical differences are $8$ and $6$, so $AB=\sqrt{8^2+6^2}=\boxed{10}$.
\begin{asy}
import graph;import geometry;usepackage("amsmath");size(7cm);
pair A=(3,-2),B=(-5,4),C=(-5,-2);draw(A--B);draw(A--C--B,dashed);label("$A(3,-2)$",A,(1,0));label("$B(-5,4)$",B,(0,1));label("$8$",(A+C)/2,(0,-1));label("$6$",(B+C)/2,(-1,0));
void ra(pair U,pair V,pair W,real s){pair u=s*unit(U-V),w=s*unit(W-V);draw(V+u--V+u+w--V+w);}
ra(A,C,B,.4);
\end{asy}
