\textbf{(c)} Let the slopes be $m$ and $n$. Translating the lines does not change their slopes or the angle between them, so we may assume both pass through the origin $O$. They meet the vertical line $x=1$ at $P=(1,m)$ and $Q=(1,n)$, and $\angle POQ=90^\circ$.

\begin{asy}
usepackage("amsmath");
size(4.8cm);
defaultpen(fontsize(10pt));
pair O=(0,0);
draw((-2.2,0)--(2.4,0),gray,Arrow(4)); draw((0,-2.6)--(0,2.2),gray,Arrow(4));
draw(-1.8*(1,0.5)--2.2*(1,0.5),blue+linewidth(0.9));
draw(-0.9*(1,-2)--1.25*(1,-2),red+linewidth(0.9));
draw((1,-2.6)--(1,2.0),dashed);
pair P=(1,0.5), Q=(1,-2);
draw(O--P--Q--cycle,linewidth(0.6));
pair u=unit(P), v=unit(Q);
draw(0.22*u--0.22*u+0.22*v--0.22*v);
dot(P); dot(Q); dot(O);
label("$P(1,m)$",P,NE); label("$Q(1,n)$",Q,E); label("$O$",O,NW);
label("$q$",2.2*(1,0.5),N,blue); label("$r$",-0.9*(1,-2),W,red);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}

By the Pythagorean theorem in triangle $POQ$,
\[OP^2+OQ^2=PQ^2:\qquad \left(1+m^2\right)+\left(1+n^2\right)=(m-n)^2=m^2-2mn+n^2 ,\]
so $2=-2mn$ and $mn=-1$. The product is always $-1$. $\blacksquare$

\textbf{(a)} By (c), $x=-1$, so $C=\boxed{(-1,-1)}$.

\textbf{(b)} A rotation of $270^\circ$ clockwise is the same as a rotation of $90^\circ$ counterclockwise, which sends $(u,v)$ to $(-v,u)$. So $C=(-1,-1)$ goes to $\boxed{(1,-1)}$.

\begin{asy}
usepackage("amsmath");
size(5cm);
defaultpen(fontsize(10pt));
pair O=(0,0), C=(-1,-1), D=(1,-1);
draw((-1.8,0)--(1.8,0),gray,Arrow(4)); draw((0,-1.8)--(0,1.4),gray,Arrow(4));
draw(O--C,dashed); draw(O--D,dashed);
draw(arc(O,sqrt(2),225,315),blue,Arrow(5));
dot(C); dot(D); dot(O);
label("$C(-1,-1)$",C,W); label("$C'(1,-1)$",D,E); label("$O$",O,NW);
label("$90^\circ$ counterclockwise",(0,-1.45),S,blue+fontsize(8pt));
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
