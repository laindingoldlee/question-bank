For a real number $r$ and a nonnegative integer $k$, define
\[
\dbinom{r}{k}=\dfrac{r(r-1)(r-2)\cdots(r-k+1)}{k!}.
\]
\begin{enumerate}
\item[(a)] Compute $\dbinom{-5}{3}$.
\item[(b)] Prove that $\dbinom{-n}{k}=(-1)^k\dbinom{n+k-1}{k}$ for every positive integer $n$.
\item[(c)] Show that there is no way to define $(-1)!$, and hence $(-4)!$, as numbers so that $m!=m\cdot(m-1)!$ holds for all integers $m\ge -3$ (with $0!=1$).
\end{enumerate}
