Let the parallelogram be $ABCD$ with $AB=30$, $AD=26$, shorter diagonal $BD=28$ and other diagonal $AC=d$.

\begin{asy}
usepackage("amsmath");
size(7cm);
defaultpen(fontsize(10pt));
real a=30,b=26;real ct=792/1560;real th=aCos(ct);
pair A=(0,0),B=(a,0),D=b*dir(th),C=B+D;
draw(A--B--C--D--cycle,linewidth(0.9));
draw(B--D,blue);draw(A--C,red+dashed);
label("$A$",A,SW);label("$B$",B,SE);label("$C$",C,NE);label("$D$",D,NW);
label("$30$",(A+B)/2,S);label("$26$",(A+D)/2,W);
label("$28$",B+0.3*(D-B),NE,blue);label("$d$",A+0.3*(C-A),SE,red);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}

\textbf{Parallelogram law.} If $\angle DAB=\theta$, then $\angle ABC=180^\circ-\theta$, and the law of cosines in triangles $ABD$ and $ABC$ gives
\[
BD^2=30^2+26^2-2\cdot30\cdot26\cos\theta,\qquad AC^2=30^2+26^2+2\cdot30\cdot26\cos\theta .
\]
Adding, $BD^2+AC^2=2\left(30^2+26^2\right)=3152$. Hence
\[
d^2=3152-28^2=3152-784=2368=64\cdot37,\qquad d=\boxed{8\sqrt{37}}\text{ cm}\approx48.7\text{ cm}.
\]
