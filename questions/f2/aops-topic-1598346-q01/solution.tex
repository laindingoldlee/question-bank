Solve for $y$: $5y=5x-7$, so
\[
y=x-\dfrac{7}{5} .
\]
In the form $y=mx+c$ the slope is $m=\boxed{1}$: moving $1$ unit to the right raises the line by $1$ unit.

\begin{asy}
usepackage("amsmath");
size(5.5cm);
defaultpen(fontsize(10pt));
import graph;
draw((-1.5,0)--(3.2,0),gray,Arrow(TeXHead));draw((0,-2.4)--(0,2.4),gray,Arrow(TeXHead));
label("$x$",(3.2,0),E,gray);label("$y$",(0,2.4),N,gray);
draw((-1,-2.4)--(3.2,1.8),blue+linewidth(0.9));
pair P=(1.4,0),Q=(2.4,1);dot(P);dot(Q);
draw(P--(2.4,0)--Q,red+dashed);
label("$1$",(1.9,0),S,red);label("$1$",(2.4,0.5),E,red);
label("$5x-5y=7$",(3.2,1.8),NW,blue);label("$\left(\dfrac{7}{5},0\right)$",P,NW);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
