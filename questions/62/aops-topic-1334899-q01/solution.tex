The circle has area $\pi\cdot4^2=16\pi$. Each of the $8$ equal sectors (central angle $45^\circ$) has area
\[
\dfrac{16\pi}{8}=2\pi\approx\boxed{6.28}.
\]

\begin{asy}
usepackage("amsmath");
size(4cm);
defaultpen(fontsize(10pt));
for(int k=0;k<8;++k) draw((0,0)--4*dir(45*k));
fill((0,0)--arc((0,0),4,0,45)--cycle,paleblue);
draw(circle((0,0),4),linewidth(0.9));
draw((0,0)--4*dir(0)); draw((0,0)--4*dir(45));
draw(arc((0,0),0.9,0,45)); label("$45^\circ$",1.6*dir(22.5),fontsize(8pt));
label("$4$",(2,0),S);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
