For a positive integer $x$, define
\[g(x)=\lfloor x\rfloor+\left\lfloor\dfrac x2\right\rfloor+\left\lfloor\dfrac x3\right\rfloor+\cdots+\left\lfloor\dfrac xx\right\rfloor.\]
\begin{enumerate}
\item[(a)] Evaluate $g(7)-g(6)$.
\item[(b)] Evaluate $g(160)-g(159)$.
\item[(c)] Find all integers $n\ge2$ such that $g(n)-g(n-1)=11$.
\end{enumerate}
