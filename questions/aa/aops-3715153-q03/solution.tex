The radius is perpendicular to the tangent. In right triangle $OTP$, $OP=\sqrt{5^2+12^2}=\boxed{13\text{ cm}}$.
\begin{asy}
import graph;import geometry;usepackage("amsmath");size(7cm);
pair O=(0,0),T=(0,-5),P=(12,-5);draw(circle(O,5));draw(T--P--O--T);dot(O);label("$O$",O,(-1,0));label("$T$",T,(0,-1));label("$P$",P,(1,0));label("$5$",(O+T)/2,(-1,0));label("$12$",(T+P)/2,(0,-1));
void ra(pair U,pair V,pair W,real s){pair u=s*unit(U-V),w=s*unit(W-V);draw(V+u--V+u+w--V+w);}
ra(O,T,P,.5);
\end{asy}
