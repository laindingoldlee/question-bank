\begin{enumerate}
\item[(a)] Prove that if $|r|<1$, then $a+ar+ar^2+\cdots=\dfrac{a}{1-r}$.
\item[(b)] For $|r|<1$, find the value of the infinite nested expression $1+r\left(1+r\left(1+r\left(1+\cdots\right)\right)\right)$.
\end{enumerate}
