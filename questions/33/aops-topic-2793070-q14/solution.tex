Angles $AMB$ and $ANB$ are right, so $A,M,B,N$ lie on the circle with diameter $AB=3$. The angle $MBN=\angle EBD=36^\circ$ subtends chord $MN$. The chord formula gives $MN=3\sin36^\circ=\boxed{3\cos54^\circ}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=dir(90),B=dir(162),C=dir(234),D=dir(306),Ee=dir(18);pair M=B+dot(A-B,Ee-B)/dot(Ee-B,Ee-B)*(Ee-B),Nn=B+dot(A-B,D-B)/dot(D-B,D-B)*(D-B);draw(A--B--C--D--Ee--cycle);draw(B--Ee);draw(B--D);draw(A--M--Nn--A,dashed);label("$A$",A,N);label("$B$",B,W);label("$C$",C,SW);label("$D$",D,SE);label("$E$",Ee,E);label("$M$",M,N);label("$N$",Nn,S);
\end{asy}
