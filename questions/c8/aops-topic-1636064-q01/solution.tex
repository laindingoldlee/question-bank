The slope is
\[
m=\dfrac{1-(-5)}{4-3}=6 .
\]
Using the point $P(4,1)$: $y-1=6(x-4)$, so $y=6x-24+1$, that is,
\[
\boxed{y=6x-23}.
\]
Check with $Q$: $6\cdot3-23=-5$.

\begin{asy}
usepackage("amsmath");
size(5cm);
defaultpen(fontsize(10pt));
draw((-1,0)--(6,0),gray,Arrow(TeXHead));draw((0,-8)--(0,3),gray,Arrow(TeXHead));
label("$x$",(6,0),E,gray);label("$y$",(0,3),N,gray);
draw((2.5,-8)--(4.3,2.8),blue+linewidth(0.9));
pair P=(4,1),Q=(3,-5);
draw(Q--(4,-5)--P,red+dashed);
label("$1$",(3.55,-5),N,red);label("$6$",(4,-2),E,red);
dot(P);dot(Q);
label("$P(4,1)$",P,W);label("$Q(3,-5)$",Q,SE);
label("$y=6x-23$",(3.0,-7.3),E,blue);
for(int i=1;i<=5;++i){draw((i,-0.12)--(i,0.12),gray);}
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
