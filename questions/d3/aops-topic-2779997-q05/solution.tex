The original area is $\dfrac{1}{2}\cdot6\cdot8=24 $. The midpoint triangle is similar with scale factor $\dfrac{1}{2}$, so its area is $\dfrac{1}{4}\cdot24=\boxed6 $.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,6),B=(0,0),C=(8,0),X=(A+B)/2,Y=(B+C)/2,Z=(C+A)/2;draw(A--B--C--cycle);draw(X--Y--Z--cycle,blue);label("$A$",A,N);label("$B$",B,SW);label("$C$",C,SE);label("$X$",X,W);label("$Y$",Y,S);label("$Z$",Z,NE);
\end{asy}
