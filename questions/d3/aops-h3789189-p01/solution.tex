The minute hand sweeps $10\cdot6=60^\circ$ and the hour hand sweeps $10\cdot\dfrac{1}{2}=5^\circ$. Their total movement is $\boxed{65^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
draw(unitcircle);draw((0,0)--dir(90),gray);draw((0,0)--dir(30),blue);draw((0,0)--.65*dir(85),red);draw(arc((0,0),.8,30,90),blue);label("minute: 60 degrees",(1.2,.5),E);label("hour: 5 degrees",(-.6,.5),W);
\end{asy}
