\begin{enumerate}
\item[(a)] Prove that for every positive integer $n$,
\[a^n-b^n=(a-b)\left(a^{n-1}+a^{n-2}b+\cdots+ab^{n-2}+b^{n-1}\right).\]
\item[(b)] Deduce that for every odd positive integer $n$,
\[a^n+b^n=(a+b)\left(a^{n-1}-a^{n-2}b+a^{n-3}b^2-\cdots-ab^{n-2}+b^{n-1}\right).\]
\item[(c)] Factor $x^5+32$, and show that $2^{2021}+3^{2021}$ is divisible by $5$.
\end{enumerate}
