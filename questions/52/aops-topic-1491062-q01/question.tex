\begin{enumerate}
\item[(a)] Find the range of each of the functions $2^x$, $\log x$ and $x^2+1$ (each on its largest real domain).
\item[(b)] Let $f(x)=x+2$, and let $f^6$ denote $f$ applied six times, $f^6(x)=f(f(f(f(f(f(x))))))$. Find $f^6(x)$.
\end{enumerate}
