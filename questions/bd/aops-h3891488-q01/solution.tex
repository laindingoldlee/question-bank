Yes. Choose $A=(0,0)$, $B=(1,0)$, $C=(\dfrac{3}{2},\dfrac{\sqrt{3}}{2})$, $D=(\dfrac{1}{2},\dfrac{\sqrt{3}}{2})$. All consecutive sides have length $1$. This rhombus has angles $60^\circ,120^\circ,60^\circ,120^\circ$, which are not all equal.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(220);
defaultpen(fontsize(9pt));
pair A=(0,0),B=(1,0),C=(1.5,sqrt(3)/2),D=(.5,sqrt(3)/2);draw(A--B--C--D--cycle);label("$A$",A,(-1,-1));label("$B$",B,(1,-1));label("$C$",C,(1,1));label("$D$",D,(-1,1));label("$1$",(A+B)/2,(0,-1));label("$1$",(B+C)/2,(1,0));label("$1$",(C+D)/2,(0,1));label("$1$",(D+A)/2,(-1,0));
\end{asy}
