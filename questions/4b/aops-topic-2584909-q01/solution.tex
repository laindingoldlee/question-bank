Let the incircle, with centre $I$, touch $BC$, $CA$, $AB$ at $D$, $E$, $F$. Tangent segments from a point to a circle are equal, so let
\[x=AE=AF,\qquad y=BF=BD,\qquad z=CD=CE.\]
Then $x+y=c$, $y+z=a$, $z+x=b$. Adding, $x+y+z=s$, so $x=s-a$.

Now $IF\perp AB$ with $IF=r$, and $AI$ bisects angle $A$, so in the right triangle $AFI$ the angle at $A$ is $\dfrac A2$. Hence
\[\cot\dfrac A2=\dfrac{AF}{IF}=\dfrac{s-a}{r}.\qquad\blacksquare\]

\begin{asy}
usepackage("amsmath");
size(5cm);
defaultpen(fontsize(10pt));
pair B=(0,0),C=(7,0),A=(2,5);
real a=abs(B-C),b=abs(C-A),c=abs(A-B);pair I=(a*A+b*B+c*C)/(a+b+c);
real r=I.y;
pair ft(pair P,pair U,pair V){return U+dot(P-U,unit(V-U))*unit(V-U);}
pair Ff=ft(I,A,B),Ee=ft(I,A,C),Dd=ft(I,B,C);
draw(A--B--C--cycle,linewidth(0.8));draw(circle(I,r));
draw(A--I,dashed);draw(I--Ff);draw(I--Ee,dotted);draw(I--Dd,dotted);
pair u=unit(A-Ff),v=unit(I-Ff);draw(Ff+0.3*u--Ff+0.3*u+0.3*v--Ff+0.3*v);
dot(I);dot(Ff);dot(Ee);dot(Dd);
label("$A$",A,N);label("$B$",B,SW);label("$C$",C,SE);label("$I$",I,SE);
label("$F$",Ff,W);label("$E$",Ee,NE);label("$D$",Dd,S);
label("$r$",(I+Ff)/2,S,fontsize(9pt));label("$s-a$",(A+Ff)/2,W,fontsize(9pt));
\end{asy}
