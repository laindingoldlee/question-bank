Take $BE=1$, so $EC=2$. In triangle $ABE$, $\angle ABE=180^\circ-120^\circ-15^\circ=45^\circ$, and $\angle AEC=180^\circ-120^\circ=60^\circ$.


\begin{asy}

defaultpen(fontsize(10pt));

pair A=((3+sqrt(3))/2,(3+sqrt(3))/2);
draw(A--B--C--cycle,linewidth(0.9));draw(A--Ee,linewidth(0.9));
draw(arc(B,0.35,0,45));label("$45^\circ$",B+0.55*dir(20),fontsize(9pt));
draw(arc(Ee,0.18,60,180));label("$120^\circ$",Ee+0.4*dir(95),fontsize(9pt));

\textbf{Auxiliary point.} Let $F$ be the foot of the perpendicular from $C$ to line $AE$, and join $F$ to $B$ (dashed in the second sketch; equal ticks mark $EB=EF$ and $FA=FB=FC$).

\begin{asy}
usepackage("amsmath");
size(7cm);

pair B=(0,0),Ee=(1,0),C=(3,0);
pair A=((3+sqrt(3))/2,(3+sqrt(3))/2);
pair F=Ee+0.5*(1,sqrt(3));

\begin{itemize}
\item In the right triangle $CFE$, $\angle CEF=60^\circ$, so it is a $30$-$60$-$90$ triangle: $EF=\dfrac{1}{2}EC=1=BE$ and $\angle ECF=30^\circ$. Moreover $F$ lies strictly between $E$ and $A$: the angle $\angle AEC=60^\circ$ is acute, so $F$ is on ray $EA$, and in triangle $ABE$ the side $AE$, opposite the $45^\circ$ angle, is longer than the side $BE=1=EF$, opposite the $15^\circ$ angle.
\item Triangle $BEF$ is isosceles with $EB=EF$ and apex angle $\angle BEF=120^\circ$, so $\angle EBF=\angle EFB=30^\circ$.
\item Then $\angle FBA=\angle ABE-\angle EBF=45^\circ-30^\circ=15^\circ=\angle FAB$, so triangle $ABF$ is isosceles with $FA=FB$.
\item In triangle $BFC$, $\angle FBC=30^\circ=\angle FCB$, so $FB=FC$.
\end{itemize}
Hence $FA=FC$, and since $\angle AFC=90^\circ$, triangle $AFC$ is an isosceles right triangle. Therefore
\[
\angle CAE=\angle CAF=\boxed{45^\circ}.
\]
\textbf{Check by trigonometry.} By the law of sines in triangle $ABE$, $AE=\dfrac{\sin45^\circ}{\sin15^\circ}=1+\sqrt3$. In triangle $AEC$, the law of cosines gives $AC^2=(1+\sqrt3)^2+4-2\cdot2(1+\sqrt3)\cdot\dfrac{1}{2}=6$, and then $\sin\angle CAE=\dfrac{2\sin60^\circ}{\sqrt6}=\dfrac{\sqrt2}{2}$, consistent with $45^\circ$ (the angle is acute, as $\angle ACE=75^\circ$).
