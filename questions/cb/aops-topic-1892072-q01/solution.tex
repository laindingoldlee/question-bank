A line is tangent to a circle exactly when its distance from the centre equals the radius. The line $x-y+k=0$ is at distance $\dfrac{|k|}{\sqrt{1^2+1^2}}=\dfrac{|k|}{\sqrt2}$ from the origin, so we need $\dfrac{|k|}{\sqrt2}=5$, i.e.
\[
k=\boxed{\pm5\sqrt2}.
\]
Alternatively, substitute $y=x+k$: $x^2+(x+k)^2=25$, i.e.\ $2x^2+2kx+k^2-25=0$. Tangency means a double root, so the discriminant $4k^2-8\left(k^2-25\right)=200-4k^2$ is $0$, giving $k^2=50$ again.

\begin{asy}
usepackage("amsmath");
size(6.5cm);
defaultpen(fontsize(10pt));
real r=5, q=5*sqrt(2);
draw((-9,0)--(9,0),gray(0.5),Arrow(5));draw((0,-9)--(0,9),gray(0.5),Arrow(5));
draw(circle((0,0),r),linewidth(0.9));
draw((-9,-9+q)--(9-q,9),blue+linewidth(0.9));
draw((-9+q,-9)--(9,9-q),blue+linewidth(0.9));
pair T1=(-q/2,q/2),T2=(q/2,-q/2);
draw((0,0)--T1,red+dashed);draw((0,0)--T2,red+dashed);
void ra(pair P,pair u,pair v){real s=0.6; draw((P+s*u)--(P+s*u+s*v)--(P+s*v),linewidth(0.5));}
ra(T1,unit(-T1),unit((1,1)));ra(T2,unit(-T2),unit((-1,-1)));
dot(T1);dot(T2);
label("$5$",T1/2,NE,red);
label(rotate(45)*"$y=x+5\sqrt{2}$",(-5,-5+q)+0.9*dir(135),blue+fontsize(8pt));
label(rotate(45)*"$y=x-5\sqrt{2}$",(5,5-q)+0.9*dir(-45),blue+fontsize(8pt));
label("$x$",(9,0),N);label("$y$",(0,9),W);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
