Let $\ell$ be the line $ax+by+c=0$, where $a$ and $b$ are not both $0$, and let $P=(p,q)$ be a point.
\begin{enumerate}
\item[(a)] Show that the reflection of $P$ in $\ell$ is
\[
P'=\left(p-\dfrac{2a(ap+bq+c)}{a^2+b^2},\ q-\dfrac{2b(ap+bq+c)}{a^2+b^2}\right).
\]
\item[(b)] Find the reflection of the point $(3,4)$ in the line $2x-y+3=0$.
\end{enumerate}
