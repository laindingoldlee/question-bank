The triangle inequality is $|2-1|<\sqrt n<2+1$, so $1<n<9$. The possible integers are $2,3,\ldots,8$, giving $\boxed7$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair[] pts={(0,0),(2,0),(0.25,sqrt(15)/4)}; string[] names={"$A$","$B$","$C$"}; path boundary=pts[0]; for(int j=1;j<pts.length;++j)boundary=boundary--pts[j];draw(boundary--cycle); for(int i=0;i<pts.length;++i) label(names[i],pts[i],dir(90+360*i/pts.length));
\end{asy}
