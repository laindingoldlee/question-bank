\begin{enumerate}
\item[(a)] Prove that for all numbers $n$ and $k$, $\ n^2=k^2+(n-k)(n+k)$.
\item[(b)] Prove that $(m+1)^2=m^2+(2m+1)$, and deduce that for integers $n>k$,
\[n^2=k^2+\sum_{m=k}^{n-1}(2m+1).\]
\item[(c)] Use these identities to compute $413^2$ and $1997^2$ without a calculator.
\end{enumerate}
