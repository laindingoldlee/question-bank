Volume gives an upper bound $\dfrac{15\cdot12\cdot9}{0.2}=8100$. It is attained by aligning the 0.2-inch edges along the 15-inch direction, producing $75\cdot12\cdot9$ bars. Thus $\boxed{8100}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),B=(4,0),C=(4,2.5),D=(0,2.5),v=(1,1);draw(A--B--C--D--cycle);draw(D--(D+v)--(C+v)--(B+v)--B);draw(C--(C+v));label("15",(2,0),S);label("9",(0,1.25),W);label("12",B+v/2,SE);
\end{asy}
