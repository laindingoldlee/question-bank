In triangle $ABC$, let $D$ be the midpoint of $BC$. If $\angle ADB=45^\circ$ and $\angle ACD=30^\circ$, determine $\angle BAD$.

\begin{asy}
usepackage("amsmath");
size(6cm);
defaultpen(fontsize(10pt));
pair Dd=(0,0),B=(-3,0),C=(3,0),A=3*(-(1+sqrt(3))/2,(1+sqrt(3))/2);
draw(A--B--C--cycle,linewidth(0.9));draw(A--Dd,linewidth(0.9));
draw(arc(Dd,0.8,135,180));label("$45^\circ$",Dd+1.45*dir(157.5));
draw(arc(C,1.3,150,180));label("$30^\circ$",C+2.0*dir(165));
draw((B+Dd)/2+(0,-0.15)--(B+Dd)/2+(0,0.15));draw((C+Dd)/2+(0,-0.15)--(C+Dd)/2+(0,0.15));
label("$A$",A,NW);label("$B$",B,S);label("$C$",C,SE);label("$D$",Dd,S);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
