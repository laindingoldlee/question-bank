\textbf{Angles in triangle $ADC$.} $\angle ADC=180^\circ-45^\circ=135^\circ$, so $\angle DAC=180^\circ-135^\circ-30^\circ=15^\circ$.

\textbf{An auxiliary point.} Let $E$ be the point on the same side of $BC$ as $A$ for which triangle $BDE$ is equilateral. Then $DE=DB=DC$ and $\angle EDC=180^\circ-60^\circ=120^\circ$, so the isosceles triangle $DEC$ has base angles $\angle DCE=\angle DEC=30^\circ$. Since $\angle DCE=30^\circ=\angle DCA$, the point $E$ lies on ray $CA$. Measured from ray $DC$, the rays $DE$ and $DA$ make angles $120^\circ$ and $135^\circ$, so $E$ lies inside angle $ADC$; hence $E$ is on segment $CA$.

\begin{asy}
usepackage("amsmath");

defaultpen(fontsize(10pt));
pair Dd=(0,0),B=(-3,0),C=(3,0),A=3*(-(1+sqrt(3))/2,(1+sqrt(3))/2);
pair Ep=C+dot(B-C,A-C)/dot(A-C,A-C)*(A-C);
draw(A--B--C--cycle,linewidth(0.9));draw(A--Dd,linewidth(0.9));
draw(B--Ep,blue+dashed);draw(Dd--Ep,blue+dashed);
pair u=unit(C-Ep),v=unit(B-Ep);draw(Ep+0.3u--Ep+0.3u+0.3v--Ep+0.3v,blue);
draw((B+Dd)/2+(0,-0.15)--(B+Dd)/2+(0,0.15));draw((C+Dd)/2+(0,-0.15)--(C+Dd)/2+(0,0.15));
draw(arc(C,1.3,150,180));label("$30^\circ$",C+2.0*dir(165));

\end{asy}

\begin{itemize}
\item $\angle BEC=\angle BED+\angle DEC=60^\circ+30^\circ=90^\circ$, so $BE\perp AC$.
\item $\angle ADE=\angle ADC-\angle EDC=135^\circ-120^\circ=15^\circ=\angle DAE$, so triangle $ADE$ is isosceles with $EA=ED$.
\end{itemize}
Therefore $EA=ED=EB$, and the right triangle $AEB$ (right angle at $E$) is isosceles, so $\angle BAE=45^\circ$. Since $D$ lies on segment $BC$, ray $AD$ lies between rays $AB$ and $AC$, and
\[
\angle BAD=\angle BAE-\angle DAE=45^\circ-15^\circ=\boxed{30^\circ}.
\]
