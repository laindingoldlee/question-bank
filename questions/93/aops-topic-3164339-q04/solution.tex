Use $A=(0,0),B=(2,0),C=(2,2),D=(0,2),E=(0,-2)$. Line $EC$ is $y=2x-2$, so $X=(1,0)$. Its intersections with $x^2+y^2=4$ satisfy $5x^2-8x=0$. Besides $E$ this gives $Y=(\dfrac{8}{5},\dfrac{6}{5})$, on the required minor arc. Thus $XY=\sqrt{(\dfrac{3}{5})^2+(\dfrac{6}{5})^2}=\boxed{\dfrac{3\sqrt5}{5}}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),B=(2,0),C=(2,2),D=(0,2),Ee=(0,-2),X=(1,0),Y=(8/5,6/5);draw(circle(A,2));draw(A--B--C--D--cycle^^Ee--C);draw(X--Y,red+1bp);label("$A$",A,SW);label("$B$",B,SE);label("$C$",C,NE);label("$D$",D,N);label("$E$",Ee,S);label("$X$",X,S);label("$Y$",Y,E);
\end{asy}
