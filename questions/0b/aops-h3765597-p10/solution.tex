Put $\phi=\angle BAK$. Since $AD\parallel KC$, $\angle ADK=\angle CKD=71^\circ$. The equality $AD=BC=AK$ makes triangle $ADK$ isosceles, so $\angle AKD=71^\circ$ and $\angle DAK=38^\circ$. Because $DH\perp AK$, triangle $ADH$ gives $\angle ADH=90^\circ-38^\circ=\boxed{52^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
real w=5*cos(52*pi/180),k=5*sin(52*pi/180);pair A=(0,0),B=(w,0),C=(w,5),D=(0,5),K=(w,k),H=(dot(D,K)/dot(K,K))*K;draw(A--B--C--D--cycle);draw(A--K--D);draw(D--H,blue);label("A",A,SW);label("B",B,SE);label("C",C,NE);label("D",D,NW);label("K",K,E);label("H",H,SE);
\end{asy}
