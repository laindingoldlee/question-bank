For fixed base $BC=2$, the altitude from $E$ is at most the distance from $A$ to $BC$ plus $AE$, namely $\sqrt3+2\sqrt2$. This maximum occurs when the diagonal $AE$ points from $BC$ directly away through $A$. Therefore the maximum area is $\boxed{\sqrt3+2\sqrt2}$. 
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(220);
defaultpen(fontsize(9pt));
pair A=(0,sqrt(3)),B=(-1,0),C=(1,0),E=A+(0,2*sqrt(2)),D=(sqrt(2),sqrt(3)+sqrt(2)),F=(-sqrt(2),sqrt(3)+sqrt(2));draw(A--B--C--cycle);draw(A--D--E--F--cycle);draw((0,0)--E,dashed);label("$A$",A,(-1,0));label("$B$",B,(-1,-1));label("$C$",C,(1,-1));label("$D$",D,(1,0));label("$E$",E,(0,1));label("$F$",F,(-1,0));
\end{asy}
