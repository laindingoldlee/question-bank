With $A=(0,0)$, $B=(1,0)$, $C=(1,1)$, $D=(0,1)$ we have $M=\left(\dfrac{1}{2},0\right)$, $T=\left(1,\dfrac{2}{3}\right)$, $H=\left(\dfrac{1}{4},1\right)$.

\begin{asy}
usepackage("amsmath");
size(5cm);
defaultpen(fontsize(10pt));
pair A=(0,0),B=(1,0),C=(1,1),D=(0,1);
pair M=(0.5,0),T=(1,2/3),H=(0.25,1);pair Ex=extension(A,H,T,D);
fill(M--T--Ex--cycle,palegray);
draw(A--B--C--D--cycle,linewidth(0.9));draw(A--H);draw(T--D);draw(M--T--Ex--cycle,red+linewidth(0.9));
dot(M);dot(T);dot(H);dot(Ex);
label("$A$",A,SW);label("$B$",B,SE);label("$C$",C,NE);label("$D$",D,NW);
label("$M$",M,S);label("$T$",T,E);label("$H$",H,N);label("$E$",Ex,dir(210));
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}

\textbf{Point $E$.} Line $AH$ is $y=4x$; line $TD$ is $y=1-\dfrac {x}{3}$. Then $4x=1-\dfrac {x}{3}$ gives $x=\dfrac{3}{13}$, so $E=\left(\dfrac{3}{13},\dfrac{12}{13}\right)$.

\textbf{Area.} With $\overrightarrow{MT}=\left(\dfrac{1}{2},\dfrac{2}{3}\right)$ and $\overrightarrow{ME}=\left(-\dfrac{7}{26},\dfrac{12}{13}\right)$,
\[
[MTE]=\dfrac{1}{2}\left|\dfrac{1}{2}\cdot\dfrac{12}{13}-\dfrac{2}{3}\cdot\left(-\dfrac{7}{26}\right)\right|=\dfrac{1}{2}\left(\dfrac{6}{13}+\dfrac{7}{39}\right)=\dfrac{1}{2}\cdot\dfrac{25}{39}=\boxed{\textbf{(b)}\ \dfrac{25}{78}} .
\]
