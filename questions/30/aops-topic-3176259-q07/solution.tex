Use $A=(0,0)$, $B=(1,0)$, $C=(1,h)$, $D=(0,h)$ with $h>1$. The bisector is $y=x$, so $E=(1,1)$ and $F=(h,h)$. Then $FE=\sqrt2(h-1)$ and $FD=h$. Their equality gives $(\sqrt2-1)h=\sqrt2$, hence $\boxed{h=2+\sqrt2}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
real h=2+sqrt(2);pair A=(0,0),Bp=(1,0),C=(1,h),D=(0,h),Ep=(1,1),F=(h,h);draw(A--Bp--C--D--cycle);draw(D--F,dashed);draw(A--F,blue);label("$A$",A,SW);label("$B$",Bp,SE);label("$C$",C,N);label("$D$",D,NW);label("$E$",Ep,E);label("$F$",F,NE);
\end{asy}
