One circular segment has area $\dfrac{60}{360}\pi6^2-\dfrac{1}{2}\cdot6^2\sin60^\circ=6\pi-9\sqrt3$. One lens has two such segments, and four lenses have area $48\pi-72\sqrt3$. Thus $a+b+c=48+72+3=\boxed{123}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
real r=6,h=3*sqrt(3);pair O=(0,0),P=(2*h,0),U=(h,3),V=(h,-3);draw(circle(O,r)^^circle(P,r));draw(O--U--P--V--O,dashed);draw(U--V);label("$O_1$",O,W);label("$O_2$",P,E);label("$U$",U,N);label("$V$",V,S);label("$60^\circ$",(2,0));
\end{asy}
