The triangle angle sum gives $\angle ABC=180^\circ-55^\circ-30^\circ=95^\circ$. Because $D$ lies on $AB$, $\angle CBD=95^\circ$. In triangle $BCD$, the remaining angle is $180^\circ-95^\circ-10^\circ=\boxed{75^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,0),B=(4,0),C=8sin(95*pi/180)*dir(55),D=(C.x-C.y/tan(255*pi/180),0);draw(A--B--C--cycle);draw(C--D);label("$A:55^\circ$",A,(-1,-1));label("$B$",B,(1,-1));label("$C:30^\circ$",C,(0,1));label("$D$",D,(0,-1));label("$10^\circ$",C+.7dir(263),(1,0));
\end{asy}
