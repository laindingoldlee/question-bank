Let $A^*$ be a point of $\ell$ on the other side of $P$ from $A$ (so $A^*$ and $B$ are on the same side of $t$), and let $X$ be a point of $t$ with $P$ between $X$ and $Q$.

\begin{asy}
usepackage("amsmath");
size(6.5cm);
defaultpen(fontsize(10pt));
pair Q=(0,0), P=(2/Tan(60),2);
pair dt=dir(60);
draw((-2.6,2)--(4.6,2),linewidth(0.9)); draw((-2.6,0)--(4.6,0),linewidth(0.9));
draw(Q-1.3*dt--P+1.3*dt,linewidth(0.9));
draw((2.4,2)--(2.6,2),Arrow(5)); draw((2.4,0)--(2.6,0),Arrow(5));
label("$\ell$",(4.6,2),E); label("$m$",(4.6,0),E); label("$t$",P+1.3*dt,N);
dot(P); dot(Q);
dot(P); label("$P$",P,NW);
dot(Q); label("$Q$",Q,SE);
dot(P+0.9*dt); label("$X$",P+0.9*dt,W);
dot((-2,2)); label("$A$",(-2,2),N);
dot((3,0)); label("$B$",(3,0),S);
dot((4,2)); label("$A^{*}$",(4,2),N);
draw(arc(P,0.45,180,240),red); label("$\beta$",P+0.77*dir(210.0),red+fontsize(8pt));
draw(arc(Q,0.45,0,60),blue); label("$\beta$",Q+0.77*dir(30.0),blue+fontsize(8pt));
draw(arc(P,0.45,0,60),deepgreen); label("$\beta$",P+0.77*dir(30.0),deepgreen+fontsize(8pt));
\end{asy}

\begin{itemize}
\item $\angle XPA^*$ and $\angle PQB$ are corresponding angles ($A^*$ and $B$ on the same side of $t$), so $\angle XPA^*\cong\angle PQB$.
\item The rays $PX$, $PQ$ are opposite, and so are $PA^*$, $PA$. Hence $\angle XPA^*$ and $\angle QPA$ are vertical angles, and $\angle XPA^*\cong\angle APQ$.
\end{itemize}
Combining, $\angle APQ\cong\angle XPA^*\cong\angle PQB=\angle BQP$. $\blacksquare$
