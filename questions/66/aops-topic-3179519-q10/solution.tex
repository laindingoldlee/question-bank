Place ray $AB$ at direction $0^\circ$ and $AC$ at $60^\circ$. Since $D$ is interior and $\angle CAD=15^\circ$, ray $AD$ has direction $45^\circ$. Of the two equilateral choices, only the one obtained by rotating $AD$ clockwise through $60^\circ$ lets $DE$ cross $AB$. The direction from $D$ to $E$ is $45^\circ-120^\circ=-75^\circ$. Therefore $FD$ has direction $105^\circ$ while $FB$ has direction $0^\circ$, giving $\boxed{105^\circ}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),B=(4,0),C=4*dir(60),D=1.6*dir(45),Ee=rotate(-60)*D,F=extension(D,Ee,A,B);draw(A--B--C--cycle);draw(A--D--Ee--cycle);dot(F);label("$A$",A,SW);label("$B$",B,SE);label("$C$",C,N);label("$D$",D,N);label("$E$",Ee,S);label("$F$",F,SE);
\end{asy}
