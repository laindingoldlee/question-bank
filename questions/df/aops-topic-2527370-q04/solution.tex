The cake has height $6$ and base $8\times10$. A cube has no icing exactly when it does not touch the top or any of the four side faces. Removing a layer of thickness $1$ from the top and from each side face, but nothing from the bottom, leaves a block of
\[(10-2)\times(8-2)\times(6-1)=8\times6\times5=\boxed{240}\]
cubes without icing.

In the sketch, the iced faces are shaded and the un-iced block is dashed in red.

\begin{asy}
usepackage("amsmath");
size(6cm);
defaultpen(fontsize(10pt));
pair P(real x,real y,real z){return (x+0.5*y,z+0.38*y);}
real a=10,c=8,h=6;
fill(P(0,0,h)--P(a,0,h)--P(a,c,h)--P(0,c,h)--cycle,paleblue);
fill(P(0,0,0)--P(a,0,0)--P(a,0,h)--P(0,0,h)--cycle,paleblue);
fill(P(a,0,0)--P(a,c,0)--P(a,c,h)--P(a,0,h)--cycle,paleblue);
draw(P(0,0,0)--P(a,0,0)--P(a,0,h)--P(0,0,h)--cycle,linewidth(0.8));
draw(P(a,0,0)--P(a,c,0)--P(a,c,h)--P(a,0,h),linewidth(0.8));
draw(P(0,0,h)--P(0,c,h)--P(a,c,h),linewidth(0.8));
draw(P(0,0,0)--P(0,c,0)--P(a,c,0),dashed);draw(P(0,c,0)--P(0,c,h),dashed);
draw(P(1,1,0)--P(9,1,0)--P(9,1,5)--P(1,1,5)--cycle,red+dashed);
draw(P(9,1,5)--P(9,7,5)--P(1,7,5)--P(1,1,5),red+dashed);draw(P(9,1,0)--P(9,7,0)--P(9,7,5),red+dashed);
label("$10$",P(a/2,0,0),S);label("$8$",P(a,c/2,0),SE);label("$6$",P(0,0,h/2),W);
label("un-iced block: $8\times6\times5$",P(a/2,0,0)+(0,-1.6),S,red+fontsize(8pt));
\end{asy}
