Let the trapezoid height be $ h $, the semicircle radius $ r $, and $ KA=x $. The perpendicular distances from $ A $ to the two legs are $ x\sin K=\dfrac{xh}{65}$ and $(80-x)\sin L=\dfrac{(80-x)h}{60}$. Both equal $ r $. Cancelling $ h>0 $ gives $ 60x=65(80-x)$, hence $\boxed{KA=\dfrac{208}{5}}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
real u=3989/116,h=sqrt(65^2-u^2),x=208/5,r=x*h/65;pair K=(0,0),L=(80,0),Nn=(u,h),M=(u+22,h),A=(x,0);draw(K--L--M--Nn--cycle);draw(arc(A,r,0,180));dot(A);label("$K$",K,SW);label("$L$",L,SE);label("$M$",M,NE);label("$N$",Nn,NW);label("$A$",A,S);label("$65$",Nn/2,W);label("$60$",(L+M)/2,E);label("$22$",(Nn+M)/2,N);
\end{asy}
