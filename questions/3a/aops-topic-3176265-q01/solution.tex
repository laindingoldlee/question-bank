The perpendicular legs are $3,4$, so the area is $\dfrac{1}{2}\cdot3\cdot4=\boxed6$.
\begin{asy}
usepackage("amsmath");

pair[] pts={(0,3),(0,0),(4,0)};string[] names={"$A$","$B$","$C$"};path boundary=pts[0];for(int j=1;j<pts.length;++j)boundary=boundary--pts[j];draw(boundary--cycle);for(int j=0;j<pts.length;++j)label(names[j],pts[j],dir(90+360*j/pts.length));
\end{asy}
