$\dfrac{\sqrt3}{4}a^2=\sqrt3\,b^2$, so $a^2=4b^2$ and $\dfrac ab=\boxed{2}$.

draw(shift(2.6,0)*scale(1)*unitsquare,linewidth(0.8));label("$b$",(3.1,0),S);
\end{asy}
