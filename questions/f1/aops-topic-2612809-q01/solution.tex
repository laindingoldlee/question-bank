$OA=OB$ (radii) and $AB=OA$, so triangle $OAB$ is equilateral and $\angle AOB=60^\circ$. The measure of a minor arc equals its central angle, so the minor arc $AB$ measures $\boxed{60^\circ}$.

\begin{asy}
usepackage("amsmath");
size(3.5cm);
defaultpen(fontsize(10pt));
pair O=(0,0),A=dir(120),B=dir(60);
draw(circle(O,1)); draw(arc(O,1,60,120),linewidth(2));
draw(O--A--B--cycle,dashed); draw(A--B,linewidth(0.9));
draw(arc(O,0.2,60,120)); label("$60^\circ$",0.42*dir(90),fontsize(7pt));
dot(O);dot(A);dot(B);
label("$O$",O,S);label("$A$",A,NW);label("$B$",B,NE);
label("$r$",(O+A)/2,W,fontsize(9pt));label("$r$",(O+B)/2,E,fontsize(9pt));label("$r$",(A+B)/2,S,fontsize(9pt));
\end{asy}
