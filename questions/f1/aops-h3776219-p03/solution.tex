Use two tiles along each row, making width $28$, and seven rows of height $4$, making height $28$. This uses $2\cdot7=\boxed{14}$ tiles and produces area $28^2=\boxed{784}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
draw(box((0,0),(28,28)));draw((14,0)--(14,28));for(int k=1;k<7;++k)draw((0,4*k)--(28,4*k));label("28",(14,0),S);label("28",(28,14),E);label("14",(7,28),N);label("4",(0,26),W);
\end{asy}
