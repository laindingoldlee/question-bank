Ptolemy\textquotesingle s theorem for cyclic quadrilateral $ABPC$ gives $AP\cdot BC=AB\cdot PC+AC\cdot PB$. Since $AB=AC$ and $AC:BC=3:4$, $AP=\dfrac{3}{4}(PB+PC)=\boxed{12}$.

import graph;
usepackage("amsmath");
import geometry;

pair A=(0,sqrt(5)),B=(-2,0),C=(2,0),O=(0,1/(2sqrt(5))),P=O+(9/(2sqrt(5)))*dir(-70);draw(circle(O,9/(2sqrt(5))));draw(A--B--P--C--cycle);draw(A--P);label("$A$",A,(0,1));label("$B$",B,(-1,0));label("$C$",C,(1,0));label("$P$",P,(0,-1));
\end{asy}
