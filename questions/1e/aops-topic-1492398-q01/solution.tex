Let $x=\cos A$, $y=\cos B$, $z=\cos C$ and $s=x+y+z$.

\begin{asy}
usepackage("amsmath");
size(4.5cm);
defaultpen(fontsize(10pt));
pair A=(0,0),B=(5,0),C=(1.6,3);
draw(A--B--C--cycle,linewidth(0.9));
draw(arc(A,0.6,0,degrees(C-A)),red); draw(arc(B,0.6,degrees(C-B),180),red); draw(arc(C,0.55,degrees(A-C),degrees(B-C)+360),red);
label("$A$",A,SW); label("$B$",B,SE); label("$C$",C,N);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}

\textbf{Step 1: $1<s\le\dfrac{3}{2}$.} Using $C=180^\circ-(A+B)$, sum-to-product and the double-angle formula,
\[
\cos A+\cos B=2\cos\dfrac{A+B}{2}\cos\dfrac{A-B}{2}=2\sin\dfrac{C}{2}\cos\dfrac{A-B}{2},\qquad \cos C=1-2\sin^2\dfrac{C}{2}.
\]
Put $t=\sin\dfrac{C}{2}$, so $0<t<1$. Since $\cos\dfrac{A-B}{2}\le1$,
\[
s\le1+2t-2t^2=\dfrac{3}{2}-2\left(t-\dfrac{1}{2}\right)^2\le\dfrac{3}{2} .
\]
Also, writing $\sin\dfrac{C}{2}=\cos\dfrac{A+B}{2}$,
\[
s=1+2\sin\dfrac{C}{2}\left(\cos\dfrac{A-B}{2}-\cos\dfrac{A+B}{2}\right)=1+4\sin\dfrac{A}{2}\sin\dfrac{B}{2}\sin\dfrac{C}{2}>1 .
\]

\textbf{Step 2.} For all real numbers, $(x-y)^2+(y-z)^2+(z-x)^2\ge0$ gives $x^2+y^2+z^2\ge xy+yz+zx$, hence
\[
s^2=x^2+y^2+z^2+2(xy+yz+zx)\ge3(xy+yz+zx).
\]
So, using $s>0$ and $s\le\dfrac{3}{2}$,
\[
2(xy+yz+zx)\le\dfrac{2s^2}{3}=s\cdot\dfrac{2s}{3}\le s\cdot\dfrac{2}{3}\cdot\dfrac{3}{2}=s .
\]
This is the required inequality. $\blacksquare$

Equality needs $x=y=z$ and $s=\dfrac{3}{2}$, that is, an equilateral triangle; there both sides equal $\dfrac{3}{2}$.
