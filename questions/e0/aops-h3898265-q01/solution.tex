Let $AC=BC=s$ and $CD=t$. Then $\dfrac{s}{t}=\tan75^\circ=2+\sqrt3$, so $s^2+t^2=4st$. The area gives $st=162$. Therefore $BD^2=(s-t)^2=s^2+t^2-2st=2st=324$. Since $BD>0$, $\boxed{BD=18}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");

real s=3,t=3*(2-sqrt(3));pair C=(0,0),A=(0,s),B=(s,0),D=(t,0);draw(A--B--C--cycle);draw(A--D,blue);label("A",A,N);label("B",B,SE);label("C",C,SW);label("D",D,S);label("$s$",(0,1.5),W);label("$t$",(t/2,0),S);label("$75^\circ$",D+(-.3,.35));
\end{asy}
