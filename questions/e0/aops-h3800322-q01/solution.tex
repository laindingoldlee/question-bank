Let $R$ and $r$ be the circumradius and inradius, and join the centre to a vertex and to the midpoint of an adjacent side. This forms a right triangle with hypotenuse $R$, legs $r$ and $5$. Therefore $R^2-r^2=25$. The requested annular area is $\pi R^2-\pi r^2=\boxed{25\pi}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
real R=5/sin(pi/8),r=R*cos(pi/8);fill(circle((0,0),R),lightblue);fill(circle((0,0),r),white);draw(circle((0,0),R));draw(circle((0,0),r));pair[] v;for(int i=0;i<8;++i)v.push(R*dir(45i));for(int i=0;i<8;++i)draw(v[i]--v[(i+1)%8]);pair M=(v[0]+v[1])/2;draw((0,0)--v[0]);draw((0,0)--M);label("$O$",(0,0),(-1,0));label("$R$",v[0]/2,(0,-1));label("$r$",M/2,(-1,1));label("$M$",M,(1,1));label("$5$",(M+v[0])/2,(1,0));
\end{asy}
