Let $\alpha=\angle PAB$ and $\beta=\angle PBA$. Triangle $APB$ gives $\alpha+\beta=90^\circ$. Thus $A+B=(\alpha+20^\circ)+(\beta+30^\circ)=140^\circ$, leaving $\boxed{C=40^\circ}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),B=(4,0),P=(3,sqrt(3)),C=(4,4*tan(50*pi/180));draw(A--B--C--cycle);draw(A--P--B);label("$A$",A,SW);label("$B$",B,SE);label("$C$",C,N);label("$P$",P,W);label("$20^\circ$",A+0.9*dir(40));label("$30^\circ$",B+0.9*dir(105));
\end{asy}
