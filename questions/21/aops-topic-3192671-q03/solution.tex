Similar figures with length scale factor $2$ have area scale factor $2^2=\boxed4$. The areas are $6$ and $24$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair[] pts={(0,0),(4,0),(0,3)}; string[] names={"$A$","$B$","$C$"}; path boundary=pts[0]; for(int j=1;j<pts.length;++j)boundary=boundary--pts[j];draw(boundary--cycle); for(int i=0;i<pts.length;++i) label(names[i],pts[i],dir(90+360*i/pts.length));
\end{asy}
