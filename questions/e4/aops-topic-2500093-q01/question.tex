\begin{enumerate}
\item[(a)] Prove that for every point $P$ in the plane of a rectangle $ABCD$,
\[PA^2+PC^2=PB^2+PD^2.\]
\item[(b)] In square $ABCD$, a point $P$ inside the square has $PA=3$, $PB=5$ and $PC=7$. Find $PD$, and show that such a point really exists.
\end{enumerate}
