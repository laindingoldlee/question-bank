With radius $r$ and height $h=2r$, the volume is $\pi r^2\cdot2r=2\pi r^3=6.28\,r^3=401.92$, so $r^3=64$ and $r=4$. Hence the radius is $\boxed{4\text{ cm}}$ and the height is $\boxed{8\text{ cm}}$.

\begin{asy}
usepackage("amsmath");
size(2.5cm);
defaultpen(fontsize(10pt));
real r=1,h=2,e=0.3;
draw(xscale(r)*yscale(e)*arc((0,0),1,180,360),linewidth(0.8));draw(xscale(r)*yscale(e)*arc((0,0),1,0,180),dashed);
draw(shift(0,h)*xscale(r)*yscale(e)*unitcircle,linewidth(0.8));
draw((-r,0)--(-r,h),linewidth(0.8));draw((r,0)--(r,h),linewidth(0.8));
draw((0,h)--(r,h));dot((0,h));label("$r$",(r/2,h),S,fontsize(8pt));
draw((1.3,0)--(1.3,h),Arrows(4));label("$h=2r$",(1.3,h/2),E,fontsize(8pt));
\end{asy}
