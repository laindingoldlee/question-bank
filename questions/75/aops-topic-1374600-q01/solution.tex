A line with slope $2$ through $(-4,0)$ has point-slope form $y-0=2\left(x-(-4)\right)$, i.e.
\[
\boxed{y=2x+8}.
\]
Check: at $x=-4$, $y=0$, and the $y$-intercept is $(0,8)$, so the slope is $\dfrac{8-0}{0-(-4)}=2$.

\begin{asy}
usepackage("amsmath");
size(4.5cm);
defaultpen(fontsize(10pt));
import graph;
draw((-6,0)--(3,0),gray,Arrow(5)); draw((0,-2)--(0,11),gray,Arrow(5));
label("$x$",(3,0),E,gray); label("$y$",(0,11),N,gray);
draw((-5,-2)--(1.5,11),blue+linewidth(1));
dot((-4,0)); dot((0,8));
label("$(-4,0)$",(-4,0),SE); label("$(0,8)$",(0,8),E);
label("$y=2x+8$",(-1.4,6.4),W,blue);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
