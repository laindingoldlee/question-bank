Choose $A=(0,0)$, $B=(16,0)$, $C=(16,16)$, $D=(0,16)$. Then $E=(\dfrac{32}{3},16)$ and $F=(\dfrac{16}{3},16)$. The lines have equations
\[
AE:\ y=\dfrac{3}{2}x,\qquad BF:\ y=-\dfrac{3}{2}(x-16).
\]
They intersect at $M=(8,12)$. Therefore $[MAB]=\dfrac{1}{2}\cdot16\cdot12=\boxed{96}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,0),B=(16,0),C=(16,16),D=(0,16),E=(32.0/3,16),F=(16.0/3,16),M=(8,12);fill(A--B--M--cycle,lightblue);draw(A--B--C--D--cycle);draw(A--E);draw(B--F);label("$A$",A,SW);label("$B$",B,SE);label("$C$",C,NE);label("$D$",D,NW);label("$E$",E,(0,1));label("$F$",F,(0,1));label("$M$",M,(0,-1));
\end{asy}
