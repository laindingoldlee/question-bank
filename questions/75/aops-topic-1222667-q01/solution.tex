Arc length $=r\theta$ with $\theta$ in radians, so $\theta=\dfrac{6\pi}{8}=\boxed{\dfrac{3\pi}{4}}$ radians $=135^\circ$.

\begin{asy}
usepackage("amsmath");
size(2.5cm);
defaultpen(fontsize(10pt));
draw(unitcircle); draw((0,0)--(1,0)); draw((0,0)--dir(135)); draw(arc((0,0),1,0,135),red+linewidth(1.5)); label("$8$",(0.5,0),S); label("$\theta$",(0,0),3*dir(67));
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
