Use vectors from $A$: let $\vec u=\overrightarrow{AB}$, $\vec v=\overrightarrow{AD}$, $\vec m=\overrightarrow{AM}$, $\vec n=\overrightarrow{AN}$. Then
\[
\vec m=\vec u+\dfrac{1}{2}\vec v,\qquad \vec n=\vec v+\dfrac{1}{2}\vec u .
\]
Doubling the first equation and subtracting the second gives $\dfrac{3}{2}\vec u=2\vec m-\vec n$, so $\vec u=\dfrac{4\vec m-2\vec n}{3}$. We know $|\vec m|=2$, $|\vec n|=1$ and $\vec m\cdot\vec n=2\cdot1\cdot\cos60^\circ=1$. Hence
\[
AB^2=|\vec u|^2=\dfrac{16|\vec m|^2-16\,\vec m\cdot\vec n+4|\vec n|^2}{9}=\dfrac{64-16+4}{9}=\dfrac{52}{9},
\]
and $AB=\boxed{\dfrac{2\sqrt{13}}{3}}$.

\begin{asy}
usepackage("amsmath");
size(6cm);
defaultpen(fontsize(10pt));
pair m0=(2,0),n0=dir(60);
pair u0=(4*m0-2*n0)/3,v0=(4*n0-2*m0)/3;
real t=-degrees(u0);
pair u=rotate(t)*u0,v=rotate(t)*v0;
pair A=(0,0),B=u,C=u+v,D=v,M=(B+C)/2,N=(C+D)/2;
draw(A--B--C--D--cycle,linewidth(0.9));
draw(A--M,blue); draw(A--N,blue);
draw(arc(A,0.35,degrees(M),degrees(N)),red);
dot(A);dot(B);dot(C);dot(D);dot(M);dot(N);
label("$A$",A,SW); label("$B$",B,SE); label("$C$",C,NE); label("$D$",D,NW); label("$M$",M,E); label("$N$",N,N);
label("$2$",(A+M)/2,SE,blue); label("$1$",(A+N)/2+(-0.1,0),W,blue);
label("$60^\circ$",A+0.72*dir((degrees(M)+degrees(N))/2),red+fontsize(8pt));
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
