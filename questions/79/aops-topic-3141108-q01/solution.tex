Let $DC=x$, so $AD=2x$ and $AC=3x$. Stewart's theorem gives $9^2(2x)+12^2x=3x(6^2+2x^2)$. Dividing yields $306=108+6x^2$, so $x^2=33$. Thus $AC=3\sqrt{33}\approx\boxed{17.2}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
real L=3*sqrt(33),bx=180/L;pair A=(0,0),C=(L,0),D=(2*L/3,0),B=(bx,sqrt(144-bx^2));draw(A--B--C--cycle);draw(B--D,blue);label("$A$",A,SW);label("$B$",B,N);label("$C$",C,SE);label("$D$",D,S);label("$12$",(A+B)/2,NW);label("$9$",(B+C)/2,NE);label("$6$",(B+D)/2,E);
\end{asy}
