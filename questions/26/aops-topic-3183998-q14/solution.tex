Let each removed right triangle have legs $x$. Its hypotenuse is $x\sqrt2$, which equals the remaining straight side $2000-2x$. Thus the octagon side is $\dfrac{2000\sqrt2}{2+\sqrt2}=\boxed{2000(\sqrt2-1)}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
real t=2/(2+sqrt(2));draw((0,0)--(2,0)--(2,2)--(0,2)--cycle,dashed);draw((t,0)--(2-t,0)--(2,t)--(2,2-t)--(2-t,2)--(t,2)--(0,2-t)--(0,t)--cycle);label("$2000$",(1,-.2));label("$x$",(t/2,0),S);
\end{asy}
