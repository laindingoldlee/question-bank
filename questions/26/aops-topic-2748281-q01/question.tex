\begin{enumerate}
\item[(a)] Prove that $3$ is the only prime that is $1$ less than a perfect square.
\item[(b)] More generally, let $k\ge1$ be an integer. Prove that if a prime $p$ equals $m^2-k^2$ for some positive integer $m$, then $m=k+1$ and $p=2k+1$.
\end{enumerate}
