Let the base be the equilateral triangle $ABC$ and the apex $D$. By symmetry, the foot of the perpendicular from $D$ to the base is the centre $G$ of triangle $ABC$ (the point equidistant from $A$, $B$, $C$, since $DA=DB=DC$).

\begin{asy}
usepackage("amsmath");
size(6cm);
defaultpen(fontsize(10pt));
pair A=(0,0),B=(6,0),C=(3.9,1.7);
pair G=(A+B+C)/3;pair D=G+(0,4.9);
draw(A--B,linewidth(0.9));draw(A--D--B,linewidth(0.9));draw(D--C,dashed);draw(A--C--B,dashed);
draw(D--G,red+dashed);draw(G--A,blue+dashed);
dot(G);
real s=0.25;pair u=unit(D-G),v=unit(A-G);draw(G+s*u--G+s*(u+v)--G+s*v,red);
label("$A$",A,SW);label("$B$",B,SE);label("$C$",C,NE);label("$D$",D,N);label("$G$",G,dir(-30));
label("$6$",(A+B)/2,S);label("$6$",(A+D)/2,NW);
label("$h$",(D+G)/2,E,red);draw((A+G)/2--(1.4,-0.55),blue+linewidth(0.4));label("$2\sqrt{3}$",(1.4,-0.55),S,blue);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}

\textbf{Base.} The area of an equilateral triangle with side $6$ is $\dfrac{\sqrt3}{4}\cdot36=9\sqrt3$.

\textbf{Distance $AG$.} The centre lies on each median, two thirds of the way from the vertex. A median of $ABC$ has length $\dfrac{\sqrt3}{2}\cdot6=3\sqrt3$, so $AG=\dfrac{2}{3}\cdot3\sqrt3=2\sqrt3$.

\textbf{Height.} Triangle $AGD$ has a right angle at $G$, so
\[
h=DG=\sqrt{AD^2-AG^2}=\sqrt{36-12}=2\sqrt6 .
\]

\textbf{Volume.}
\[
V=\dfrac{1}{3}\cdot9\sqrt3\cdot2\sqrt6=6\sqrt{18}=\boxed{18\sqrt2}\approx25.46 .
\]
(The same steps for edge $a$ give $V=\dfrac{\sqrt2}{12}a^3$; here $\dfrac{\sqrt2}{12}\cdot216=18\sqrt2$.)
