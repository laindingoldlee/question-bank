Use position vectors with $B=0$. The extension ratios give
\[
D=2C,\qquad E=A+2(A-C)=3A-2C,\qquad F=-3A.
\]
Their sum is zero, so the centroid $K$ of $DEF$ is the origin. Therefore
\[
GK=\sqrt{32^2+24^2}=\boxed{40}.
\]
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair B=(0,0),A=(2,2),C=(4,0),D=2C,E=3A-2C,F=-3A,G=(A+C)/3;draw(A--B--C--cycle,blue);draw(D--E--F--cycle,gray);draw(C--D,dashed);draw(A--E,dashed);draw(B--F,dashed);dot(G);label("$A$",A,(0,1));label("$B=K$",B,(0,-1));label("$C$",C,(0,-1));label("$D$",D,(1,0));label("$E$",E,(0,1));label("$F$",F,SW);label("$G$",G,(1,0));
\end{asy}
