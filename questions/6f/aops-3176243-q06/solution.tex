Scale the three equal lengths to one, and put $AB=BC=s>1$. If $\theta=\angle B$, the cosine rule in $BCD$ gives $\cos\theta=\dfrac{s}{2}$. Applying it in $ABC$ gives $1=2s^2(1-\cos\theta)=2s^2-s^3$. Thus $(s-1)(s^2-s-1)=0$, and interiority excludes $s=1$. Therefore $s=\dfrac{(1+\sqrt5)}{2}$ and $\cos\theta=\dfrac{(1+\sqrt5)}{4}=\cos36^\circ$. Hence $\boxed{36^\circ}$.

import graph;import geometry;usepackage("amsmath");size(7cm);
pair B=(0,0),A=(0,(1+sqrt(5))/2),C=((1+sqrt(5))/2*sin(pi/5),(1+sqrt(5))/2*cos(pi/5)),D=(0,1);draw(A--B--C--cycle);draw(C--D,blue);label("$A$",A,(-1,1));label("$B$",B,(-1,-1));label("$C$",C,(1,0));label("$D$",D,(-1,0));
\end{asy}
