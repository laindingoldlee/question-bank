For angles with nonzero cosines,
\[
\tan\alpha-\tan\beta=\dfrac{\sin\alpha\cos\beta-\cos\alpha\sin\beta}{\cos\alpha\cos\beta}=\dfrac{\sin(\alpha-\beta)}{\cos\alpha\cos\beta}.
\]
With $\alpha=60^\circ$, $\beta=10^\circ$ and $\cos60^\circ=\dfrac{1}{2}$,
\[
\tan60^\circ-\tan10^\circ=\dfrac{\sin50^\circ}{\dfrac{1}{2}\cos10^\circ}=\dfrac{2\sin50^\circ}{\cos10^\circ}.
\]
Now $\cos10^\circ=\sin80^\circ=\sin100^\circ=2\sin50^\circ\cos50^\circ$, so
\[
\tan60^\circ-\tan10^\circ=\dfrac{2\sin50^\circ}{2\sin50^\circ\cos50^\circ}=\dfrac{1}{\cos50^\circ}.
\]
$\blacksquare$

\emph{Geometric picture.} Put $O$ at the origin, $F=(1,0)$, and let $P$, $Q$ be the points of the line $x=1$ with $\angle FOP=10^\circ$ and $\angle FOQ=60^\circ$. Then $FP=\tan10^\circ$, $FQ=\tan60^\circ$, so $PQ=\tan60^\circ-\tan10^\circ$. In triangle $OPQ$ we have $\angle POQ=50^\circ$, $\angle OQP=30^\circ$ and $OP=\dfrac{1}{\cos10^\circ}$, and the law of sines gives $PQ=\dfrac{OP\sin50^\circ}{\sin30^\circ}=\dfrac{2\sin50^\circ}{\cos10^\circ}$, the same expression as above.

\begin{asy}
usepackage("amsmath");
size(4.5cm);
defaultpen(fontsize(10pt));
pair O=(0,0),F=(1,0),P=(1,tan(10*pi/180)),Q=(1,tan(pi/3));
draw(O--F--Q--cycle,linewidth(0.9)); draw(O--P,blue);
draw(F+(-0.07,0)--F+(-0.07,0.07)--F+(0,0.07));
draw(arc(O,0.3,10,60),red);
draw(arc(Q,0.32,240,270),red);
dot(O);dot(F);dot(P);dot(Q);
label("$O$",O,SW); label("$F$",F,SE); label("$P$",P,E); label("$Q$",Q,NE);
label("$1$",(O+F)/2,S);
label("$50^\circ$",O+0.43*dir(35),red+fontsize(8pt)); label("$30^\circ$",Q+0.62*dir(256),red+fontsize(8pt));
label("$\angle FOP=10^\circ$",(0.5,-0.3),fontsize(8pt));
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
