An interior angle and its adjacent exterior angle form a straight angle, so the exterior angle is $180^\circ-120^\circ=\boxed{60^\circ}$.

\begin{asy}
usepackage("amsmath");
size(3cm);
defaultpen(fontsize(10pt));
draw((0,0)--(2,0)); draw((0,0)--1.6*dir(120)); draw((0,0)--(-1.5,0),dashed); draw(arc((0,0),0.4,0,120)); draw(arc((0,0),0.6,120,180),red); label("$120^\circ$",(0,0),(1.5,2)); label("$60^\circ$",0.6*dir(150),NW);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
