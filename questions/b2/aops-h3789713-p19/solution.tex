Scale to $A=(0,0)$, $C=(1,0)$, $M=(3,0)$. The angle at $C$ is $120^\circ$. The rays $AB$ and $CB$ therefore make angles $45^\circ$ and $60^\circ$ with the positive horizontal axis, so $B=(t,t)$ with $t=\dfrac{3+\sqrt3}{2}$. The tangent of $\angle AMB$ is $\dfrac{t}{3-t}=2+\sqrt3=\tan75^\circ$. The angle is acute, hence $\boxed{75^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,0),C=(1,0),M=(3,0),B=((3+sqrt(3))/2,(3+sqrt(3))/2);draw(A--B--C--A);draw(C--M--B);label("A",A,SW);label("C",C,S);label("M",M,SE);label("B",B,N);label("1",(.5,0),S);label("2",(2,0),S);
\end{asy}
