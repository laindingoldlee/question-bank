The line is $y=\dfrac79(x-3)$. At $x=0$, $y=-\dfrac73$, so the $y$-intercept is $\boxed{\left(0,-\dfrac73\right)}$.
\begin{asy}
usepackage("amsmath");
size(4cm);
defaultpen(fontsize(10pt));
import graph;draw((-1.5,0)--(5,0),Arrow(4));draw((0,-2.8)--(0,2),Arrow(4));
draw((-1,-28/9)--(5,14/9),linewidth(0.9));
dot((3,0));dot((0,-7/3));label("$(3,0)$",(3,0),SE,fontsize(8pt));label("$\left(0,-\dfrac73\right)$",(0,-7/3),W,fontsize(8pt));label("slope $\dfrac79$",(4.3,1.1),NW,fontsize(8pt));
\end{asy}
