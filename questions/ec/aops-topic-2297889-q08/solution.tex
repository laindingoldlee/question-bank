Let $A^*$ be a point of $\ell$ on the other side of $P$ from $A$, so that $A^*$ and $B$ are on opposite sides of $t$.

\begin{asy}
usepackage("amsmath");
size(6.5cm);
defaultpen(fontsize(10pt));
pair Q=(0,0), P=(2/Tan(60),2);
pair dt=dir(60);
draw((-2.6,2)--(4.6,2),linewidth(0.9)); draw((-2.6,0)--(4.6,0),linewidth(0.9));
draw(Q-1.3*dt--P+1.3*dt,linewidth(0.9));
draw((2.4,2)--(2.6,2),Arrow(5)); draw((2.4,0)--(2.6,0),Arrow(5));
label("$\ell$",(4.6,2),E); label("$m$",(4.6,0),E); label("$t$",P+1.3*dt,N);
dot(P); dot(Q);
dot(P); label("$P$",P,NW);
dot(Q); label("$Q$",Q,SE);
dot((4,2)); label("$A$",(4,2),N);
dot((3,0)); label("$B$",(3,0),S);
dot((-2,2)); label("$A^{*}$",(-2,2),N);
draw(arc(P,0.45,240,360),red); label("$\delta$",P+0.77*dir(300.0),red+fontsize(8pt));
draw(arc(Q,0.45,0,60),blue); label("$\varepsilon$",Q+0.77*dir(30.0),blue+fontsize(8pt));
draw(arc(P,0.45,180,240),deepgreen); label("$\varepsilon$",P+0.77*dir(210.0),deepgreen+fontsize(8pt));
\end{asy}

\begin{itemize}
\item $\angle A^*PQ$ and $\angle BQP$ are alternate interior angles, so $m\angle A^*PQ=m\angle BQP$ (marked $\varepsilon$).
\item $\angle APQ$ and $\angle A^*PQ$ form a linear pair (the rays $PA$ and $PA^*$ are opposite), so $m\angle APQ+m\angle A^*PQ=180^\circ$.
\end{itemize}
Substituting, $m\angle APQ+m\angle BQP=180^\circ$, so the two angles are supplementary. $\blacksquare$
