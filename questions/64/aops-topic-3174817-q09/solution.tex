Menelaus's theorem for transversal $Q,R,P$ gives $\dfrac{AQ}{QB}\cdot\dfrac{BP}{PC}\cdot\dfrac{CR}{RA}=1$. Hence $1\cdot\dfrac{BC+4}{4}\cdot\dfrac{3}{5}=1$, so $\boxed{BC=\dfrac{8}{3}}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair Bp=(0,0),C=(8/3,0),P=(20/3,0),A=(0,sqrt(512)/3),Q=A/2,R=3*A/8+5*C/8;draw(A--Bp--C--cycle);draw(C--P);draw(Q--P,blue);pair[]v={A,Bp,C,P,Q,R};string[]names={"A","B","C","P","Q","R"};for(int j=0;j<v.length;++j)label("$"+names[j]+"$",v[j],dir(90+50*j));
\end{asy}
