From $\angle D+\angle E=180^\circ$ and $\angle D+\angle F=180^\circ$ we get $\angle E=\angle F$. Since $\angle E+\angle F=90^\circ$, each of them is $45^\circ$. Then
\[m\angle D=180^\circ-45^\circ=\boxed{135^\circ}.\]

\begin{asy}
usepackage("amsmath");
size(9cm);
defaultpen(fontsize(10pt));
pair O=(0,0);
draw((-2.6,0)--(2.6,0),linewidth(0.9)); draw(O--2.4*dir(45),linewidth(0.9));
draw(arc(O,0.6,0,45),blue); draw(arc(O,0.75,45,180),red);
label("$E=45^\circ$",0.75*dir(22),E,blue+fontsize(8pt)); label("$D=135^\circ$",0.85*dir(120),NW,red+fontsize(8pt));
label("$D$ and $E$: supplementary",(0,-0.4),S,fontsize(8pt));
pair Q=(5.2,0);
draw(Q--Q+2.4*dir(0),linewidth(0.9)); draw(Q--Q+2.4*dir(45),linewidth(0.9)); draw(Q--Q+2.4*dir(90),linewidth(0.9));
draw(arc(Q,1.0,0,45),blue); draw(arc(Q,1.2,45,90),heavygreen+linewidth(0.8));
label("$E$",Q+1.3*dir(22),blue); label("$F$",Q+1.5*dir(67),heavygreen);
draw(Q+(0.25,0)--Q+(0.25,0.25)--Q+(0,0.25),gray(0.5));
label("$E$ and $F$: complementary, $E=F=45^\circ$",Q+(1.2,-0.4),S,fontsize(8pt));
\end{asy}
