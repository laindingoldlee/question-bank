The two altitude lines are perpendicular to $BC$ and $AC$. Rotating both lines through $90^\circ$ preserves their smaller angle, which is therefore $\angle C=58^\circ$. The two angle sizes at their intersection are $\boxed{58^\circ\text{ and }122^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(-3,0),B=(3,0),C=(0,3/tan(29*pi/180));pair K=B+dot(A-B,C-B)/dot(C-B,C-B)*(C-B),M=A+dot(B-A,C-A)/dot(C-A,C-A)*(C-A);draw(A--B--C--cycle);draw(A--K,blue);draw(B--M,red);label("A",A,SW);label("B",B,SE);label("C",C,N);label("K",K,NE);label("M",M,NW);
\end{asy}
