The boundary has four straight segments of length 5 and four quarter-circles of radius 1. Thus its perimeter is $\boxed{20+2\pi}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
draw((0,0)--(5,0)--(5,5)--(0,5)--cycle,dashed);draw((0,-1)--(5,-1));draw((6,0)--(6,5));draw((5,6)--(0,6));draw((-1,5)--(-1,0));draw(arc((0,0),1,180,270));draw(arc((5,0),1,270,360));draw(arc((5,5),1,0,90));draw(arc((0,5),1,90,180));label("$5$",(2.5,0),N);label("offset $1$",(6,2.5),E);
\end{asy}
