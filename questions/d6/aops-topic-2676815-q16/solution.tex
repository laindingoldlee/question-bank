Let $\angle BCD=\angle DCA=x$, so $\angle B=\angle ACB=2x$. Triangle $BCD$ has $CB=CD$, so $\angle CDB=\angle CBD=2x$. Its angles give $x+2x+2x=180^\circ$, so $x=36^\circ$. Then $\angle ADC=180^\circ-\angle CDB=180^\circ-72^\circ=\boxed{108^\circ}$.

\begin{asy}
usepackage("amsmath");
size(3cm);
defaultpen(fontsize(10pt));
pair B=(0,0),C=(1,0),A=(0.5,0.5*Tan(72));pair D=B+2*Cos(72)*unit(A-B);
draw(A--B--C--cycle,linewidth(0.8));draw(C--D,linewidth(0.8));
label("$A$",A,N);label("$B$",B,SW);label("$C$",C,SE);label("$D$",D,W);
label("$2x$",B+0.27*dir(30),fontsize(7pt));label("$x$",C+0.3*dir(162),fontsize(6pt));label("$x$",C+0.3*dir(126),fontsize(6pt));label("$2x$",D+0.22*dir(300),fontsize(7pt));
\end{asy}
