The sine ratio gives $\sin 34^\circ=\dfrac{CB}{15}$, so $\boxed{CB=15\sin 34^\circ\approx 8.388}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(220);
defaultpen(fontsize(9pt));
pair A=(0,0),B=(15*cos(34*pi/180),15*sin(34*pi/180)),C=(15*cos(34*pi/180),0); draw(A--B--C--cycle); label("$A$",A,(-1,-1)); label("$B$",B,(1,-1)); label("$C$",C,(0,1)); label("$15$",(A+B)/2,(-1,1));label("$34^\circ$",A+(2,.4));
\end{asy}
