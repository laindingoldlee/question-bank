The third angle is $B=45^\circ$. Use the altitude foot $D=(0,0)$, with $A=(0,6\sqrt3)$, $C=(6,0)$, and $B=(-6\sqrt3,0)$, consistent with $AC=12$ and angles $C=60^\circ,B=45^\circ$. The slope of $AC$ is $-\sqrt3$, so the altitude from $B$ has slope $\dfrac{1}{\sqrt3}$. At $x=0$ it has height 6, hence its intersection $H$ with $AD$ satisfies $\boxed{DH=6}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,6*sqrt(3)),B=(-6*sqrt(3),0),C=(6,0),D=(0,0),H=(0,6);draw(A--B--C--cycle);draw(A--D,blue);draw(B--H,red);label("A",A,N);label("B",B,SW);label("C",C,SE);label("D",D,S);label("H",H,E);label("12",(A+C)/2,NE);
\end{asy}
