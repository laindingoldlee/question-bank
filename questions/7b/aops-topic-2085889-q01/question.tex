\begin{enumerate}
\item[(a)] Prove that for every odd positive integer \(a\),
\[
a^2+\left\lfloor\dfrac{a^2}{2}\right\rfloor^2=\left\lceil\dfrac{a^2}{2}\right\rceil^2.
\]
\item[(b)] Show that the identity fails for every even positive integer \(a\).
\end{enumerate}
