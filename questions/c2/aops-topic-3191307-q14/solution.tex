Use $A=(6,0)$, $B=(3,3\sqrt3)$, $D=(-6,0)$, $E=(-3,-3\sqrt3)$ and $O=(0,0)$. The shoelace sum gives area $\dfrac{1}{2}(18\sqrt3+18\sqrt3+18\sqrt3)=\boxed{27\sqrt3}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair[] H;for(int k=0;k<6;++k)H.push(6*dir(60*k));path boundary=H[0]; for(int j=1;j<H.length;++j)boundary=boundary--H[j];draw(boundary--cycle,gray);pair O=(0,0);filldraw(H[0]--H[1]--O--H[3]--H[4]--cycle,lightgray);string[] names={"A","B","C","D","E","F"};for(int k=0;k<6;++k)label("$"+names[k]+"$",H[k],dir(60*k));label("$O$",O,NE);
\end{asy}
