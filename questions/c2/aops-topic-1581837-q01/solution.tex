Let the smaller angle be $x$; the other is $2x$. Supplementary angles add to $180^\circ$, so $x+2x=180^\circ$ and $x=60^\circ$. The angles are $\boxed{60^\circ\text{ and }120^\circ}$.

\begin{asy}
usepackage("amsmath");
size(5cm);
defaultpen(fontsize(10pt));
pair O=(0,0);
draw((-3,0)--(3,0),linewidth(1));draw(O--3*dir(60),linewidth(1));
draw(arc(O,0.8,0,60),red);label("$60^\circ$",1.2*dir(30),red);
draw(arc(O,0.6,60,180),blue);label("$120^\circ$",1.05*dir(120),blue);
dot(O);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
