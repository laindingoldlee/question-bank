Let $\angle AOB=x$. Then $x+5x+4x+2x=180^\circ$, so $x=15^\circ$: $\boxed{\angle AOB=15^\circ,\ \angle BOC=75^\circ,\ \angle COD=60^\circ,\ \angle DOE=30^\circ}$.
\begin{asy}
usepackage("amsmath");
size(3cm);
defaultpen(fontsize(10pt));
draw(arc((0,0),1,0,180));draw((-1,0)--(1,0));pair O=(0,0);draw(O--dir(165));draw(O--dir(90));draw(O--dir(30));label("$A$",(-1,0),W);label("$E$",(1,0),E);label("$B$",dir(165),NW);label("$C$",dir(90),N);label("$D$",dir(30),NE);label("$O$",O,S);
\end{asy}
