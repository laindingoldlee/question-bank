Let $A=\{a_1,a_2,\dots,a_{100}\}$ and $B=\{1,2,\dots,50\}$.
\begin{enumerate}
\item[(a)] Determine the number of surjective functions $f:A\to B$ such that $f(a_1)\le f(a_2)\le\cdots\le f(a_{100})$.
\item[(b)] How many functions $f:A\to B$ with $f(a_1)\le f(a_2)\le\cdots\le f(a_{100})$ are there if $f$ need not be surjective?
\end{enumerate}
