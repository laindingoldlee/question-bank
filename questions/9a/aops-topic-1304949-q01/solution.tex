The square of the expression is
\[
x^2+\dfrac{1}{2}\left(x^2-3\right)^2=\dfrac{x^4-4x^2+9}{2}=\dfrac{\left(x^2-2\right)^2+5}{2},
\]
which is smallest when $x^2=2$, with value $\dfrac{5}{2}$. So the minimum is $\sqrt{\dfrac{5}{2}}=\boxed{\dfrac{\sqrt{10}}{2}}$, at $x=\pm\sqrt2$.

\begin{asy}
usepackage("amsmath");
size(4.5cm);
defaultpen(fontsize(10pt));
import graph; real f(real x){return (x^2-3)/sqrt(2);}
draw(graph(f,-2.6,2.6),blue); draw((-3,0)--(3,0),gray); draw((0,-2.5)--(0,2.6),gray);
pair P=(sqrt(2),f(sqrt(2))); draw((0,0)--P,red); dot(P); dot((0,0)); label("$O$",(0,0),NW); label("$(\sqrt2,-\dfrac{1}{\sqrt2})$",P,SE);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
