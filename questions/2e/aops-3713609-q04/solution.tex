A $30^\circ$--$60^\circ$--$90^\circ$ triangle has its shortest side half the hypotenuse, hence the required length is $\boxed5$.
\begin{asy}
import graph;import geometry;usepackage("amsmath");size(7cm);
pair A=(0, 5),B=(0, 0),C=(8.660254, 0);draw(A--B--C--cycle);label("$A$",A,(-1,1));label("$B$",B,(-1,-1));label("$C$",C,(1,-1));label("$5$",(0,2.5),(-1,0));label("$10$",(4.330127,2.5),(1,1));label("$30^\circ$",(6.8,.45));
void ra(pair U,pair V,pair W,real s){pair u=s*unit(U-V),w=s*unit(W-V);draw(V+u--V+u+w--V+w);}
ra(A,B,C,.4);
\end{asy}
