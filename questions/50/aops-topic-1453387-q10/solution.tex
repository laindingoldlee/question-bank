Completing the square, $x^2+(y-3)^2=25$, a circle with centre $(0,3)$ and radius $5$. Its circumference is $2\pi\cdot5=\boxed{10\pi}$.
\begin{asy}
usepackage("amsmath");
size(5cm);
defaultpen(fontsize(10pt));
import graph;
pair C=(0,3); real rr=5;
xaxis(-6.5,6.5,Arrow); yaxis(-3,9,Arrow);
draw(circle(C,rr),blue+linewidth(0.9));
draw(C--C+rr*dir(30),red); dot(C);
label("$(0,3)$",C,W); label("$5$",C+rr/2*dir(30),NW,red);
label("$x$",(6.5,0),S); label("$y$",(0,9),E);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
