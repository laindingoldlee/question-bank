Take triangle $BCD$ (half of the bottom face) as the base of the tetrahedron. Its area is $\dfrac{1}{2}\cdot12\cdot8=48$. The fourth vertex $H$ is directly above $D$, so the height is $DH=AE=6$.

\begin{asy}
usepackage("amsmath");
size(5.5cm);
defaultpen(fontsize(10pt));
pair A=(0.0000,0.0000),B=(12.0000,0.0000),C=(15.6000,2.5600),D=(3.6000,2.5600),E=(0.0000,6.0000),F=(12.0000,6.0000),G=(15.6000,8.5600),H=(3.6000,8.5600);
draw(A--B--F--E--cycle); draw(B--C--G--F); draw(E--H--G);
draw(A--D--C,dashed); draw(D--H,dashed);
filldraw(D--H--B--cycle,lightred+opacity(0.35),red+dashed);
draw(D--B,red+dashed); draw(H--B,red); draw(H--C,red); draw(B--C,red+linewidth(0.9)); draw(D--C,red+dashed); draw(D--H,red+dashed);
label("$A$",A,SW); label("$B$",B,SE); label("$C$",C,dir(-20)); label("$D$",D,NW); label("$E$",E,W); label("$F$",F,SE); label("$G$",G,NE); label("$H$",H,NW);
label("$12$",(A+B)/2,S); label("$8$",(B+C)/2,SE); label("$6$",(A+E)/2,W);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}

\[
V=\dfrac{1}{3}\cdot48\cdot6=\boxed{96}\text{ cubic units}.
\]
