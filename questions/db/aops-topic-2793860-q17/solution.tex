The line through $D$ perpendicular to $EF$ is perpendicular to $AB$, since $EF\parallel AB$. It is therefore the perpendicular bisector of $AB$. Similarly the other medial-triangle altitudes are perpendicular bisectors of the original sides. Their intersection $H$ is the circumcenter of $ABC$, so $HA=HB=HC$ and the requested difference is $\boxed0$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),B=(3,0),C=(4,4*sqrt(3)),D=(1.5,0),E0=(B+C)/2,F=C/2,H=(1.5,13/(2*sqrt(3)));draw(A--B--C--cycle);draw(D--E0--F--cycle,blue);draw(H--D,dashed);label("A",A,SW);label("B",B,S);label("C",C,N);label("D",D,S);label("E",E0,E);label("F",F,W);label("H",H,W);
\end{asy}
