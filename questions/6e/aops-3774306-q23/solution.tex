The trapezoid midsegment is parallel to the bases and has length equal to their arithmetic mean, $\dfrac{6+14}{2}=\boxed{10}$. This follows by joining a diagonal midpoint to each leg midpoint and applying the triangle midpoint theorem.
\begin{asy}
import graph;
usepackage("amsmath");
import geometry;
size(7cm);
pair[] p={(0,0),(14,0),(9,5),(3,5)}; pair labelCentre=(0,0);for(pair labelVertex:p)labelCentre+=labelVertex;labelCentre/=p.length; draw(p[0]--p[1]--p[2]--p[3]--cycle); dot(p[0]);label("$A$",p[0],unit(p[0]-labelCentre));dot(p[1]);label("$B$",p[1],unit(p[1]-labelCentre));dot(p[2]);label("$C$",p[2],unit(p[2]-labelCentre));dot(p[3]);label("$D$",p[3],unit(p[3]-labelCentre));draw((1.5,2.5)--(11.5,2.5));label("$M$",(1.5,2.5),(-1,0));label("$N$",(11.5,2.5),(1,0));
\end{asy}
