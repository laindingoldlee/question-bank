Let $M$ be the midpoint of $CD$, hence also of $AB$. Since $OC=OD$, $OM\perp AB$, and $CM=3$. Pythagoras in $OCM$ gives $OM^2=36-9=27$. Also $AM=9$, so the radius satisfies $OA^2=81+27=108$. Thus $OA=6\sqrt3$ and $\boxed{a+b=9}$.
\begin{asy}
import graph;import geometry;usepackage("amsmath");size(7cm);
pair O=(0,sqrt(27)),A=(-9,0),B=(9,0),C=(-3,0),D=(3,0),M=(0,0);draw(circle(O,sqrt(108)));draw(A--B);draw(O--A^^O--C^^O--D);draw(O--M,dashed);label("$O$",O,(0,1));label("$A$",A,(-1,-1));label("$B$",B,(1,-1));label("$C$",C,(0,-1));label("$D$",D,(0,-1));label("$M$",M,(0,-1));
void ra(pair U,pair V,pair W,real s){pair u=s*unit(U-V),w=s*unit(W-V);draw(V+u--V+u+w--V+w);}
ra(O,M,B,.6);
\end{asy}
