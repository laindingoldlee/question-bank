The horizontal base $AC$ has length $8-2=6$, and the perpendicular height from $B$ is $7-3=4$. The area is $\dfrac{1}{2}\cdot6\cdot4=\boxed{12}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");

defaultpen(fontsize(9pt));
pair A=(2,3),B=(6,7),C=(8,3),H=(6,3);draw(A--B--C--cycle);draw(B--H,dashed);label("$A(2,3)$",A,(-1,-1));label("$B(6,7)$",B,(0,1));label("$C(8,3)$",C,(1,-1));label("$4$",(B+H)/2,(1,0));label("$6$",(A+C)/2,(0,-1));
\end{asy}
