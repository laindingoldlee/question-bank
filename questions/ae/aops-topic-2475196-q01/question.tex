\begin{enumerate}
\item[(a)] Show that $37\times3k$ is the three-digit number $\overline{kkk}$ for each digit $k=1,2,\dots,9$ (for example $37\times27=999$ and $37\times18=666$).
\item[(b)] A student notices that $36\times37=1332$, that $1332+333=1665$ and that $1665\div3=555$. Use (a) to explain why the result is a three-digit number with equal digits.
\end{enumerate}
