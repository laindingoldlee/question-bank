$\pi r^2=961\pi$, so $r^2=961$ and $r=\boxed{31}$ (as $31^2=961$).

\begin{asy}
usepackage("amsmath");
size(3.5cm);
defaultpen(fontsize(10pt));
draw(circle((0,0),3),linewidth(0.8));
fill(circle((0,0),3),rgb(0.85,0.9,1)); draw(circle((0,0),3),linewidth(0.8));
draw((0,0)--3*dir(30),red+linewidth(0.9)); dot((0,0));
label("$r=31$",1.5*dir(30),NW,red); label("area $961\pi$",(0,-1.3),fontsize(8pt));
\end{asy}
