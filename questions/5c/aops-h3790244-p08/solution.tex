Triangle $XYZ$ has $\angle XYZ=180^\circ-20^\circ-35^\circ=125^\circ$. Opposite angles in a convex cyclic quadrilateral are supplementary, so $\angle XTZ=180^\circ-125^\circ=\boxed{55^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair X=dir(0),Y=dir(70),Z=dir(110),T=dir(235);draw(unitcircle);draw(X--Y--Z--T--cycle);draw(X--Z,dashed);label("X",X,E);label("Y",Y,N);label("Z",Z,NW);label("T",T,SW);
\end{asy}
