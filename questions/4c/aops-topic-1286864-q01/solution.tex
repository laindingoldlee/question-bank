Take $BE=BC=AE=1$.

\begin{asy}
usepackage("amsmath");
size(5cm);
defaultpen(fontsize(10pt));
pair B=(0,0); pair Ep=dir(0); pair A=2*cos(pi/9)*dir(20);
pair C2=dir(-20); draw(A--B--C2--cycle); draw(A--Ep--B,blue); draw(Ep--C2,blue);
label("$A$",A,NE); label("$B$",B,W); label("$C$",C2,S); label("$E$",Ep,E); dot(Ep);
label("$20^\circ$",B+0.42*dir(10),fontsize(7)); label("$20^\circ$",B+0.42*dir(-10),fontsize(7));
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}

\textbf{Triangle $EBC$:} it is isosceles with apex angle $20^\circ$, so $\angle BEC=\angle BCE=80^\circ$ and $EC=2\sin10^\circ$.

\textbf{Triangle $ABE$:} it is isosceles with $AE=BE$, so $\angle BAE=20^\circ$ and $\angle AEB=140^\circ$.

\textbf{Triangle $AEC$:} $\angle AEC=360^\circ-140^\circ-80^\circ=140^\circ$, so if $x=\angle CAE$ then $\angle ACE=40^\circ-x$. The law of sines gives
\[
\dfrac{\sin x}{\sin(40^\circ-x)}=\dfrac{EC}{AE}=2\sin10^\circ .
\]
The value $x=10^\circ$ works, since $\dfrac{\sin10^\circ}{\sin30^\circ}=2\sin10^\circ$. The left side increases strictly with $x$ on $(0^\circ,40^\circ)$, so the solution is unique, and $\angle CAE=\boxed{10^\circ}$.
