\textbf{Construction.} Let $AB$ be the given segment, $AB=1$.
\begin{enumerate}
\item Extend $AB$ beyond $B$ and draw the circle with centre $B$ and radius $AB$; it meets the extension at $C$, so $BC=1$ and $AC=2$.
\item Erect the perpendicular to line $AC$ at $A$: the circle with centre $A$ and radius $AB$ meets line $AC$ again at a point $P$ on the other side of $A$, so $AP=AB$; the circles with centres $P$ and $B$ and radius $PB$ meet at a point $X$ above the line; draw line $AX$.
\item The circle with centre $A$ and radius $AB$ meets line $AX$ at $D$, so $AD=1$.
\item Draw the segment $CD$.
\end{enumerate}

\begin{asy}
usepackage("amsmath");
size(5.5cm);
defaultpen(fontsize(10pt));
pair A=(0,0),B=(1,0),C=(2,0),D=(0,1),P=(-1,0),X=(0,sqrt(3));
draw((-1.4,0)--(2.4,0),gray); draw((0,-0.4)--(0,2.1),gray);
draw(A--B,linewidth(1.2)); draw(B--C,linewidth(1.2)); draw(C--D,red+linewidth(1));
draw(A--D,linewidth(1.2));
draw(arc(B,1,-20,20),dotted); draw(arc(A,1,160,200),dotted); draw(arc(A,1,75,105),dotted);
draw(arc(P,2,50,70),dotted); draw(arc(B,2,110,130),dotted);
draw((0,0)--(0.12,0)--(0.12,0.12)--(0,0.12));
dot(A);dot(B);dot(C);dot(D);dot(P);dot(X);
label("$A$",A,SW);label("$B$",B,S);label("$C$",C,SE);label("$D$",D,W);label("$P$",P,SW);label("$X$",X,W);
label("$1$",(A+B)/2,N); label("$1$",(B+C)/2,N); label("$1$",(A+D)/2,E); label("$\sqrt5$",(C+D)/2,NE,red);
\end{asy}

\textbf{Proof.} $X$ is equidistant from $P$ and $B$, and so is $A$; so line $AX$ is the perpendicular bisector of $PB$, and $AD\perp AC$. Triangle $CAD$ therefore has a right angle at $A$, with legs $AC=2$ and $AD=1$. By the Pythagorean theorem,
\[CD=\sqrt{2^2+1^2}=\boxed{\sqrt5}.\qquad\blacksquare\]
