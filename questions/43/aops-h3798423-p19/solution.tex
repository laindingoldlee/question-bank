Triangle angle sums give
\[\angle ABC-\angle ADC=(\angle CAD+\angle ACD)-(\angle BAC+\angle ACB)=14^\circ-10^\circ=\boxed{4^\circ}.\]
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,0),C=(5,0),B=extension(A,dir(40),C,C+dir(150)),D=extension(A,dir(-44),C,C+dir(210));draw(A--B--C--D--cycle);draw(A--C,dashed);label("$A$",A,(-1,0));label("$B$",B,(0,1));label("$C$",C,(1,0));label("$D$",D,(0,-1));
\end{asy}
