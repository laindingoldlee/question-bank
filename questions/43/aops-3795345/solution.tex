Use vector coordinates $A=0,B=u,D=v,C=u+v$, so $E=\dfrac{2}{3}u,F=\dfrac{2}{3}v$. On $BF$ write $K=(1-t)u+\dfrac{2}{3}tv$; on $DE$ write $K=\dfrac{2}{3}su+(1-s)v$. Equating coefficients gives $1-t=\dfrac{2}{3}s$ and $\dfrac{2}{3}t=1-s$, so $s=t=\dfrac{3}{5}$. Hence $K=\dfrac{2}{5}(u+v)=\dfrac{2}{5}C$, and $CK=\dfrac{3}{5}AC=\boxed6$.
\begin{asy}
import graph;
usepackage("amsmath");
import geometry;
size(7cm);
pair A=(0,0),B=(5,0),D=(1,3),C=B+D,E=2B/3,F=2D/3,K=2C/5;draw(A--B--C--D--cycle);draw(B--F);draw(D--E);draw(A--C,dashed);label("$A$",A,(-1,-1));label("$B$",B,(1,-1));label("$C$",C,(1,1));label("$D$",D,(-1,1));label("$E$",E,(0,-1));label("$F$",F,(-1,0));label("$K$",K,(1,-1));
\end{asy}
