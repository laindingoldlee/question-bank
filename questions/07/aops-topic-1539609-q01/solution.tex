Let $\alpha=\angle BAC$, $\beta=\angle ABC$, $\gamma=\angle ACB$.

\textbf{Where $D$ is.} If $D$ were on segment $AC$ we would have $CD\le CA<CB$; so $D$ lies on the ray beyond $C$, and $C$ is between $A$ and $D$:

\begin{asy}
usepackage("amsmath");
size(7.5cm);
defaultpen(fontsize(10pt));
pair A=(0,0),C=(3,0),B=(1.2,2.6);
real r=abs(B-C);pair D=C+(r,0);
draw(A--B--C--cycle,linewidth(0.9));draw(C--D,linewidth(0.9));draw(B--D,blue+linewidth(0.9));
draw(D--D+(0.6,0),dashed);
pair m1=(B+C)/2,m2=(C+D)/2;
pair u=unit(B-C),v=rotate(90)*u;draw(m1-0.1*v--m1+0.1*v);
draw(m2+(0,-0.1)--m2+(0,0.1));
label("$A$",A,SW);label("$B$",B,N);label("$C$",C,S);label("$D$",D,S);
draw(arc(D,0.75,180-degrees(atan2(B.y,D.x-B.x)),180),red);
draw(arc(B,0.4,degrees(D-B),degrees(A-B)),blue);
label("$\gamma$",C+0.35*dir(155),dir(155));
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}

\textbf{Angles.} Then $\angle BCD=180^\circ-\gamma$, and triangle $BCD$ is isosceles with $CB=CD$, so
\[
\angle CDB=\angle CBD=\dfrac{180^\circ-(180^\circ-\gamma)}{2}=\dfrac{\gamma}{2} .
\]
In triangle $ABD$,
\[
\angle ABD=180^\circ-\angle BAD-\angle ADB=180^\circ-\alpha-\dfrac{\gamma}{2} .
\]

\textbf{Using $BC>AC$.} In a triangle the larger side is opposite the larger angle; $BC$ is opposite $\alpha$ and $AC$ is opposite $\beta$, so $\alpha>\beta$. Hence
\[
2\alpha>\alpha+\beta=180^\circ-\gamma,\qquad\text{so}\qquad \alpha+\dfrac{\gamma}{2}>90^\circ ,
\]
and therefore $\angle ABD=180^\circ-\alpha-\dfrac{\gamma}{2}<90^\circ$. $\blacksquare$
