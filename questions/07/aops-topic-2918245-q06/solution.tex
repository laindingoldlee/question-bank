$\angle PAB=90^\circ-27^\circ=63^\circ$, and since $BA=BP$, $\angle BPA=63^\circ$ and $\angle ABP=54^\circ$. So $\angle OBC=90^\circ-54^\circ=36^\circ$, $\angle OCB=45^\circ$, and in triangle $BOC$, $\angle BOC=180^\circ-36^\circ-45^\circ=\boxed{99^\circ}$.
\begin{asy}
usepackage("amsmath");
size(3cm);
defaultpen(fontsize(10pt));
pair A=(0,0),B=(1,0),C=(1,1),D=(0,1);pair P=B+dir(126);pair O=extension(A,C,B,P);draw(A--B--C--D--cycle);draw(A--C);draw(B--P);draw(A--P,dashed);dot(O);label("$A$",A,SW);label("$B$",B,SE);label("$C$",C,NE);label("$D$",D,NW);label("$P$",P,N);label("$O$",O,E);
\end{asy}
