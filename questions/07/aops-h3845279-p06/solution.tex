If the rectangle sides are $a\ge b$, then $a+b=23$ and $a-b=7$, giving $a=15,b=8$. By Pythagoras, its diagonal is $\boxed{\sqrt{15^2+8^2}=17}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
fill(box((0,0),(15,8)),lightgray);fill(box((15,0),(23,15)),paleblue);fill(box((8,15),(23,23)),lightgray);fill(box((0,8),(8,23)),paleblue);draw(box((0,0),(15,8)));draw(box((15,0),(23,15)));draw(box((8,15),(23,23)));draw(box((0,8),(8,23)));draw((0,0)--(15,8),red);label("$23$",(11.5,-1),S);label("$7$",(11.5,11.5));
\end{asy}
