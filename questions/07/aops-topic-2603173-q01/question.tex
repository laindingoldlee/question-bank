Let $f(x)=\ln x$, $g(x)=e^x$ and $h(x)=x^2$ (real functions).
\begin{enumerate}
\item[(a)] Find and simplify $f(g(x))$ and $f(h(x))$, and give the domain of each.
\item[(b)] Find the value of $x$ for which $h(g(f(x)))=100$.
\end{enumerate}
