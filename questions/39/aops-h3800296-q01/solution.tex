Writing $T=[ABE]$ gives $[ABCD]=T+5T=6T$. But $T=\dfrac{1}{2} AB\cdot BE$ and $[ABCD]=AB\cdot BC$. Cancelling $AB$ gives $BC=3BE$, so $\boxed{\dfrac{BC}{BE}=3}$.

texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,0),B=(4,0),C=(4,3),D=(0,3),E=(4,1);filldraw(A--B--E--cycle,lightblue);draw(A--B--C--D--cycle);draw(A--E);label("$A$",A,(-1,-1));label("$B$",B,(1,-1));label("$C$",C,(1,1));label("$D$",D,(-1,1));label("$E$",E,(1,0));label("$T$",(A+B+E)/3);label("$5T$",(1.7,2));
\end{asy}
