Scale so $C=(0,0)$, $B=(1,0)$, $D=(3,0)$. Since $\angle C=45^\circ$, write $A=(t,t)$. Angle $B$ is $120^\circ$, so line $BA$ makes $60^\circ$ with the positive horizontal, giving $t=\sqrt3(t-1)$ and $t=\dfrac{3+\sqrt3}{2}$. Therefore $\tan\angle ADC=\dfrac{t}{3-t}=2+\sqrt3=\tan75^\circ$. Since the angle is acute, $\boxed{\angle ADC=75^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
real t=(3+sqrt(3))/2;pair C=(0,0),B=(1,0),D=(3,0),A=(t,t);draw(C--D--A--C);draw(A--B);label("A",A,N);label("B",B,S);label("C",C,SW);label("D",D,SE);label("1",(.5,0),S);label("2",(2,0),S);label("$45^\circ$",(.55,.2));label("$75^\circ$",(2.8,.35));
\end{asy}
