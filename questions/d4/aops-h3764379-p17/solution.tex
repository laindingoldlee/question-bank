If the square side is $s$ and the given circle radius is $R$, then $4s=2\pi R$. The inscribed radius is $r=\dfrac{s}{2}$, so $\dfrac{r}{R}=\dfrac{s}{2R}=\boxed{\dfrac{\pi}{4}}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
draw(box((-1,-1),(1,1)));draw(circle((0,0),1),blue);draw((0,0)--(1,0));label("$r$",(.5,0),S);draw(circle((4,0),4/pi),red);draw((4,0)--(4+4/pi,0));label("$R$",(4+2/pi,0),S);label("equal perimeters",(2,-2));
\end{asy}
