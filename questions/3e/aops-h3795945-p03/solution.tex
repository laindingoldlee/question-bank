A diagonal of an $m\times n$ grid passes through $m+n-\gcd(m,n)$ cell interiors: it crosses $m-1$ vertical and $n-1$ horizontal internal grid lines, with $\gcd(m,n)-1$ simultaneous vertex crossings counted twice. Here $\gcd(200,3)=1$, giving $200+3-1=\boxed{202}$ cells.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
for(int x=0;x<=8;++x)draw((x,0)--(x,3),gray);for(int y=0;y<=3;++y)draw((0,y)--(8,y),gray);draw((0,0)--(8,3),blue+1);label("8 by 3 illustration of grid crossings",(4,-.7));
\end{asy}
