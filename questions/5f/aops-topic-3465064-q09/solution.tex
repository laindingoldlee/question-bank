Use $B=(0,0),C=(6,0),A=(3,3\sqrt3)$. Then $M=(\dfrac{3}{2},\dfrac{3\sqrt3}{2})$, $N=(\dfrac{9}{2},\dfrac{3\sqrt3}{2})$, $D=(2,0),E=(4,0)$. Reflection symmetry puts $P$ on $x=3$. Along $DN$, moving 1 out of the total horizontal change $\dfrac{5}{2}$ gives height $\dfrac{2}{5}\dfrac{3\sqrt3}{2}=\dfrac{3\sqrt3}{5}$. Since $DE=2$, the area is $\boxed{\dfrac{3\sqrt3}{5}}$.
\begin{asy}
import geometry; usepackage("amsmath"); size(240);
pair B=(0,0),C=(6,0),A=(3,3*sqrt(3)),M=(A+B)/2,Npoint=(A+C)/2,D=(2,0),Epoint=(4,0),P=(3,3*sqrt(3)/5);draw(A--B--C--cycle);draw(D--Npoint^^Epoint--M);filldraw(D--P--Epoint--cycle,lightblue,blue);label("$A$",A,N);label("$B$",B,SW);label("$C$",C,SE);label("$M$",M,NW);label("$N$",Npoint,NE);label("$D$",D,S);label("$E$",Epoint,S);label("$P$",P,N);
\end{asy}
