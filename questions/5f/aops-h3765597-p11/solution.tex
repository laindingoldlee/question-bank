Write $\angle A=\alpha$ and scale $AB=1$. In right triangle $ABH$, $AH=\cos\alpha$ and $BH=\sin\alpha$. The condition gives $CH=1+\cos\alpha$, so $\tan C=\dfrac{\sin\alpha}{1+\cos\alpha}=\tan\dfrac{\alpha}{2}$. Both angles are acute, hence $C=\dfrac{\alpha}{2}$. Then $\alpha+\dfrac{\alpha}{2}+84^\circ=180^\circ$, giving $\boxed{\alpha=64^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
real a=64*pi/180;pair A=(0,0),H=(cos(a),0),B=(cos(a),sin(a)),C=(1+2*cos(a),0);draw(A--B--C--cycle);draw(B--H,dashed);label("A",A,SW);label("B",B,N);label("C",C,SE);label("H",H,S);label("1",(A+B)/2,W);
\end{asy}
