The parallel sides are $24$ apart, so $648=\dfrac{1}{2}(AB+DC)24$, giving $AB+DC=54$. In a tangential quadrilateral the two sums of opposite sides are equal, by pairing equal tangent segments from each vertex. Thus $AD+BC=54$ and $AD=54-25=\boxed{29}$.

\begin{asy}
import geometry;
size(240);
real u=sqrt(265),L=(61+u)/2;pair D=(0,0),C=(L,0),A=(u,24),B=(L-7,24),O=((29+u)/2,12);draw(A--B--C--D--cycle);draw(circle(O,12));draw(O--(O.x,0),dashed);label("$A$",A,NW);label("$B$",B,NE);label("$C$",C,SE);label("$D$",D,SW);label("$O$",O,N);label("$12$",(O+(O.x,0))/2,E);label("$25$",(B+C)/2,E);
\end{asy}
