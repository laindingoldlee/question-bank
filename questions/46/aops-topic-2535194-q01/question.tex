\begin{enumerate}
\item[(a)] For $x>0$, solve $2^y=x$ for $y$ in terms of $x$.
\item[(b)] A student, having forgotten about logarithms, proposes the formula $y=\dfrac{\sqrt{x+1}}{\sqrt2}+1$ for the solution of $2^y=x$. Show that this formula is wrong.
\end{enumerate}
