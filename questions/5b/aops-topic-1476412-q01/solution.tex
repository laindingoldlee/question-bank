For a positive integer $k$, multiply $\dfrac{1}{x}+\dfrac{1}{y}=\dfrac{1}{k}$ by $kxy$: $ky+kx=xy$, that is,
\[
xy-kx-ky+k^2=k^2\qquad\Longleftrightarrow\qquad (x-k)(y-k)=k^2 .
\]
Since $\dfrac{1}{x}<\dfrac{1}{k}$, we have $x>k$, and similarly $y>k$. So $x-k$ and $y-k$ are positive integers whose product is $k^2$, and each factorisation $k^2=uv$ gives the solution $(x,y)=(k+u,k+v)$.
\begin{enumerate}
\item[(a)] $k=4$, $16=1\cdot16=2\cdot8=4\cdot4=8\cdot2=16\cdot1$:
\[
(x,y)=\boxed{(5,20),\ (6,12),\ (8,8),\ (12,6),\ (20,5)} .
\]
\item[(b)] $k=3$, $9=1\cdot9=3\cdot3=9\cdot1$: $\boxed{(4,12),\ (6,6),\ (12,4)}$.
\item[(c)] $k=2$, $4=1\cdot4=2\cdot2=4\cdot1$: $\boxed{(3,6),\ (4,4),\ (6,3)}$.
\end{enumerate}
