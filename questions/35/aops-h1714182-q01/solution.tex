Factoring gives $(3k+2)(k-2)=0$, so $k=2$ or $k=-\dfrac{2}{3}$. At the $y$-axis, $x=0$ and $y=3k$. Therefore the two possible intercept values are $\boxed{6\text{ and }-2}$, corresponding to points $(0,6)$ and $(0,-2)$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
import graph;real f(real x){return 2*(x*x+3);}real g(real x){return -2.0/3*(x*x+3);}draw((-2,0)--(2,0),gray,Arrow);draw((0,-5)--(0,12),gray,Arrow);draw(graph(f,-1.5,1.5),blue);draw(graph(g,-1.5,1.5),red);dot((0,6));dot((0,-2));label("$6$",(0,6),SW);label("$-2$",(0,-2),NW);
\end{asy}
