The centers lie on the angle bisector. For radius $r$, the center is $\dfrac{r}{\sin30^\circ}=2r$ from the vertex. Thus the center separation is $10-2r$, while external tangency makes it $5+r$. Solving $10-2r=5+r$ gives $\boxed{r=\dfrac{5}{3}}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(220);
defaultpen(fontsize(9pt));
pair V=(0,0),O=(10,0),P=(10/3,0);draw(V--(16,16/sqrt(3)));draw(V--(16,-16/sqrt(3)));draw(circle(O,5));draw(circle(P,5/3));draw(V--O,dashed);label("$V$",V,(-1,0));label("$O$",O,(1,0));label("$P$",P,(-1,0));label("$5$",O+(0,2.5));label("$r$",P+(0,.8));
\end{asy}
