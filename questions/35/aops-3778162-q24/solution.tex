Triangle $AHC$ is right at $H$, and $M$ is the midpoint of its hypotenuse $AC$. The midpoint of a right triangle\textquotesingle s hypotenuse is equidistant from its three vertices, so $AC=2MH=\boxed{20}$ cm.

pair[] p={(0,0),(7,0),(4,5)}; pair labelCentre=(0,0);for(pair labelVertex:p)labelCentre+=labelVertex;labelCentre/=p.length; draw(p[0]--p[1]--p[2]--cycle); dot(p[0]);label("$A$",p[0],unit(p[0]-labelCentre));dot(p[1]);label("$B$",p[1],unit(p[1]-labelCentre));dot(p[2]);label("$C$",p[2],unit(p[2]-labelCentre));pair H=(4,0),M=(2,2.5);draw(p[2]--H);draw(p[1]--M);draw(H--M,dashed);label("$H$",H,(0,-1));label("$M$",M,(-1,1));
\end{asy}
