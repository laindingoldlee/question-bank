\begin{enumerate}
\item[(a)] Prove that if $p$ is a prime, then $(a+b)^p\equiv a^p+b^p\pmod p$ for all integers $a$ and $b$.
\item[(b)] Show that the congruence can fail when $p$ is not prime.
\end{enumerate}
