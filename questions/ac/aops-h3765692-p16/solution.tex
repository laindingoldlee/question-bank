The space diagonal of a cube of side 3 has length $\sqrt{3^2+3^2+3^2}=\boxed{3\sqrt3}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair O=(0,0),A=(3,0),B=(1,1),C=(0,3);draw(O--A--A+B--B--cycle);draw(O--C--C+A--A);draw(C--C+B--C+A+B--C+A);draw(A+B--A+B+C);draw(B--B+C,dashed);draw(O--A+B+C,red);label("X",O,SW);label("Y",A+B+C,NE);label("3",A/2,S);
\end{asy}
