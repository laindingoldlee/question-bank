Corresponding angles of similar triangles are equal, so $\angle D=30^\circ$ and $\sin D=\boxed{\dfrac12}$.

\begin{asy}
usepackage("amsmath");
size(4cm);
defaultpen(fontsize(10pt));
pair A=(0,0),B=(2,0),C=(2,2*Tan(30));draw(A--B--C--cycle,linewidth(0.8));label("$A$",A,W);label("$30^\circ$",A+(0.62,0.12),fontsize(6pt));
pair D=(3,0),E=(4.5,0),F=(4.5,1.5*Tan(30));draw(D--E--F--cycle,linewidth(0.8));label("$D$",D,W);label("$30^\circ$",D+(0.55,0.1),fontsize(6pt));
\end{asy}
