Reflection in the $y$-axis gives $B=(-a,b)$, and reflection in the $x$-axis then gives $C=(-a,-b)$. Hence $ef=(-a)(-b)=ab$, so $\boxed{ab-ef=0}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
draw((-3,0)--(3,0),Arrow);draw((0,-3)--(0,3),Arrow);pair A=(2,1),B=(-2,1),C=(-2,-1);dot(A);dot(B);dot(C);draw(A--B--C,dashed);label("A(a,b)",A,NE);label("B(-a,b)",B,NW);label("C(-a,-b)",C,SW);
\end{asy}
