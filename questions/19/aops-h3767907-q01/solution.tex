The angles sum to $180^\circ$, so the third is $180^\circ-47^\circ-55^\circ=\boxed{78^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,0),B=(4,0);real x=4*tan(55*pi/180)/(tan(47*pi/180)+tan(55*pi/180));pair C=(x,x*tan(47*pi/180));draw(A--B--C--cycle);label("A",A,SW);label("B",B,SE);label("C",C,N);label("$47^\circ$",(.65,.2));label("$55^\circ$",(3.3,.2));label("$78^\circ$",C-(0,.4));
\end{asy}
