Adjacent points among five equally spaced points subtend a central angle of $\dfrac{360^\circ}{5}=72^\circ$, so sector $AOB$ is $\dfrac15$ of its disc. Among eight points the angle is $45^\circ$, so sector $CPD$ is $\dfrac18$ of its disc. Each disc has area $400\pi$, so the difference is
\[\dfrac{400\pi}{5}-\dfrac{400\pi}{8}=80\pi-50\pi=\boxed{30\pi}.\]

\begin{asy}
usepackage("amsmath");
size(6.5cm);
defaultpen(fontsize(10pt));
pair O=(0,0),P=(5.2,0); real R=2;
draw(circle(O,R),linewidth(0.8)); draw(circle(P,R),linewidth(0.8));
fill(O--arc(O,R,90,162)--cycle,rgb(0.8,0.88,1)); fill(P--arc(P,R,90,135)--cycle,rgb(1,0.85,0.8));
for(int i=0;i<5;++i){draw(O--O+R*dir(90+72*i),gray); dot(O+R*dir(90+72*i));}
for(int i=0;i<8;++i){draw(P--P+R*dir(90+45*i),gray); dot(P+R*dir(90+45*i));}
draw(O--O+R*dir(90),linewidth(0.9)); draw(O--O+R*dir(162),linewidth(0.9));
draw(P--P+R*dir(90),linewidth(0.9)); draw(P--P+R*dir(135),linewidth(0.9));
label("$A$",O+R*dir(90),N); label("$B$",O+R*dir(162),W); label("$O$",O,SE);
label("$C$",P+R*dir(90),N); label("$D$",P+R*dir(135),NW); label("$P$",P,SE);
label("$72^\circ$",O+0.75*dir(126),fontsize(7pt)); label("$45^\circ$",P+0.95*dir(112),fontsize(7pt));
\end{asy}
