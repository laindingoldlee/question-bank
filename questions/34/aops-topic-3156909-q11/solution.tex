Let $ MT=2 $. With $ M=(-1,0),T=(1,0)$, the other points are $ L=(0,\sqrt3),A=(0,1),H=(0,-1)$. Thus $ HL=\sqrt3+1 $ and $ AL=\sqrt3-1 $, giving $\boxed{\dfrac{\sqrt3+1}{\sqrt3-1}=2+\sqrt3}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair M=(-1,0),T=(1,0),L=(0,sqrt(3)),A=(0,1),H=(0,-1);draw(L--M--T--cycle);draw(M--A--T--H--cycle,blue);draw(L--H,dashed);label("$M$",M,W);label("$T$",T,E);label("$L$",L,N);label("$A$",A,E);label("$H$",H,S);
\end{asy}
