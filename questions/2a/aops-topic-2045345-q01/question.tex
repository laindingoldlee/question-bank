\[
\begin{aligned}
&SO+MANY+MORE+MEN+SEEM+TO+SAY+THAT+THEY+MAY+SOON+TRY+TO+STAY+AT+HOME\\
&+SO+AS+TO+SEE+OR+HEAR+THE+SAME+ONE+MAN+TRY+TO+MEET+THE+TEAM+ON+THE\\
&+MOON+AS+HE+HAS+AT+THE+OTHER+TEN=TESTS.
\end{aligned}
\]
