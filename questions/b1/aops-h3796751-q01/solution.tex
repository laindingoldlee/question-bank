The right triangle $AHC$ has a $45^\circ$ angle, so $AH=CH=x$. Thus $HB=AB-AH=x=CH$. The right triangle $BHC$ is also isosceles, giving $\boxed{\angle B=45^\circ}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,0),B=(2,0),C=(1,1),H=(1,0);draw(A--B--C--cycle);draw(C--H,dashed);draw((1.1,0)--(1.1,.1)--(1,.1));label("A",A,SW);label("B",B,SE);label("C",C,N);label("H",H,S);label("$x$",(1,.5),E);label("$x$",(.5,0),S);label("$x$",(1.5,0),S);label("$45^\circ$",(.35,.12));
\end{asy}
