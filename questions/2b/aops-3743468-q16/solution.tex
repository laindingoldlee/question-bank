Write $y=a(x-20)(x-26)$. Its axis is $x=23$, where $y=-9a=9$, so $a=-1$. Hence $\boxed{y=-(x-20)(x-26)=9-(x-23)^2}$.
\begin{asy}
import graph;
usepackage("amsmath");
import geometry;
size(7cm);
draw((19,0)--(27,0));draw((23,-2)--(23,10));draw(graph(new real(real x){return 9-(x-23)^2;},20,26));label("20",(20,0),(0,-1));label("26",(26,0),(0,-1));label("$(23,9)$",(23,9),(0,1));
\end{asy}
