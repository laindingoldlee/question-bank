Set $A=(0,0),B=(6,0),C=(6,6)$, so $E=(-6,-6)$. If $F=(f,0)$ with $0\le f\le6$, then
\[
(f+6)^2+36=100,\qquad f+6=8.
\]
Thus $AF=2$ and $\boxed{BF=4}$.
\begin{asy}
texpreamble("\usepackage{amsmath,amssymb}");
size(240);
defaultpen(fontsize(9pt));
pair A=(0,0),B=(6,0),C=(6,6),D=(0,6);draw(A--B--C--D--cycle);label("$A$",A,(-1,-1));label("$B$",B,(1,-1));label("$C$",C,(1,1));label("$D$",D,(-1,1));pair E=(-6,-6),F=(2,0);draw(circle(E,10),gray);draw(E--C,dashed);draw(E--F,blue);dot(E);dot(F);label("$E$",E,(-1,-1));label("$F$",F,(0,-1));label("$10$",(E+F)/2,(0,-1));
\end{asy}
