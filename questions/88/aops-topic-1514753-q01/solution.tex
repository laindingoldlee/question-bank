\textbf{Classify the unit cubes} by how many faces of the box they touch.

\begin{asy}
usepackage("amsmath");
size(4.5cm);
defaultpen(fontsize(10pt));
int L=4,H=5,Dp=3; pair k=(0.5,0.38);
pair T(real x,real z){return (x,H)+z*k;}
pair R(real z,real y){return (L,y)+z*k;}
fill((0,0)--(L,0)--(L,H)--(0,H)--cycle,gray(0.93));
fill(T(0,0)--T(L,0)--T(L,Dp)--T(0,Dp)--cycle,gray(0.85));
fill(R(0,0)--R(Dp,0)--R(Dp,H)--R(0,H)--cycle,gray(0.78));
for(int i=1;i<L;++i){draw((i,0)--(i,H),gray(0.5));draw(T(i,0)--T(i,Dp),gray(0.5));}
for(int j=1;j<H;++j){draw((0,j)--(L,j),gray(0.5));draw(R(0,j)--R(Dp,j),gray(0.5));}
for(int z=1;z<Dp;++z){draw(T(0,z)--T(L,z),gray(0.5));draw(R(z,0)--R(z,H),gray(0.5));}
draw((0,0)--(L,0)--(L,H)--(0,H)--cycle,linewidth(1));
draw(T(0,0)--T(0,Dp)--T(L,Dp)--T(L,0),linewidth(1));
draw(R(0,0)--R(Dp,0)--R(Dp,H),linewidth(1));
label("$4$",(L/2,0),S);label("$5$",(0,H/2),W);label("$3$",R(Dp/2,0),SE);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}

\begin{itemize}
\item Interior cubes: $(3-2)(4-2)(5-2)=6$; they touch no face.
\item Cubes on exactly one face (face interiors): $2\big(1\cdot2+1\cdot3+2\cdot3\big)=22$.
\item Cubes on exactly two faces (edges without corners): $4\big((3-2)+(4-2)+(5-2)\big)=24$.
\item Corner cubes, on three faces: $8$.
\end{itemize}
(Check: $6+22+24+8=60$.)

\textbf{Probabilities.} A cube touching $j$ faces stays unpainted exactly when each of the four independent choices avoids those $j$ faces, which has probability $\left(\dfrac{6-j}{6}\right)^4$.

\textbf{Linearity of expectation.} The expected number of unpainted cubes is the sum of these probabilities over all cubes:
\[
6+22\left(\dfrac{5}{6}\right)^4+24\left(\dfrac{4}{6}\right)^4+8\left(\dfrac{3}{6}\right)^4=6+\dfrac{22\cdot625+24\cdot256+8\cdot81}{1296}=6+\dfrac{20542}{1296}=\dfrac{14159}{648}\approx21.85 .
\]
To the nearest whole number the answer is $\boxed{22}$.

(The poster's lower bound of $24$ is not right: only the $6$ interior cubes are certain to stay unpainted; every other cube can be painted.)
