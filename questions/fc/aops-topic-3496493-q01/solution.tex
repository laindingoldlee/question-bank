Since $5^2+12^2=13^2$, the area is $\dfrac{1}{2}\cdot5\cdot12=30$. For side length $b$, its altitude is $\dfrac{60}{b}$, minimized when $b=13$. Hence the shortest altitude is $\boxed{\dfrac{60}{13}}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),B=(12,0),C=(0,5),H=B+dot(A-B,C-B)/dot(C-B,C-B)*(C-B);draw(A--B--C--cycle);draw(A--H,dashed);label("$A$",A,SW);label("$B$",B,SE);label("$C$",C,N);label("$H$",H,NE);label("$12$",(6,0),S);label("$5$",(0,2.5),W);label("$13$",(6,2.5),N);
\end{asy}
