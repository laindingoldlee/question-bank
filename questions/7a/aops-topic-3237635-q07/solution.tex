Drop a perpendicular from $C$ to the line $AB$, meeting it at $D$. The $30$-$60$-$90$ triangle $ACD$ gives $CD=\dfrac{1}{2}AC=3\sqrt3$ and $AD=\dfrac{\sqrt3}{2}AC=9$. Since $AB=6<9$, $D$ lies beyond $B$, and $BD=3$. The right triangle $BCD$ has legs $3$ and $3\sqrt3$, so $\angle CBD=60^\circ$. Because $BA$ and $BD$ are opposite rays, $\boxed{\angle ABC=120^\circ}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
pair A=(0,0),B=(6,0),D=(9,0),C=(9,3*sqrt(3));draw(A--B--C--cycle);draw(B--D--C,dashed);draw(D+(-0.4,0)--D+(-0.4,0.4)--D+(0,0.4));label("$A$",A,SW);label("$B$",B,S);label("$D$",D,SE);label("$C$",C,N);label("$6$",(3,0),S);label("$3$",(7.5,0),S);label("$3\sqrt3$",(9,2.5),E);label("$30^\circ$",(1.7,0.45));
\end{asy}
