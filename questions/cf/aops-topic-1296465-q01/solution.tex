The centroid is the average of the vertices:
\[
\left(\dfrac{12+18+21}{3},\ \dfrac{37+64+43}{3}\right)=\boxed{(17,48)}.
\]

\begin{asy}
usepackage("amsmath");
size(3.5cm);
defaultpen(fontsize(10pt));
pair A=(12,37),B=(18,64),C=(21,43),G=(A+B+C)/3; draw(A--B--C--cycle,linewidth(0.9)); dot(G,red);
draw(A--(B+C)/2,gray+dashed); draw(B--(A+C)/2,gray+dashed); label("$(12,37)$",A,SW); label("$(18,64)$",B,N); label("$(21,43)$",C,SE); label("$G$",G,E,red);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}
