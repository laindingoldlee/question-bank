Prove that for every positive integer $n$,
\[
(1+2+3+\cdots+n)+\bigl(1+2+3+\cdots+(n-1)\bigr)=n^2,
\]
\begin{enumerate}
\item[(a)] algebraically, using $1+2+\cdots+k=\dfrac{k(k+1)}{2}$;
\item[(b)] with a picture.
\end{enumerate}
