Let $a$ and $b$ be positive integers with $b>a$, and write $b=ax+y$ with integers $x\ge1$ and $0\le y<a$ (division with remainder).
\begin{enumerate}
\item[(a)] Prove that
\[
\dfrac {a}{b}=\dfrac{1}{x+1}+\dfrac{a-y}{b(x+1)}.
\]
\item[(b)] Show that if $y\ge1$, the numerator $a-y$ of the second fraction satisfies $1\le a-y<a$.
\end{enumerate}
