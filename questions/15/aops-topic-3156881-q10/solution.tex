The rectangle sides are $ L,L+s $, so $ L(L+s)=60 $ and $ s^2=24 $. Its diagonal squared is $ L^2+(L+s)^2=2L(L+s)+s^2=120+24=144 $. Thus its diagonal is $\boxed{12}$.
\begin{asy}
usepackage("amsmath");
size(7cm);
real s=sqrt(24),L=(-s+sqrt(s*s+240))/2;draw((0,0)--(L+s,0)--(L+s,L)--(0,L)--cycle);draw((L,0)--(L,L));draw((L,s)--(L+s,s));draw((0,0)--(L+s,L),blue);label("$L$",(0,L/2),W);label("$s$",(L+s/2,0),S);
\end{asy}
