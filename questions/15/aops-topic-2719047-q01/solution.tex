The volume is $18\cdot9\cdot15=2430\text{ dm}^3=2430$ litres, which fills $\dfrac{2430}{0.5}=4860$ bottles. The revenue is $4860\cdot\$2=\boxed{\$9720}$.

\begin{asy}
usepackage("amsmath");
size(4cm);
defaultpen(fontsize(10pt));
real L=18,H=15;pair d=(5,3.5);
draw((0,0)--(L,0)--(L,H)--(0,H)--cycle,linewidth(0.8));
draw((L,0)--(L,0)+d--(L,H)+d--(0,H)+d--(0,H),linewidth(0.8));draw((L,H)--(L,H)+d,linewidth(0.8));
draw((0,0)--d--(L,0)+d,dashed);draw(d--(0,H)+d,dashed);
label("$18$ dm",(L/2,0),S,fontsize(8pt));label("$15$ dm",(0,H/2),W,fontsize(8pt));label("$9$ dm",(L,0)+d/2,SE,fontsize(8pt));
\end{asy}
