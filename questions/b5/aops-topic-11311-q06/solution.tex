Compute the values needed, from the top down.
\begin{itemize}
\item $f(6)=4$, $f(7)=5$, $f(8)=6$.
\item $f(5)=5-f(f(8))=5-f(6)=5-4=1$.
\item $f(4)=4-f(f(7))=4-f(5)=4-1=3$.
\item $f(3)=3-f(f(6))=3-f(4)=3-3=0$.
\item $f(1)=1-f(f(4))=1-f(3)=1-0=\boxed{1}$.
\end{itemize}
