\begin{enumerate}
\item[(a)] Let $m_1,\dots,m_k$ be pairwise relatively prime positive integers and $M=m_1m_2\cdots m_k$. Prove the Chinese remainder theorem: for any integers $a_1,\dots,a_k$ there is an integer $x$ with $x\equiv a_i\pmod{m_i}$ for every $i$, and the solutions form exactly one residue class modulo $M$.
\item[(b)] (Sunzi's problem.) Find the smallest positive integer that leaves remainder $2$ when divided by $3$, remainder $3$ when divided by $5$, and remainder $2$ when divided by $7$.
\end{enumerate}
