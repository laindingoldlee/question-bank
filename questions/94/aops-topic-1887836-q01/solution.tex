Write $[\,\cdot\,]$ for area.

\textbf{Construction.}
\begin{enumerate}
\item Through $C$ draw the line parallel to $BD$; let it meet ray $AB$ at $C'$.
\item Through $D$ draw the line parallel to $XC'$; let it meet ray $AB$ at $Y$.
\end{enumerate}

\begin{asy}
usepackage("amsmath");
size(8cm);
defaultpen(fontsize(10pt));
pair A=(0,0),B=(4,0),C=(5,3),D=(1,3.5);
pair X=0.75*D;
pair Cp=(53/7,0);
real s=3.5/(0.75*3.5);
pair Y=D+s*(Cp-X);
fill(A--X--Y--cycle,rgb(1,0.95,0.75));
draw(A--Y,gray);
draw(B--D,blue+dashed);draw(C--Cp,blue+dashed);
draw(X--Cp,red+dashed);draw(D--Y,red+dashed);
draw(A--B--C--D--cycle,linewidth(1));
dot(A);dot(B);dot(C);dot(D);dot(X);dot(Cp);dot(Y);
label("$A$",A,SW);label("$B$",B,S);label("$C$",C,NE);label("$D$",D,N);label("$X$",X,W);
label("$C'$",Cp,S);label("$Y$",Y,S);
label("\mbox{}",(min(currentpicture,user=true)+max(currentpicture,user=true))/2,black); // keeps the text after the figure black when LaTeX typesets the labels inline
\end{asy}

\textbf{Step 1: $[AC'D]=[ABCD]$.} As $ABCD$ is convex, $A$ and $C$ are on opposite sides of line $BD$, and the parallel to $BD$ through $C$ meets ray $AB$ beyond $B$. Triangles $BCD$ and $BC'D$ have the common base $BD$, and $C$, $C'$ are at the same distance from $BD$ because $CC'\parallel BD$. So $[BC'D]=[BCD]$ and
\[
[AC'D]=[ABD]+[BC'D]=[ABD]+[BCD]=[ABCD].
\]

\textbf{Step 2: $[AXY]=[AC'D]$.} Since $XC'\parallel DY$, triangles $AXC'$ and $ADY$ are similar, so $\dfrac{AY}{AC'}=\dfrac{AD}{AX}$, i.e.\ $AY=\dfrac{AC'\cdot AD}{AX}$. With $\alpha=\angle DAB$,
\[
[AXY]=\dfrac{1}{2}\,AX\cdot AY\sin\alpha=\dfrac{1}{2}\,AX\cdot\dfrac{AC'\cdot AD}{AX}\sin\alpha=\dfrac{1}{2}\,AD\cdot AC'\sin\alpha=[ADC'].
\]
Together, $[AXY]=[ABCD]$, as required.

\textbf{$Y$ is beyond $B$.} If $Y$ were on segment $AB$, then $AX\le AD$ and $AY\le AB$ would give $[AXY]\le\dfrac{1}{2}AD\cdot AB\sin\alpha=[ABD]<[ABCD]$. So $Y$ never lies on the segment $AB$; it lies on ray $AB$ beyond $B$ (indeed $AY\ge AC'>AB$).
